\subsection{The claim, verbatim}
\label{sec:claimverbatim}

The object of this project is the following assertion of Schouten--Isachenkov
\cite{Schouten:2025dssyk} (arXiv:2512.10101v2; all quotations and equation numbers refer to v2;
the full quotation ledger with locations is \texttt{lit/notes/2512.10101.md}):

\begin{quote}
``Lastly, we make remarks on the correlation function related to the strange series
representation, which is argued to interpolate between the AdS and dS regions of our
$q$-homogeneous space.'' \hfill (abstract)
\end{quote}
\begin{quote}
``such that the strange series has a non-zero amplitude between the lower part and upper part of
the spectrum~. It was proposed in [arXiv:2411.16922] and [arXiv:2505.08116] that zooming into the
upper part of the spectrum, i.e.\ $\theta=|\ln(q^2)|k$, corresponds to de Sitter JT-gravity.
If this proposal were proven to be correct, the strange series would therefore have a non-zero
amplitude between the anti-de Sitter and de Sitter regions.'' \hfill (\S4.3, end)
\end{quote}
\begin{quote}
``Moreover, supposing that dS space corresponds to the upper part of the spectrum, it was shown
in Section~4.3 that the strange series gives non-zero amplitudes between AdS and dS regions, and
thus in some sense interpolates between them. It is currently unknown what the correct physics
interpretation of such an amplitude would be.'' \hfill (\S5, Discussion, item 1)
\end{quote}

The authors' own epistemic labelling is therefore: \emph{argued}, twice conditional --- on the
dS identification of the upper spectral edge, and on an interpretation of the amplitude that they
declare unknown. The phrase ``in some sense interpolates'' is theirs; making the sense precise is
the purpose of this section.

\subsection{Setting and the paper's computation}
\label{sec:setting}

Throughout, conventions are those of Section~\ref{sec:conventions}. The DSSYK spectral parameter
is $\theta\in[0,\pi]$, $E(\theta)=2\cos\theta/\sqrt{1-q^{2}}$ (Convention~\ref{conv:q}); the
principal-series eigenvectors $\ket{\P^{\theta}}$ and the chord states $\ket{k}$ are as in
Conventions \ref{conv:casimir} and \ref{conv:chord}.

\begin{definition}[edge regimes]\label{def:edges}
For $k\ge0$ fixed and $q\to1^{-}$, the \emph{lower-edge regime} is the family
$\theta^{L}_{q}(k)\coloneqq\pi-|\ln q^{2}|\,k$ and the \emph{upper-edge regime} is
$\theta^{U}_{q}(k)\coloneqq|\ln q^{2}|\,k$.
\end{definition}

\begin{definition}[AdS region]\label{def:ads}
The \emph{AdS region} is the lower-edge regime. Justification (not a convention): in this regime
the $q\to1$ limit of DSSYK is the Schwarzian theory on AdS$_2$ [SI \S4.2, eq.\ (4.17); standard
DSSYK literature \cite{Berkooz:2018largeN}], and every bulk-dual proposal on the market agrees on
this identification.
\end{definition}

\begin{definition}[dS region --- two inequivalent candidates]\label{def:ds}
\emph{(a) Edge-dS (dS-JT):} the upper-edge regime $\theta^{U}_{q}(k)$
[arXiv:2411.16922 \cite{Blommaert:2024entropic}; arXiv:2505.08116 \cite{Okuyama:2025dsjt}].
\emph{(b) Centre-dS:} the middle of the band, $\theta=\pi/2$, $E=0$
[arXiv:2310.16994 \cite{Narovlansky:2023lfz} and successors].
The claim of \cite{Schouten:2025dssyk} is formulated with (a); its footnote records (b) as a
rival. Any statement below that depends on the choice says so explicitly.
\end{definition}

The paper's strange-series computation concerns the bilocal operators of
Convention~\ref{conv:chord},
$\O^{\ell}$ (discrete series) and $\mathcal{S}^{\ell}$ (strange series), $\ell\in\tfrac12\Z_{>0}$,
through their equal-time overlap kernels
\begin{equation}\label{eq:kernels}
  \K^{\O}_{q}(\theta_{1},\theta_{2})
  \coloneqq\bra{\P^{\theta_1}}\O^{\ell}(0)\ket{\P^{\theta_2}}
  =\frac{(q^{4\ell};q^{2})_{\infty}}{(q^{2\ell}e^{i(\pm\theta_1\pm\theta_2)};q^{2})_{\infty}},
  \qquad
  \K^{\mathcal{S}}_{q}(\theta_{1},\theta_{2})
  \coloneqq\bra{\P^{\theta_1}}\mathcal{S}^{\ell}(0)\ket{\P^{\theta_2}},
\end{equation}
where the denominator is the product over the four sign choices [SI (4.15)--(4.16), (4.19)].

\begin{lemma}[$q$-Mehler form; chord-side characterisation]\label{lem:mehler}
\DERIVED\ For $t\in\C$, $|t|<1$,
\begin{equation}\label{eq:qmehler}
  \sum_{k\ge0}\frac{t^{k}}{(q^{2};q^{2})_{k}}\,H_{k}(\cos\theta_1;q^{2})H_{k}(\cos\theta_2;q^{2})
  =\frac{(t^{2};q^{2})_{\infty}}{(t\,e^{i(\pm\theta_1\pm\theta_2)};q^{2})_{\infty}},
\end{equation}
the $q$-Mehler formula for continuous $q$-Hermite polynomials
(Gasper--Rahman; \cite{Berkooz:2018largeN} eq.\ (2.28)). Consequently
$\K^{\O}_{q}$ is \eqref{eq:qmehler} at $t=q^{2\ell}$, and
$\K^{\mathcal{S}}_{q}$ is \eqref{eq:qmehler} at $t=-q^{2\ell}$:
the strange bilocal inserts matter with \emph{alternating chord weights}
$(-1)^{k}q^{2k\ell}$, versus $q^{2k\ell}$ for the discrete bilocal. In particular
$\K^{\mathcal{S}}_{q}(\theta_1,\theta_2)
=(q^{4\ell};q^{2})_{\infty}/(-q^{2\ell}e^{i(\pm\theta_1\pm\theta_2)};q^{2})_{\infty}$,
which is SI (4.19) before its limit.
\end{lemma}

\begin{proof}
$\hat\rho_{st}\ket{k}=\tfrac12 q^{-1}q^{-2k}\ket{k}$ [SI (4.13)] gives
$(2q\hat\rho_{st})^{-\ell-\frac{i\pi}{2\ln q}}\ket{k}
=q^{2k(\ell+\frac{i\pi}{2\ln q})}\ket{k}=(-1)^{k}q^{2k\ell}\ket{k}$, since
$q^{2k\cdot i\pi/(2\ln q)}=e^{i\pi k}$. Expanding both $\ket{\P^{\theta_i}}$ in chord states
(Convention~\ref{conv:casimir}) yields the left side of \eqref{eq:qmehler} with $t=\mp q^{2\ell}$;
\eqref{eq:qmehler} itself is classical.
\end{proof}

\begin{lemma}[exact reflection identity]\label{lem:reflection}
\DERIVED\ For all $q\in(0,1)$, $\ell\in\tfrac12\Z_{>0}$ and $\theta_{1},\theta_{2}$,
\begin{equation}\label{eq:reflection}
  \K^{\mathcal{S}}_{q}(\theta_{1},\theta_{2})
  =\K^{\O}_{q}(\theta_{1},\,\pi-\theta_{2})
  =\K^{\O}_{q}(\pi-\theta_{1},\,\theta_{2}).
\end{equation}
That is: at finite $q$, the strange-series equal-time kernel is the discrete-series kernel with
one spectral argument reflected through the middle of the band ($E\to-E$).
\end{lemma}

\begin{proof}
$e^{\pm i(\pi-\theta_2)}=-e^{\mp i\theta_2}$, so the four denominator arguments
$-q^{2\ell}e^{i(\pm\theta_1\pm\theta_2)}$ of $\K^{\mathcal{S}}_{q}(\theta_1,\theta_2)$
(Lemma~\ref{lem:mehler}) coincide, as a set, with the four arguments
$q^{2\ell}e^{i(\pm\theta_1\pm(\pi-\theta_2))}$ of
$\K^{\O}_{q}(\theta_1,\pi-\theta_2)$; the numerators are equal. The second equality holds because
the kernel depends on $(\theta_1,\theta_2)$ only through the four sums $\pm\theta_1\pm\theta_2$,
and reflecting both arguments returns the original kernel:
$\K^{\mathcal{S}}$ and $\K^{\O}$ are each invariant under simultaneous reflection.
\end{proof}

\begin{proposition}[the paper's limit statements, re-derived]\label{prop:limits}
\DERIVED\ Fix $\ell\in\tfrac12\Z_{>0}$ and $k_{1},k_{2}\ge0$. As $q\to1^{-}$:
\begin{enumerate}[label=(\roman*)]
\item \emph{Same edge, discrete (Schwarzian benchmark):}
$\K^{\O}_{q}(\theta^{L}_{q}(k_1),\theta^{L}_{q}(k_2))
\sim N(q)\,\dfrac{\Gamma(\ell\pm ik_1\pm ik_2)}{\Gamma(2\ell)}$,
where $\Gamma(\ell\pm ik_1\pm ik_2)$ denotes the product over the four sign choices and
$N(q)\coloneqq(1-q^{2})^{2\ell-3}(q^{2};q^{2})_{\infty}^{-3}$
[SI (4.17); the $q$-gamma limit $\Gamma_{q^2}\to\Gamma$].
\item \emph{Same edge, strange:} the normalisation-free ratio vanishes,
$\K^{\mathcal{S}}_{q}(\theta^{L}_{q}(k_1),\theta^{L}_{q}(k_2))\,/\,
 \K^{\O}_{q}(\theta^{L}_{q}(k_1),\theta^{L}_{q}(k_2))\longrightarrow 0$
[this is the content of SI (4.19)].
\item \emph{Opposite edges, strange:} exactly, at every $q$,
$\K^{\mathcal{S}}_{q}(\theta^{L}_{q}(k_1),\theta^{U}_{q}(k_2))
=\K^{\O}_{q}(\theta^{L}_{q}(k_1),\theta^{L}_{q}(k_2))$,
hence with the same normalisation $N(q)$ as in (i) the limit is the Schwarzian $3j$-symbol
$\Gamma(\ell\pm ik_1\pm ik_2)/\Gamma(2\ell)$ [this is SI (4.20), sharpened to an identity].
\item \emph{Opposite edges, discrete:} the ratio
$\K^{\O}_{q}(\theta^{L}_{q}(k_1),\theta^{U}_{q}(k_2))\,/\,
 \K^{\O}_{q}(\theta^{L}_{q}(k_1),\theta^{L}_{q}(k_2))\longrightarrow 0$:
ordinary (discrete-series) matter does \emph{not} connect the two edges.
\end{enumerate}
\end{proposition}

\begin{proof}
$\theta^{L}_{q}(k)$ gives $e^{i\theta^{L}_{q}(k)}=-q^{2ik}$ and $\theta^{U}_{q}(k)$ gives
$e^{i\theta^{U}_{q}(k)}=q^{-2ik}$. (iii) is then Lemma~\ref{lem:reflection} evaluated at
$\pi-\theta^{U}_{q}(k_2)=\theta^{L}_{q}(k_2)$ --- the reflection is exact, no limit needed ---
combined with (i). For (i): each denominator factor is
$(q^{2(\ell\pm ik_1\pm ik_2)};q^{2})_{\infty}$, and
$(q^{2x};q^{2})_{\infty}=(1-q^{2})^{1-x}(q^{2};q^{2})_{\infty}/\Gamma_{q^{2}}(x)$
[SI (A.3)] with $\Gamma_{q^{2}}(x)\to\Gamma(x)$ gives the stated asymptotics; the imaginary
parts cancel in the accumulated exponent of $(1-q^{2})$. For (ii) and (iv): by
Lemma~\ref{lem:reflection} the two ratios are equal, so it suffices to prove (iv). By (i)'s
factorisation, the ratio in (iv) is
$\prod_{\pm\pm}(q^{2(\ell\pm ik_1\pm ik_2)};q^{2})_{\infty}
/\prod_{\pm\pm}(-q^{2(\ell\pm ik_1\pm ik_2)};q^{2})_{\infty}$.
For fixed $x$ with $\mathrm{Re}\,x=\ell>0$,
$\bigl|(-q^{2x};q^{2})_{\infty}\bigr|=\prod_{n\ge0}\bigl|1+q^{2x+2n}\bigr|
\ge\prod_{0\le n\le N(q)}\bigl(1+\tfrac12 q^{2\ell}\bigr)\to\infty$
as $q\to1$, where $N(q)\coloneqq\lfloor |2\ln q|^{-1}\ln 2\rfloor\to\infty$ counts the factors
with $q^{2n}\ge\tfrac12$ (and $|1+q^{2x+2n}|\ge 1+\cos(2k\ln q\cdots)q^{2\ell+2n}\ge
1-q^{2\ell+2n}>0$ keeps all other factors bounded below away from $0$ uniformly in $q$ near
$1$; the phases $q^{\pm2ik_i}\to1$), while the numerator tends to the finite nonzero limit
$(1-q^2)$-renormalised in (i). A cleaner route to the same conclusion: the ratio equals
$\prod_{\pm\pm}\Gamma_{q^{2}}(\tfrac{i\pi}{2\ln q}+\ell\pm ik_1\pm ik_2)
/\Gamma_{q^{2}}(\ell\pm ik_1\pm ik_2)^{\vphantom{1}}$ up to a phase, and the half-period-shifted
$q$-gamma tends to $0$; the elementary product bound above is verified numerically in
\texttt{src/check\_claim\_kernels.py} (output \texttt{results/check\_claim\_kernels.out}: the
ratio falls to $10^{-21419}$ by $q=0.9999$, the renormalised mixed-edge strange kernel agrees
with $\Gamma(\ell\pm ik_1\pm ik_2)/\Gamma(2\ell)$ to $4.7\times10^{-4}$ relative at $q=0.9999$,
and the reflection identity holds to better than $9\times10^{-37}$ relative at all sampled $q$),
with further scans in issue~\#10. \emph{Note}: SI's printed (4.19) writes the shift as $i\pi/\ln q$ (the full period of
$\Gamma_{q^{2}}$, which would \emph{not} vanish); consistency with their own (4.18) and \S4.3
text requires $i\pi/(2\ln q)$; see \texttt{lit/notes/2512.10101.md} \S7.
\end{proof}

\subsection{Formalisations of the claim}
\label{sec:formalisations}

The informal claim admits inequivalent formalisations. We state all that we will use, from
weakest to strongest, and identify which one this project attacks.

\begin{claim}[C-amp: non-vanishing AdS$\leftrightarrow$dS amplitude; SI's literal claim]
\label{claim:camp}
Assume Definition~\ref{def:ds}(a) (edge-dS). Then the strange-series bilocals
$\mathcal{S}^{\ell}$, $\ell\in\tfrac12\Z_{>0}$, have equal-time overlap kernels between the AdS
region and the dS region with finite non-zero $q\to1$ limit (after the normalisation $N(q)$ that
renders the discrete-series AdS kernel finite), given by the Schwarzian $3j$-symbol; their
AdS$\leftrightarrow$AdS kernels vanish in the same normalisation; and for the discrete-series
bilocals the situation is reversed. \emph{Status:} \DERIVED\ --- this is
Proposition~\ref{prop:limits}\,(ii)--(iv), conditional \emph{only} on the meaning of ``dS''.
\end{claim}

\begin{claim}[C-geo: geometric interpolation; the open claim this project attacks]
\label{claim:cgeo}
There exists a radial Casimir problem on the quantum homogeneous space carrying the DSSYK
dynamics (issues \#5--\#7) --- a second-order $q$-difference operator in an invariant coordinate,
with an effective potential depending on the representation label --- together with a criterion,
fixed in advance (issue \#8), for when such a radial problem is of AdS$_2$ type and when of
dS$_2$ type; and under this criterion the strange series realises the dS$_2$ type, or crosses
from one type to the other as its label varies, while the representations with classical
analogues realise the AdS$_2$ type. \emph{Status:} \CONJ\ --- this is the precise reading of
``genuinely interpolate''; nothing in \cite{Schouten:2025dssyk} proves or refutes it, and the
authors call the interpretation of the amplitude ``currently unknown''.
\end{claim}

\begin{remark}[what would settle the question]\label{rem:settle}
Claim~\ref{claim:camp} is settled (Proposition~\ref{prop:limits}), \emph{given} its
conditionality; what remains open in it is exactly its hypothesis, Definition~\ref{def:ds}(a).
Claim~\ref{claim:cgeo} is open and falsifiable: computing the radial Casimir operator on the
space fixed in issue \#5, restricted to the strange series (issues \#6--\#7), and evaluating the
issue-\#8 criterion on it, either exhibits the AdS$\to$dS crossing (proving
Claim~\ref{claim:cgeo}), or exhibits an obstruction (refuting it). A specialist can check either
outcome against Sections \ref{sec:space}--\ref{sec:criterion}.
\end{remark}

\begin{remark}[dS-identification dependence]\label{rem:dsdependence}
\ARGUED\ Claim~\ref{claim:camp} is \emph{not} robust against replacing
Definition~\ref{def:ds}(a) by (b). At $\theta_2=\pi/2$ the four denominator arguments of
$\K^{\mathcal{S}}_{q}(\theta^{L}_{q}(k_1),\pi/2)$ tend to $\pm i$, and the standard asymptotics
$\ln(u;q^{2})_{\infty}\sim-\mathrm{Li}_2(u)/|\ln q^{2}|$ gives
\[
\K^{\mathcal{S}}_{q}(\theta^{L}_{q}(k_1),\pi/2)\,/\,
\K^{\O}_{q}(\theta^{L}_{q}(k_1),\theta^{L}_{q}(k_2))
\;\sim\;\exp\!\Bigl(-\frac{3\pi^{2}}{4\,|\ln q^{2}|}\Bigr)\longrightarrow 0
\]
(using $\mathrm{Li}_2(i)+\mathrm{Li}_2(-i)=-\pi^{2}/24$, $\mathrm{Li}_2(1)=\pi^{2}/6$):
the strange amplitude between the AdS edge and the band centre is exponentially small in the
same normalisation in which the edge-to-edge amplitude is finite. Numerical check (issue \#10
scans pending): the ratio is $5.6\times10^{-11}$ at $q=0.9$, $1.8\times10^{-153}$ at $q=0.99$,
$1.3\times10^{-313}$ at $q=0.995$; the measured rate $|\ln q^{2}|\cdot\ln(\mathrm{ratio})$ is
$-4.97,\,-7.07,\,-7.22$ against the predicted $-3\pi^{2}/4=-7.40$ (subleading corrections are
$O(|\ln q^{2}|\ln|\ln q^{2}|)$ in the exponent). So if the centre-dS identification
(b) is the correct one, the strange series does \emph{not} connect AdS to dS in the amplitude
sense --- the claim's truth value depends on which dS proposal one adopts. This
criterion-sensitivity is exactly the situation flagged in issue \#8 (``if the criterion is
ambiguous, that is itself the finding''); the quantitative treatment belongs to \#8/\#11.
\end{remark}

\begin{remark}[discreteness of the strange family]\label{rem:discreteness}
``Interpolate'' suggests a continuous deformation. Three parameters could supply it:
(1) the abstract strange label $a>0$, continuous in the classification
(Convention~\ref{conv:series}) but \emph{discrete} ($a\in\tfrac12\Z_{>0}$) in the Plancherel
decomposition of the regular representation (Convention~\ref{conv:plancherel}) --- and the
bilocal construction of SI uses $\ell\in\tfrac12\Z_{>0}$ because integer powers of the spherical
element are what exists in the coordinate algebra (their footnote on the universal cover for
continuous weights applies here too);
(2) the spectral parameter $\theta$ (continuous; this is the parameter in
Claim~\ref{claim:camp});
(3) the radial coordinate of the homogeneous space (continuous $q$-lattice; this is the
parameter in Claim~\ref{claim:cgeo}).
Any proof or refutation of Claim~\ref{claim:cgeo} must say against which parameter the
``crossing'' happens; a formalisation in which no continuous parameter crosses anything is not
an interpolation statement but a two-point statement, and must be reported as such.
\end{remark}

\begin{remark}[degenerate readings, excluded]\label{rem:degenerate}
Two readings are excluded as contentless. (a) ``The strange series are the negative-$\tCas$
counterparts of the discrete series'' is true by construction (Convention~\ref{conv:series});
without an independent geometric criterion it says nothing about AdS or dS. (b) Given
Definition~\ref{def:ds}(a) and Proposition~\ref{prop:limits}, calling the non-zero mixed-edge
kernel ``interpolation'' is definitionally true; the content resides in the hypothesis (a) and in
the \emph{selectivity} of the property --- among the bilocal families computed so far (discrete,
strange) exactly the strange one connects the edges. Whether the selectivity survives the
inclusion of principal-series bilocals is, to our knowledge, not computed anywhere; we record it
as an open sub-question for issue \#6.
\end{remark}

\subsection{The normaliser origin of the reflection identity}
\label{sec:normaliserorigin}

% Issue #15. Connects Lemma \ref{lem:reflection} to the Z_2-grading of the dual von Neumann
% algebra of GKK; establishes the fusion of strange and discrete series inside single
% irreducible corepresentations of the normaliser quantum group; and derives the covariance
% obstruction that informs issue #8. Numerical verification:
% src/check_normaliser_grading.py (results/check_normaliser_grading.out, 72/72 checks pass;
% cited as (G1)--(G7) below).

Lemma~\ref{lem:reflection} and the normaliser structure of
Convention~\ref{conv:normaliser} are two occurrences of one mechanism. This subsection makes
the connection precise: the extra factor distinguishing the strange bilocal from the discrete
bilocal is a self-adjoint unitary $\Pi$ that anticommutes with the transfer matrix ---
the chord-sector shadow of the $\Z_2$-grading of the dual von Neumann algebra of
\cite{Groenevelt:2009gkk} --- and at the unrescaled level the strange and discrete series are
\emph{fused into single irreducible corepresentations} of $\SUq\rtimes\Z_2$. The structural
consequence (Theorem~\ref{thm:covobstruction}) is an obstruction relevant to issue \#8.

\begin{lemma}[the grading unitary on the chord sector]\label{lem:gradingchord}
\PROVED\ Let $q\in(0,1)$, let $H_{\mathrm{ch}}$ be the closure of the span of the chord states
$\{\ket{k}\}_{k\in\Z_{\ge0}}$ (an orthonormal basis, Convention~\ref{conv:chord}, boundary
conditions $\eta=\xi=1$), and let $2q\hat\rho_{st}$ be the positive self-adjoint diagonal
operator $2q\hat\rho_{st}\ket{k}=q^{-2k}\ket{k}$ [SI (4.13)]. Define, by the Borel functional
calculus with the real branch of $\ln$ on $(0,\infty)$,
\begin{equation}\label{eq:Pidef}
  \Pi \coloneqq \bigl(2q\hat\rho_{st}\bigr)^{-\frac{i\pi}{2\ln q}},
  \qquad z^{-\frac{i\pi}{2\ln q}}\coloneqq\exp\Bigl(-\frac{i\pi\ln z}{2\ln q}\Bigr).
\end{equation}
Then:
\begin{enumerate}[label=(\roman*)]
\item $\Pi\ket{k}=(-1)^{k}\ket{k}$; consequently $\Pi=\Pi^{*}=\Pi^{-1}$, $\Pi^{2}=1$:
$\Pi$ is the chord-parity involution, the nontrivial character of the spectral lattice
$q^{-2\Z_{\ge0}}$ of $2q\hat\rho_{st}$.
\item $\Pi\,\tCas\,\Pi=-\tCas$ on $H_{\mathrm{ch}}$, hence $\Pi T\Pi=-T$ and
$\Pi e^{\tau T}\Pi=e^{-\tau T}$ for every $\tau\in\C$
($T=2\tCas/\sqrt{1-q^{2}}$, Convention~\ref{conv:casimir}).
\item Coefficientwise on principal eigenvectors: with
$v_{k}(\theta)\coloneqq H_{k}(\cos\theta;q^{2})/\sqrt{(q^{2};q^{2})_{k}}$
(Convention~\ref{conv:casimir}), $(-1)^{k}v_{k}(\theta)=v_{k}(\pi-\theta)$ for all
$k\ge0$ and $\theta\in[0,\pi]$; that is, $\Pi\ket{\P^{\theta}}=\ket{\P^{\pi-\theta}}$.
\end{enumerate}
\end{lemma}

\begin{proof}
(i) On the eigenvalue $z=q^{-2k}$, $\ln z=-2k\ln q$, so
$z^{-i\pi/(2\ln q)}=\exp(i\pi k)=(-1)^{k}$. The function takes values in $\{\pm1\}\subset\R$,
so $\Pi$ is a self-adjoint unitary with $\Pi^{2}=1$.
(ii) $\tCas$ is bounded on $H_{\mathrm{ch}}$: writing $\tCas=A+A^{*}$ with
$A\ket{k}=\tfrac12\sqrt{1-q^{2k}}\ket{k-1}$, $\|A\|\le\tfrac12$ gives $\|\tCas\|\le1$. On
basis vectors,
$\Pi\tCas\Pi\ket{k}
=(-1)^{k}\bigl(\tfrac12\sqrt{1-q^{2k+2}}(-1)^{k+1}\ket{k+1}
+\tfrac12\sqrt{1-q^{2k}}(-1)^{k-1}\ket{k-1}\bigr)=-\tCas\ket{k}$,
and both sides are bounded, so the identity extends to $H_{\mathrm{ch}}$ by linearity and
continuity. The exponential statement follows term by term from the (norm-convergent, since
$T$ is bounded) power series of $e^{\tau T}$: $\Pi T^{n}\Pi=(-T)^{n}$.
(iii) $H_{k}(-x;q^{2})=(-1)^{k}H_{k}(x;q^{2})$ by induction on $k$ from the recurrence
$H_{k+1}(x)=2xH_{k}(x)-(1-q^{2k})H_{k-1}(x)$: it holds for $H_{0}=1$ and $H_{1}=2x$, and if it
holds at $k-1,k$ then
$H_{k+1}(-x)=-2xH_{k}(-x)-(1-q^{2k})H_{k-1}(-x)
=(-1)^{k+1}\bigl(2xH_{k}(x)-(1-q^{2k})H_{k-1}(x)\bigr)=(-1)^{k+1}H_{k+1}(x)$. Apply this at
$x=\cos\theta$, $-x=\cos(\pi-\theta)$. (Numerical confirmation to $10^{-59}$ or better:
checks (G1)--(G3).)
\end{proof}

\begin{proposition}[factorisation of the strange bilocal; the exact content of
Lemma~\ref{lem:reflection}]\label{prop:factorisation}
\PROVED\ Let $q\in(0,1)$ and $\ell\in\tfrac12\Z_{>0}$, with
$\O^{\ell},\mathcal{S}^{\ell}$ as in Convention~\ref{conv:chord}. Then:
\begin{enumerate}[label=(\alph*)]
\item \emph{(factorisation)} As bounded operators on $H_{\mathrm{ch}}$,
\begin{equation}\label{eq:factorisation}
  \mathcal{S}^{\ell}(0)=\O^{\ell}(0)\,\Pi=\Pi\,\O^{\ell}(0),
  \qquad
  \mathcal{S}^{\ell}(\tau)=\O^{\ell}(\tau)\;e^{\tau T}\Pi e^{-\tau T}
  =\O^{\ell}(\tau)\,\Pi\,e^{-2\tau T}\quad(\tau\in\C).
\end{equation}
\item \emph{(the conjugation reading is false)} $\Pi\,\O^{\ell}(0)\,\Pi^{-1}=\O^{\ell}(0)$:
conjugation by $\Pi$ fixes every function of $\hat\rho_{st}$, and
$\Pi\O^{\ell}(0)\Pi^{-1}\neq\mathcal{S}^{\ell}(0)$ (they differ by the factor $(-1)^{k}$ on
every chord state with $k$ odd). The strange bilocal is \emph{not} the discrete bilocal
conjugated by a normaliser element; it is the discrete bilocal \emph{multiplied} by the
grading unitary $\Pi$.
\item \emph{(kernel identity through $\Pi$)} For $\theta_{1},\theta_{2}\in[0,\pi]$ the chord
sums
\begin{equation}\label{eq:kernelsums}
  \sum_{k\ge0}v_{k}(\theta_{1})\,(-1)^{k}q^{2\ell k}\,v_{k}(\theta_{2})
  \;=\;\sum_{k\ge0}v_{k}(\theta_{1})\,q^{2\ell k}\,v_{k}(\pi-\theta_{2})
\end{equation}
converge absolutely and are equal; i.e.\
$\bra{\P^{\theta_1}}\mathcal{S}^{\ell}(0)\ket{\P^{\theta_2}}
=\bra{\P^{\theta_1}}\O^{\ell}(0)\ket{\P^{\pi-\theta_2}}$ termwise. Combined with
Lemma~\ref{lem:mehler} this reproduces Lemma~\ref{lem:reflection}; the identity
\eqref{eq:kernelsums} itself is proved here without the $q$-Mehler formula.
\end{enumerate}
\end{proposition}

\begin{proof}
(a) The Borel functional calculus of the positive self-adjoint operator $2q\hat\rho_{st}$ is
multiplicative, and
$z^{-\ell-\frac{i\pi}{2\ln q}}=z^{-\ell}\cdot z^{-\frac{i\pi}{2\ln q}}$ pointwise on
$(0,\infty)$; evaluating on the spectrum, $\mathcal{S}^{\ell}(0)\ket{k}=(-1)^{k}q^{2\ell k}
\ket{k}=\O^{\ell}(0)\Pi\ket{k}$. All three operators are diagonal in the chord basis, hence
commute pairwise. The $\tau$-dependent statement follows by conjugating with $e^{\tau T}$
(bounded for all $\tau\in\C$ since $T$ is bounded) and using
$\Pi e^{-\tau T}=e^{\tau T}\Pi$ (Lemma~\ref{lem:gradingchord}(ii)).
(b) Diagonal operators commute, so $\Pi\O^{\ell}(0)\Pi^{-1}=\O^{\ell}(0)$; on $\ket{k}$ with
$k$ odd, $\O^{\ell}(0)\ket{k}=q^{2\ell k}\ket{k}\neq-q^{2\ell k}\ket{k}
=\mathcal{S}^{\ell}(0)\ket{k}$.
(c) Absolute convergence: by the explicit expansion
$H_{k}(\cos\theta;q^{2})=\sum_{j=0}^{k}\genfrac[]{0pt}{1}{k}{j}_{q^{2}}e^{i(k-2j)\theta}$
[KLS \cite{KLS2010} \S14.26] and
$\genfrac[]{0pt}{1}{k}{j}_{q^{2}}\le(q^{2};q^{2})_{\infty}^{-2}$, one has
$|H_{k}(\cos\theta;q^{2})|\le(k+1)(q^{2};q^{2})_{\infty}^{-2}$, so both series in
\eqref{eq:kernelsums} are dominated by
$\sum_{k}(k+1)^{2}(q^{2};q^{2})_{\infty}^{-5}\,q^{2\ell k}<\infty$. Termwise equality is
Lemma~\ref{lem:gradingchord}(iii) applied to $v_{k}(\theta_{2})$. (Checks (G4)--(G5):
factorisation and conjugation triviality exact; kernel sums agree to relative
$3\times10^{-60}$ and match the closed form of
$\K^{\O}_{q}(\theta_{1},\pi-\theta_{2})$ to relative $4\times10^{-60}$.)
\end{proof}

\begin{proposition}[the $\Z_2$-grading of the dual algebra; GKK]\label{prop:gkkgrading}
\PROVED\ (in \cite{Groenevelt:2009gkk}; transcription, with the dictionary of
Convention~\ref{conv:translation}; numerically verified on truncations, checks (G6).)
Let $(M,\Delta)$ be the von Neumann algebraic quantum group $\SUq\rtimes\Z_2$ and $\hat M$ its
dual, acting on the GNS space $\mathcal{K}$ of the Haar weight. Then:
\begin{enumerate}[label=(\roman*)]
\item With $\mathcal{K}_{\pm}\coloneqq\overline{\mathrm{span}}\{f_{m,p,t}:\operatorname{sgn}(pt)
=\pm\}$ and $\hat M_{\pm}\coloneqq\{x\in\hat M: x\,\mathcal{K}_{\pm}\subseteq
\mathcal{K}_{\pm(\mp)}\}$ one has $\hat M=\hat M_{+}\oplus\hat M_{-}$, and the Casimir
satisfies the graded commutation relations $x\,\Omega_{\mathrm{GKK}}\subseteq
\Omega_{\mathrm{GKK}}\,x$ for $x\in\hat M_{+}$ and $y\,\Omega_{\mathrm{GKK}}\subseteq
-\Omega_{\mathrm{GKK}}\,y$ for $y\in\hat M_{-}$ [GKK Def.\ (defKpmandMpm), Prop.\
(decompMintoM+andM-andgradedcommutation)]. Since $\tCas=-\Omega_{\mathrm{GKK}}$
(Convention~\ref{conv:translation}), the same relations hold verbatim for $\tCas$.
\item $\hat M$ is generated by $K$, $E$, $U^{+-}_{0}$, $U^{-+}_{0}$, where the two extra
generators lie in the odd part $\hat M_{-}$ and are \emph{necessary}: $E$ and $K$ alone
do not suffice, precisely because they commute with the Casimir while $\hat M$ does not
[GKK Thm.\ (generatorsforhatM), Prop.\ (QppngeneratedualM), Lemma\
(polardecompositionQppn)].
\item On the eigenvectors $e^{\epsilon,\eta}_{m}(p,x)$ of $\Omega_{\mathrm{GKK}}$ for the
eigenvalue $\epsilon\eta x$ in the sector $(p,m,\epsilon,\eta)$, the generators act by
[GKK Lemma (actionsofgeneratorsoneigvetsOm), eq.\ (introprincipalseries)]
\begin{equation}\label{eq:gkkgradingaction}
\begin{split}
  K\,e^{\epsilon,\eta}_{m}(p,x)&=p^{\frac12}q^{m}\,e^{\epsilon,\eta}_{m}(p,x),\qquad
  (q^{-1}-q)\,E\,e^{\epsilon,\eta}_{m}(p,x)=q^{-m-\frac12}p^{-\frac12}
  \sqrt{1+2\epsilon\eta xq^{2m+1}p+q^{4m+2}p^{2}}\;e^{\epsilon,\eta}_{m+1}(p,x),\\
  U^{+-}_{0}\,e^{\epsilon,\eta}_{m}(p,x)&=\eta\,(-1)^{\upsilon(p)}\,
  e^{\epsilon,-\eta}_{m}(p,x),\qquad
  U^{-+}_{0}\,e^{\epsilon,\eta}_{m}(p,x)=\epsilon\,\eta^{\chi(p)}(-1)^{m}\,
  e^{-\epsilon,\eta}_{m}(p,x),
\end{split}
\end{equation}
where $\chi(p)=\log_{q}p$ and $\upsilon(p)=k$ for $p=q^{2k}$ or $p=q^{2k-1}$. In particular
both $U^{+-}_{0}$ and $U^{-+}_{0}$ flip the sign $\epsilon\eta$, hence map
$\Omega_{\mathrm{GKK}}$-eigenvectors for $\lambda$ to eigenvectors for $-\lambda$; they
commute with $K$; on the principal corepresentation space
$\mathcal{L}_{p,x}=\bigoplus_{\epsilon,\eta}\ell^{2}_{\epsilon,\eta}(p,x)$ they are unitary
and satisfy $(U^{+-}_{0})^{2}=(U^{-+}_{0})^{2}=-1$; and conjugation by them maps the
$E$-action to the $(\epsilon\eta\mapsto-\epsilon\eta)$-twisted action, with an extra factor
$-1$ for $U^{-+}_{0}$ (the $(-1)^{m}$ in \eqref{eq:gkkgradingaction} --- the same parity
twist as $\Pi$, in the $K$-weight index). (Checks (G6a)--(G6d): all verified to
$1.5\times10^{-59}$ or exactly, at $(q,\chi(p))=(0.6,0)$ and $(0.8,1)$.)
\end{enumerate}
\end{proposition}

\begin{theorem}[fusion of strange and discrete series in the normaliser quantum group]
\label{thm:fusion}
\PROVED\ (in \cite{Groenevelt:2009gkk}, Prop.\ (discretesubrepresinregrepr) and eq.\
(sudecompdiscretesubrepresinregrepr); label computation and transcription ours, numerically
verified, checks (G7).) Let $q\in(0,1)$, $p\in q^{\Z}$, and $x=\mu(\lambda)$ with
$\lambda\in-q^{2\Z+1}p\cup q^{2\Z+1}p$, $|\lambda|>1$; define $j,l\in\Z$ by
$|\lambda|=q^{1-2j}p^{-1}=q^{1+2l}p$ (equivalently $l=-j-\chi(p)$) and
\begin{equation}\label{eq:fusionlabel}
  \ell\coloneqq j+\tfrac12\chi(p)\in\tfrac12\Z,\qquad
  q^{2\ell-1}=|\lambda|^{-1},\qquad |\lambda|>1\iff\ell\ge1 .
\end{equation}
Then $(p,x)$ determines an irreducible discrete-series corepresentation $W_{p,x}$ of
$\SUq\rtimes\Z_2$ inside the left regular corepresentation, and as a $\Uq$-module
\begin{equation}\label{eq:fusion}
  W_{p,x}\big|_{\Uq}\;\cong\;\pi^{S}_{\ell-\frac12,\,\epsilon(p)}\;\oplus\;D^{-}_{\ell}
  \;\oplus\;D^{+}_{\ell},
  \qquad \epsilon(p)=\tfrac12\chi(p)\bmod 1,
\end{equation}
with $\tCas=-\mu(q^{2\ell-1})$ on the strange summand and $\tCas=+\mu(q^{2\ell-1})$ on both
discrete summands. Moreover:
\begin{enumerate}[label=(\alph*)]
\item \emph{(exhaustiveness)} $W\cong\bigoplus_{p\in q^{\Z}}\bigl(\int_{-1}^{1}W_{p,x}\,dx
\oplus\bigoplus_{x\in\sigma_{d}(\Omega_{p})}W_{p,x}\bigr)$ [GKK Thm.\
(decompositionofleftregularcorep)], and the corepresentations in the continuous part
restrict to direct sums of two principal series [GKK Rmk.\
(propintroprincipalseriescorepresentation)]. Hence \emph{every} occurrence of a strange
series in the Plancherel decomposition of $L^{2}_{q}(\mathrm{SU}(1,1)\rtimes\Z_2)$ lies
inside a single irreducible corepresentation \eqref{eq:fusion} whose other
$\Uq$-constituents are $D^{\pm}_{\ell}$ with $a=\ell-\tfrac12$ --- \emph{exactly} the label
substitution $\ell\to\ell+\tfrac{i\pi}{2\ln q}$, $a=\ell-\tfrac12$, by which
\cite{Schouten:2025dssyk} generate the strange series (Convention~\ref{conv:series}).
\item \emph{(the odd generators realise the exchange)} In the case $x>0$ the strange
summand is the block $(\epsilon,\eta)=(+,+)$ (all $m\in\Z$), and $U^{+-}_{0}$ maps the
$D^{+}_{\ell}$ summand (block $(+,-)$, $m\ge j$) isometrically into it, while $U^{-+}_{0}$
maps the $D^{-}_{\ell}$ summand (block $(-,+)$, $m\le l$) isometrically into it; restricted
to $W_{p,x}$ they are partial isometries. The cases $x<0$ are identical with the roles of
the blocks permuted [GKK Prop.\ (discretesubrepresinregrepr), cases (ii)--(iii)].
\end{enumerate}
\end{theorem}

\begin{proof}[Proof of the label computation and the transcription]
GKK state the decomposition (in their conventions, with their labels) as
$\pi^{S}_{\frac12(\chi(p)-1)+j,\,\epsilon(p)}\oplus D^{-}_{-\frac12\chi(p)-l}\oplus
D^{+}_{\frac12\chi(p)+j}$. From $l=-j-\chi(p)$ (which follows from
$q^{1-2j}p^{-1}=q^{1+2l}p$), the $D^{-}$ label is
$-\tfrac12\chi(p)-l=j+\tfrac12\chi(p)=\ell$, equal to the $D^{+}$ label, and the strange
label is $a=\tfrac12(\chi(p)-1)+j=\ell-\tfrac12$. The Casimir values: on the strange block
$\Omega_{\mathrm{GKK}}=+|x|=\mu(q^{2a})$ with $q^{2a}=q^{2j-1}p=|\lambda|^{-1}$, and on the
discrete blocks $\Omega_{\mathrm{GKK}}=-|x|=-\mu(q^{1-2\ell})$; translation by
$\tCas=-\Omega_{\mathrm{GKK}}$ gives the values stated. The identification of the blocks
with the printed matrix elements [GKK eqs.\ (strangeserrep), (posdiscrserrep),
(negdiscrserrep)] under the index maps $n+\epsilon(p)=m+\tfrac12\chi(p)$ (strange block),
$n=m-j$ ($D^{+}$ block) and $n=l-m$ ($D^{-}$ block) is a direct computation with the
factorisation $1+2\epsilon\eta xq^{2m+1}p+q^{4m+2}p^{2}
=(1+\epsilon\eta q^{2m+1}p\lambda)(1+\epsilon\eta q^{2m+1}p\lambda^{-1})$; it is carried
out numerically to $4\times10^{-61}$ relative for all three blocks at
$(q,\chi(p),j)=(0.6,0,2),(0.8,1,1),(0.7,2,3)$ (checks (G7), including the vanishing of the
lower-edge factor $1-q^{2m+2-2j}$ at $m=j-1$, which is what makes the one-sided blocks
invariant). Part (b) is \eqref{eq:gkkgradingaction} restricted to $W_{p,x}$, where
$e^{\epsilon,-\eta}_{m}(p,x)$, resp.\ $e^{-\epsilon,\eta}_{m}(p,x)$, is zero whenever the
corresponding eigenvalue is not in the discrete spectrum of the target sector [GKK,
definition of $e^{\epsilon,\eta}_{m}(p,x)$ in Sec.\ (ssec:discreteseries)].
\end{proof}

\begin{remark}[GKK's own summary]\label{rem:gkkquote}
\cite{Groenevelt:2009gkk} state the phenomenon verbatim in their introduction: ``However, in
the quantum group case the discrete series are no longer split up into a positive discrete
series and a negative discrete series.'' Theorem~\ref{thm:fusion} is the precise form: the
normaliser fuses $D^{+}_{\ell}$, $D^{-}_{\ell}$ \emph{and} $\pi^{S}_{\ell-1/2}$ into one
irreducible corepresentation. Classically the normaliser fuses only $D^{+}_{\ell}\oplus
D^{-}_{\ell}$ (an L-packet, as \cite{Schouten:2025dssyk} \S2.3 note); the adjunction of the
strange constituent is the quantum novelty.
\end{remark}

\begin{lemma}[essential uniqueness of $\Pi$]\label{lem:uniqueness}
\PROVED\ Let $V$ be a unitary operator on $H_{\mathrm{ch}}$ with $V\tCas=-\tCas V$. Then
$V=\Pi f(\tCas)$ for some Borel function $f$ with $|f|=1$ almost everywhere with respect to
the spectral measure of $\tCas$ at $\ket{0}$. In particular $\Pi$ is the unique
self-adjoint-unitary implementation of the grading on the chord sector up to a spectral
phase.
\end{lemma}

\begin{proof}
$\Pi V$ commutes with $\tCas$: $\Pi V\tCas=-\Pi\tCas V=\tCas\Pi V$ by
Lemma~\ref{lem:gradingchord}(ii). By Proposition~\ref{prop:chordprincipal}, $\tCas$ on
$H_{\mathrm{ch}}$ is unitarily equivalent to multiplication by $x$ on
$L^{2}([-1,1],\nu)$ with $\nu$ the (absolutely continuous) $q^{2}$-Hermite orthogonality
measure and cyclic vector $1$. A bounded operator $B$ on $L^{2}(\nu)$ commuting with
multiplication by $x$ commutes with multiplication by any polynomial, hence
$B(g)=B(1)\cdot g$ first for polynomials $g$ and then, by continuity of $B$ and density,
for all $g\in L^{2}(\nu)\cap L^{\infty}(\nu)$; $h\coloneqq B(1)$ satisfies
$\|hg\|_{2}\le\|B\|\,\|g\|_{2}$ for all such $g$, which forces $|h|\le\|B\|$ a.e.\ (apply it
to indicators of the sets $\{|h|>\|B\|+\epsilon\}$), so $B=M_{h}$ with
$h\in L^{\infty}(\nu)$. Thus $\Pi V=f(\tCas)$; unitarity of $\Pi V$ gives $|f|=1$
$\nu$-a.e., and $V=\Pi f(\tCas)$ since $\Pi^{2}=1$.
\end{proof}

\begin{remark}[what is \emph{not} established: $\Pi$ as a rescaled limit]\label{rem:pilimit}
\GAP{16}\ Lemma~\ref{lem:gradingchord} and Proposition~\ref{prop:gkkgrading} identify $\Pi$
with the normaliser grading \emph{structurally}: $\Pi$ anticommutes with $\tCas$ on the chord
sector exactly as the odd part $\hat M_{-}$ anticommutes with $\tCas$ on the GNS space, and
by Lemma~\ref{lem:uniqueness} it is the essentially unique such unitary. What is not
established is the \emph{limit} identification: whether $\Pi$ is the $r\to\infty$ rescaled
limit (SI (4.2)) of a concrete element of $\hat M_{-}$ (for instance of $U^{-+}_{0}$ in the
appropriate sector, whose $(-1)^{m}$ factor in \eqref{eq:gkkgradingaction} is the same
parity twist). SI's rescaling is performed on states and on the spherical element, not on
$\hat M_{-}$; issue \#16 records this gap. No statement below depends on it.
\end{remark}

\begin{definition}[normaliser-covariant assignment]\label{def:covariant}
Let $\mathcal{R}$ be the set of irreducible admissible unitary $*$-representations of $\Uq$
(Convention~\ref{conv:series}). An \emph{assignment} is a map $C$ from $\mathcal{R}$ to an
arbitrary set of values (e.g.\ $\{\mathrm{AdS},\mathrm{dS},\ldots\}$). $C$ is
\emph{normaliser-covariant} if $C(\pi)=C(\pi')$ whenever $\pi$ and $\pi'$ occur as
$\Uq$-constituents of one and the same irreducible unitary corepresentation of
$\SUq\rtimes\Z_2$ in the left regular corepresentation.
\end{definition}

\begin{theorem}[covariance obstruction]\label{thm:covobstruction}
\PROVED\ Let $C$ be any normaliser-covariant assignment. Then for every
$\ell\in\tfrac12\Z_{\ge1}$, with $\epsilon=0$ for integer $\ell$ and $\epsilon=\tfrac12$ for
half-odd $\ell$,
\begin{equation}\label{eq:covobstruction}
  C\bigl(\pi^{S}_{\ell-\frac12,\,\epsilon}\bigr)=C\bigl(D^{+}_{\ell}\bigr)
  =C\bigl(D^{-}_{\ell}\bigr).
\end{equation}
In particular no normaliser-covariant assignment separates the strange family at the
Plancherel labels $a=\ell-\tfrac12\in\tfrac12\Z_{>0}$ from the discrete family: a criterion
that classifies the discrete series as ``AdS type'' and the strange series as ``dS type''
cannot be normaliser-covariant.
\end{theorem}

\begin{proof}
Fix $\ell$. For integer $\ell$ take $p=1$ ($\chi(p)=0$), $j=\ell$; for half-odd $\ell$ take
$p=q$ ($\chi(p)=1$), $j=\ell-\tfrac12$. In both cases $\lambda=q^{1-2j}p^{-1}=q^{1-2\ell}>1$,
so Theorem~\ref{thm:fusion} applies with $x=\mu(\lambda)$ and produces an irreducible
corepresentation $W_{p,x}$ whose $\Uq$-constituents are precisely
$\pi^{S}_{\ell-1/2,\epsilon(p)}$, $D^{-}_{\ell}$ and $D^{+}_{\ell}$; covariance forces
\eqref{eq:covobstruction}. The parenthetical range statement: by
Theorem~\ref{thm:fusion}(a) the strange labels occurring in the Plancherel decomposition are
exactly $a=\ell-\tfrac12$, $\ell\in\tfrac12\Z_{\ge1}$, i.e.\ $a\in\tfrac12\Z_{>0}$
(Convention~\ref{conv:plancherel}).
\end{proof}

\begin{corollary}[radial-problem version]\label{cor:radialparity}
\PROVED\ Let $\omega\in\R$ and consider the chord-lattice radial equation
\eqref{eq:radialrecurrence} of Section~\ref{sec:matrixcoeffs}. The parity map
$(c_{k})_{k\ge0}\mapsto((-1)^{k}c_{k})_{k\ge0}$ is an involutive linear bijection from the
solution space at $\omega$ onto the solution space at $-\omega$ which preserves absolute
values pointwise --- hence preserves square-summability of the normalised sequences, decay
and growth rates, minimality of solutions, and maps the regular solution at $\omega$ (the one
with $c_{-1}=0$) to the regular solution at $-\omega$. Consequently any AdS/dS criterion
that is a function of parity-invariant data of the radial problem assigns the same type to
the spectral values $\omega=\mu(q^{2\ell-1})$ (discrete) and $\omega=-\mu(q^{2\ell-1})$
(strange).
\end{corollary}

\begin{proof}
If $2\omega c_{k}=c_{k+1}+(1-q^{2k})c_{k-1}$ for all $k\ge1$, then $c'_{k}=(-1)^{k}c_{k}$
satisfies $2(-\omega)c'_{k}=(-1)^{k+1}2\omega c_{k}
=(-1)^{k+1}c_{k+1}+(1-q^{2k})(-1)^{k-1}c_{k-1}=c'_{k+1}+(1-q^{2k})c'_{k-1}$. The map squares
to the identity and $|c'_{k}|=|c_{k}|$; $c_{-1}=0$ iff $c'_{-1}=0$. All the listed
properties depend on the sequences only through absolute values or through the vanishing of
$c_{-1}$.
\end{proof}

\begin{remark}[reading: a candidate obstruction for Claim~\ref{claim:cgeo}, and the

\end{remark}

$*$-structure $K^{*}=K$, $E^{*}=-F$; Hopf structure $\Delta(K)=K\otimes K$,
$\Delta(E)=K\otimes E+E\otimes K^{-1}$, $\Delta(F)=K\otimes F+F\otimes K^{-1}$,
$S(K)=K^{-1}$, $S(E)=-q^{-1}E$, $S(F)=-qF$, $\epsilon(K)=1$, $\epsilon(E)=\epsilon(F)=0$.
Classically $K=q^{H/2}$.
\end{convention}

\begin{convention}[Casimir and transfer matrix]\label{conv:casimir}
\begin{equation}\label{eq:ourCasimir}
  \Cas \coloneqq \frac{q^{-1}K^{-2}+qK^{2}-2}{(q-q^{-1})^{2}}+FE,
  \qquad
  \tCas \coloneqq \tfrac12\big((q-q^{-1})^{2}\Cas+2\big)
  =\tfrac12\big(qK^{2}+q^{-1}K^{-2}\big)+\tfrac12 (q-q^{-1})^{2}FE
\end{equation}
[SI (3.5) and \S3.3, \S3.5]. The DSSYK transfer matrix is $T=2\tCas/\sqrt{1-q^{2}}$
[SI \S4.2], acting on chord states by
$\tCas\ket{k}=\tfrac12\sqrt{1-q^{2k+2}}\,\ket{k+1}+\tfrac12\sqrt{1-q^{2k}}\,\ket{k-1}$
[SI (4.9)], with eigenvectors
$\ket{\P^{\theta}}=\sum_{k\ge0}H_{k}(\cos\theta;q^{2})/\sqrt{(q^{2};q^{2})_{k}}\,\ket{k}$
of eigenvalue $\cos\theta$ [SI (4.10)]. \DERIVED{} (check C5). We write
$\mu(y)\coloneqq\tfrac12(y+y^{-1})$ and $[z]_{q}\coloneqq(q^{z}-q^{-z})/(q-q^{-1})$
[SI (A.1)].
\end{convention}

\begin{convention}[Coordinate algebra and the normaliser extension]\label{conv:normaliser}
$\A_q(\mathrm{SU}(1,1))$: generators $\alpha,\gamma$ with relations, coproduct, counit, antipode
as in SI (3.1)--(3.2). By Woronowicz's no-go theorem there is no locally compact quantum group
$\SUq$; the extension $\SUq\rtimes\Z_2$ adds a central group-like self-adjoint $e$, $e^{2}=1$,
with $\alpha\alpha^{*}-q^{2}\gamma^{*}\gamma=e=\alpha^{*}\alpha-\gamma^{*}\gamma$ and the
$e$-twisted coproduct [SI (3.14)--(3.15)]; its irreducible $*$-representations $\pi_{\theta}$ act
on $L^{2}_{q^{2}}(I_{q^{2}})$, $I_{q}=(-\infty,-1)_{q}\cup[0,\infty)_{q}$, with position operator
$e\gamma^{*}\gamma$ [SI (3.18)]. The $\Z_2$ is the deck transformation between the two components
$\mathrm{SU}(1,1)$ and $\mathrm{SU}(1,1)v$ of the classical normaliser of $\mathrm{SU}(1,1)$ in
$\mathrm{SL}(2,\C)$ [SI (2.25)]; the extension is needed because only for the normaliser do the
self-adjoint extensions of $\Delta(e\gamma^{*}\gamma)$ admit coproduct-stable domains
(``smoothness conditions'', SI (3.19)), making a von Neumann algebraic quantum group possible.
\end{convention}

Admissible $*$-representations have $K$-eigenvalues in $q^{\frac12\Z}$ with finite
multiplicities [GKK \S4.4]. The irreducible admissible unitary $*$-representations, in our
conventions (dictionary of Convention~\ref{conv:translation}):

\subsection{Candidates}
\label{sec:spacecandidates}

The phrase ``quantum AdS$_2$'' is used for inequivalent quotients in the literature; the
computation of Sections \ref{sec:matrixcoeffs}--\ref{sec:criterion} depends on which one is
meant. The candidates appearing in the sources:

\begin{definition}[quantum disk]\label{def:qdisk}
The Cartan-type quotient $\SUq/\mathrm{U}(1)$ (invariant algebra generated by the spherical
element $e\gamma^{*}\gamma$ in the normaliser setting; classically the unit disk, i.e.\
Euclidean AdS$_2$). This is the quantum bounded symmetric domain of Vaksman et al. Its
transfer-matrix eigenfunctions are Al-Salam--Chihara--type (two levels up the $q$-Askey scheme
from DSSYK). \cite{Schouten:2025dssyk} \S5 states explicitly that it is ``another (distinct from
the DSSYK coset considered in this paper) quantum homogeneous space of the von Neumann algebraic
quantum group, corresponding to its quotient by a maximal compact subgroup.'' \emph{Not the DSSYK
space.}
\end{definition}

\begin{definition}[quantum hyperboloid]\label{def:qhyperboloid}
The split-torus quotient (classically the single-sheeted hyperboloid $=$ Lorentzian AdS$_2$).
It does not occur in \cite{Schouten:2025dssyk}'s reduction; we record it only to exclude it:
no chord structure arises along this quotient in any source we know of. \emph{Not the DSSYK
space.}
\end{definition}

\begin{definition}[twisted-primitive double quotient; the DSSYK space]\label{def:dssykspace}
For $s,t$ with $|s|,|t|>q$, the ``$(s,t)$-Gauss'' double quotient of
$\mathrm{SU}_q(1,1)\rtimes\Z_2$ by the one-parameter quantum subgroups generated by the twisted
primitive elements $Y_{s},Y_{t}$ [SI (3.43)], with invariant $*$-algebra
\begin{equation}\label{eq:invariantalgebra}
  \A_q\bigl((Y_s)\backslash \mathrm{SU}(1,1)\rtimes\Z_2/(Y_t)\bigr)=\langle\rho_{st},e\rangle,
  \qquad
  \rho_{st}\coloneqq e\gamma^{*}_{s,qt}\gamma_{s,qt}-s^{-2}-q^{-2}t^{-2}
\end{equation}
[SI (3.47)--(3.48)]. Functions on the total space diagonalising the symmetrised twisted
primitives $\tilde Y^{L}_{t},\tilde Y^{R}_{s}$ [SI (3.45)] with eigenvalues $\mu(\xi),\mu(\eta)$
(quantised momenta) and $\rho_{st}$-eigenvalue $\mu(y)$ are the mixed matrix elements
$\delta^{\epsilon,(st)}_{\eta,y,\xi}$ [SI \S3.4]; classically $\rho_{-1,1}=-2+4e^{2\varphi}$, so
the $\rho_{st}$-eigenvalue is the radial/length coordinate [SI below (3.49)].
\end{definition}

\subsection{The choice, justified from the sources}

\begin{proposition}[where DSSYK lives]\label{prop:where}
\DERIVED\ (from \cite{Schouten:2025dssyk} \S4, whose derivations we follow; the chord-sector
spectral statement is proved below.) The DSSYK dynamics lives on the quotient of
Definition~\ref{def:dssykspace}, in the rescaled regime SI (4.2), as follows.
\begin{enumerate}[label=(\alph*)]
\item \emph{Two-sided (chord) sector:} the span of the chord states
$\ket{k}_{\eta,\xi}$, $y=q\eta\xi q^{2k}$, $k\in\Z_{\ge0}$ [SI (4.1), (4.5)], which sit at
$\epsilon=-1$ --- the $\mathrm{SU}(1,1)v$ component of the normaliser. On it the symmetrised
Casimir acts as the DSSYK transfer matrix (Convention~\ref{conv:casimir}); DSSYK boundary
conditions are $\eta=\xi=1$.
\item \emph{One-sided space (``reduced quantum AdS$_{2,q}$''):} fixing only the right datum
$\xi\in q^{2\Z}$ gives the restricted $q$-lattice
\begin{equation}\label{eq:ourlattice}
  \R^{2}_{q^{2}}(\xi)=\{(-q^{2n},q^{2m+1})\}_{n,m\in\Z}\cup
  \{(q^{2n},q^{2m+1}):q^{2(n+m)}\le\xi\},
\end{equation}
with points $(\chi,y)$, $\chi=-\epsilon\eta^{-1}$ [SI (4.22)], inner product
\begin{equation}\label{eq:ourinnerproduct}
  \langle f,g\rangle_{\xi}
  =\int_{-\infty}^{\infty}\!\!d_{q^{2}}\chi\int_{0}^{\infty}\!\!d_{q^{2}}y\;
  (q\xi^{-1}\chi y;q^{2})_{\infty}\,\overline{f(\chi,y)}\,g(\chi,y)
\end{equation}
[SI (4.24)], and Casimir action
\begin{equation}\label{eq:ourradialoperator}
  \Cas=\frac{q^{-1}\mathcal{T}_{q}^{-2}+q\,\mathcal{T}_{q}^{2}-2}{(q-q^{-1})^{2}}
  -\mu_{\xi}\,y\,\frac{\chi}{1-q^{2}}\,\mathcal{T}_{q}^{-2},
  \qquad \mu_{\xi}\coloneqq\frac{\xi^{-1}}{1-q^{2}},
\end{equation}
where $(\mathcal{T}_{q}f)(\chi,y)=f(\chi,qy)$ [SI (4.25)]. Classical coordinates:
$p=\tfrac12\chi/(q^{-1}-q)$, $e^{\varphi}=\tfrac12 y^{-1/2}$ [SI (4.30)]; the weight in
\eqref{eq:ourinnerproduct} is the $q$-exponential
$E_{q^{2}}(-2py\xi^{-1}(1-q^{2}))\to e^{-2py\xi^{-1}}$ under asymptotic ($p_{R}\sim\lambda^{-1}$)
boundary conditions, and $\to1$ at fixed $p_R$, recovering classical AdS$_2$ data as $q\to1$
[SI \S4.5].
\end{enumerate}
The DSSYK length dictionary: chord number $k$ with $y=q\eta\xi q^{2k}$; renormalised length
$\varphi$ with $e^{\varphi}=\tfrac12y^{-1/2}$; bare geodesic length $\ell_{\mathrm{length}}
=\lambda k$ in the triple-scaling regime, $\lambda=-2\ln q$; length positivity $=$ the lattice
restriction $q^{2(n+m)}\le\xi$ [SI \S4.1].
\end{proposition}

\begin{remark}[position space is assumed, not constructed]\label{rem:positionspace}
\cite{Schouten:2025dssyk} \S4.4 constructs the space in \emph{momentum} coordinates $(\chi,y)$
and writes: ``we assume there exists an invertible $q$-Fourier transform'' and ``an appropriate
explicit expression for the inner product in the position basis, which is compatible with the
constraint and the inner product in the momentum space, is currently not available to us.''
All computations in this project stay on the momentum-space lattice \eqref{eq:ourlattice}, where
every object is defined; nothing below uses the assumed $q$-Fourier transform. If a later issue
needs the position picture, that requires a \texttt{gap} issue first.
\end{remark}

\subsection{Where the strange series lives}
\label{sec:wherestrange}

\begin{proposition}[chord-sector states are principal only]\label{prop:chordprincipal}
\PROVED\ The closure of the span of the chord states $\{\ket{k}\}_{k\ge0}$ decomposes under the
symmetrised Casimir $\tCas$ with purely absolutely continuous spectrum $[-1,1]$ and spectral
multiplicity one; i.e.\ only principal-series Casimir values occur, and no discrete, strange, or
complementary values.
\end{proposition}

\begin{proof}
$\tCas$ acts on $\ell^{2}(\Z_{\ge0})$ as the Jacobi matrix with off-diagonal
$a_{k}=\tfrac12\sqrt{1-q^{2k}}$ and zero diagonal (Convention~\ref{conv:casimir}). It is
bounded and self-adjoint (all $|a_k|\le\tfrac12$), so its spectral measure at the cyclic vector
$\ket{0}$ is the unique orthogonality measure of the corresponding orthonormal polynomials,
which are the normalised continuous $q^{2}$-Hermite polynomials; their orthogonality measure is
$\frac{(q^{2};q^{2})_{\infty}}{2\pi}\,|(e^{2i\theta};q^{2})_{\infty}|^{2}\,d\theta$ on
$\cos\theta\in[-1,1]$ --- absolutely continuous with no mass points (Askey--Wilson $a=b=c=d=0$;
\cite{KLS2010} \S14.26). Cyclicity of $\ket{0}$ (three-term recurrence with $a_{k}>0$ for
$k\ge1$) gives multiplicity one. By Convention~\ref{conv:series}, values in $[-1,1]$ are
principal-series values.
\end{proof}

\begin{proposition}[where the strange series does occur]\label{prop:strangewhere}
\PROVED\ (in \cite{Groenevelt:2009gkk}; not re-derived here.) In the decomposition of the GNS
space of the Haar weight of $L^{\infty}_{q}(\mathrm{SU}(1,1)\rtimes\Z_2)$ as a $\Uq$-module, the
strange series occurs as discrete direct summands in the sectors $(\varepsilon,\eta)\in
\{(+,-),(-,+),(+,+)\}$ with labels $q^{2a}=pq^{1+2l}$ (Convention~\ref{conv:plancherel}).
\end{proposition}

\begin{remark}[consequence for the programme; the precise sense of ``on the space'']
\label{rem:strangeonspace}
Propositions \ref{prop:chordprincipal} and \ref{prop:strangewhere} together fix what ``matrix
coefficients of the strange series on the DSSYK space'' can mean, and this constrains issues
\#6--\#8:
\begin{enumerate}[label=(\roman*)]
\item \emph{As states of the DSSYK sector:} the strange series does \textbf{not} occur. The
chord sector is principal only. Any interpolation mechanism through strange \emph{states} of the
transfer matrix is excluded.
\item \emph{As operators (bilocals):} the strange series acts on the chord sector through
$\mathcal{S}^{\ell}$ (Convention~\ref{conv:chord}), with the kernel properties of
Section~\ref{sec:intro}.
\item \emph{As functions on the group:} strange-series matrix coefficients are elements of
$L^{2}_{q}(\mathrm{SU}(1,1)\rtimes\Z_2)$ (Proposition~\ref{prop:strangewhere}); their
$(s,t)$-reduction obeys the radial Casimir equation \eqref{eq:ourradialoperator} at spectral
value $\tCas=-\mu(q^{2a})<-1$, i.e.\ \emph{below the principal band}. The radial problem is the
same $q$-difference operator for every series; series membership enters only through the
spectral value. This is the object issues \#6--\#7 compute.
\end{enumerate}
\end{remark}

\subsection{The representations, explicitly}

\begin{proposition}[strange series in our conventions]\label{prop:strangeexplicit}
\PROVED\ (transcription of \cite{Groenevelt:2009gkk} eq.\ (strangeserrep) through the verified

\begin{split}
  K e_{n}&=q^{n+\epsilon}e_{n},\\
  (q^{-1}-q)\,E\,e_{n}
  &=q^{-n-\epsilon-\frac12}\sqrt{(1+q^{2n+2\epsilon+1+2a})(1+q^{2n+2\epsilon+1-2a})}\;e_{n+1},\\
  F&=-E^{*},
\end{split}
\end{equation}
with $\tCas=-\mu(q^{2a})$, equivalently
$\Cas=-2\bigl(\mu(q^{2a})+1\bigr)/(q-q^{-1})^{2}=-\bigl(q^{a}+q^{-a}\bigr)^{2}/(q-q^{-1})^{2}$
(Convention~\ref{conv:casimir}). It is unitary (all square-root arguments are manifestly positive for every $a>0$), irreducible,
and admissible; the normaliser extension acts by the $\Z_2$-grading operators
$U^{+-}_{0},U^{-+}_{0}$ of \cite{Groenevelt:2009gkk} which map $\tCas\mapsto-\tCas$, i.e.\
exchange the strange family with the discrete family and the principal band with itself

$D^{\pm}_{\ell}$ into a single irreducible corepresentation, are
Proposition~\ref{prop:gkkgrading} and Theorem~\ref{thm:fusion}.
\end{proposition}

\subsection{Radial coefficients on the chord lattice}

By Remark~\ref{rem:strangeonspace}, the meaningful ``matrix coefficients on the DSSYK space''
are the solutions of the radial Casimir equation at the strange spectral value. On the chord
lattice the radial equation is the three-term relation of Convention~\ref{conv:casimir}; in
unnormalised form ($c_{k}=v_{k}\sqrt{(q^{2};q^{2})_{k}}$):
\begin{equation}\label{eq:radialrecurrence}
  2\omega\,c_{k}=c_{k+1}+(1-q^{2k})\,c_{k-1},\qquad k\ge1,
  \tag{R}
\end{equation}
with the \emph{regular} solution fixed by $c_{1}=2\omega c_{0}$ (equivalently $c_{-1}=0$), i.e.\
$c^{\mathrm{reg}}_{k}=H_{k}(\omega;q^{2})$.

\begin{theorem}[strange radial coefficients, closed forms]\label{thm:strangeradial}
Let $0<s<1$ and $\omega=-\mu(s)<-1$ (strange values: $s=q^{2a}$, $a>0$;
Convention~\ref{conv:series}).
\begin{enumerate}[label=(\alph*)]
\item \PROVED\ \emph{(minimal solution.)} The sequence
\begin{equation}\label{eq:minimalsolution}
  c^{S}_{k}(a)\coloneqq(-1)^{k}\,q^{2ak}\;
  {}_{2}\varphi_{1}\!\left(\begin{matrix}0,\,0\\ q^{4a+2}\end{matrix};\,q^{2},\,
  q^{2k+2}\right),
  \qquad
  {}_{2}\varphi_{1}\!\left(\begin{matrix}0,0\\ c\end{matrix};q,z\right)
  =\sum_{j\ge0}\frac{z^{j}}{(q;q)_{j}(c;q)_{j}},
\end{equation}
solves \eqref{eq:radialrecurrence} at $\omega=-\mu(q^{2a})$ for all $k\ge1$, and
$c^{S}_{k}(a)=(-1)^{k}q^{2ak}\,(1+O(q^{2k}))$; the normalised sequence
$v^{S}_{k}=c^{S}_{k}/\sqrt{(q^{2};q^{2})_{k}}$ is square-summable. (Check S1$'$;
proof below.)
\item \DERIVED\ \emph{(regular solution and growth.)} The regular solution is
$c^{\mathrm{reg}}_{k}=H_{k}(-\mu(q^{2a});q^{2})=(-1)^{k}H_{k}(\mu(q^{2a});q^{2})$ with
$H_{k}(\mu(s);q^{2})\,s^{k}\,(s^{2};q^{2})_{\infty}\to1$ as $k\to\infty$; it grows like
$q^{-2ak}$ and is not square-summable --- consistently with
Proposition~\ref{prop:chordprincipal} (no strange states). (Check S2.)
\item \DERIVED\ \emph{(Wronskian and connection formula.)} With
$W_{k}\coloneqq c^{\mathrm{reg}}_{k+1}c^{\mathrm{min}}_{k}-c^{\mathrm{min}}_{k+1}
c^{\mathrm{reg}}_{k}$ one has $W_{k}=(q^{2};q^{2})_{k}\,W_{0}$ (\PROVED, one-line induction from
\eqref{eq:radialrecurrence}), and
\begin{equation}\label{eq:jost}
  W_{0}=\frac{s^{-1}-s}{(s^{2};q^{2})_{\infty}(q^{2};q^{2})_{\infty}}\;\neq\;0
  \quad\text{for all }0<s<1,
\end{equation}
equivalently the connection formula (generic $s$, $s^{2}\notin q^{2\Z}$)
\begin{equation}\label{eq:connection}
  H_{k}(\mu(s);q^{2})
  =\frac{c^{\mathrm{min}}_{k}(s^{-1})}{(s^{2};q^{2})_{\infty}}
  +\frac{c^{\mathrm{min}}_{k}(s)}{(s^{-2};q^{2})_{\infty}},
  \qquad
  c^{\mathrm{min}}_{k}(s)\coloneqq s^{k}\,
  {}_{2}\varphi_{1}\!\left(\begin{matrix}0,0\\ s^{2}q^{2}\end{matrix};q^{2},q^{2k+2}\right),
\end{equation}
verified to relative $10^{-33}$ or better at multiple $(q,s)$ (checks S3, S3$'$, S5). At the
\emph{resonant} points $s^{2}\in q^{2\Z}$ --- which include every strange label occurring in the
Plancherel decomposition, $a\in\tfrac12\Z_{>0}$ --- the branch $c^{\mathrm{min}}(s^{-1})$
degenerates and \eqref{eq:connection} requires a confluent limit; the minimal solution
\eqref{eq:minimalsolution} itself remains well-defined there.
\end{enumerate}
\end{theorem}

\begin{proof}[Proof of (a)]
Substitute $c_{k}=s^{k}\varphi(q^{2k})$, $\varphi(z)=\sum_{j\ge0}d_{j}z^{j}$, into
\eqref{eq:radialrecurrence} at $\omega=\mu(s)$ (the value $-\mu(s)$ follows by the parity map
$c_{k}\to(-1)^{k}c_{k}$, which flips the sign of $\omega$ in \eqref{eq:radialrecurrence}).
Matching coefficients of $z^{j}$ in $(s^{2}+1)\varphi(z)=s^{2}\varphi(q^{2}z)
+(1-z)\varphi(q^{-2}z)$ gives
$d_{j}\bigl[(s^{2}+1)-s^{2}q^{2j}-q^{-2j}\bigr]=-d_{j-1}q^{2-2j}$, i.e.\
$d_{j}=d_{j-1}\,q^{2}/\bigl[(1-q^{2j})(1-s^{2}q^{2j})\bigr]$, so
$d_{j}=q^{2j}/\bigl[(q^{2};q^{2})_{j}(s^{2}q^{2};q^{2})_{j}\bigr]$ with $d_{0}=1$ ---
the series \eqref{eq:minimalsolution}. For $0<s<1$ the factors $(1-s^{2}q^{2i})$ are bounded
below by $1-s^{2}q^{2}>0$, so $|d_{j}|\le C q^{2j}$ and $\varphi$ is analytic in
$|z|<q^{-2}$, which contains every lattice point $z=q^{2k}$, $k\ge0$; the term-by-term
manipulation is therefore justified, and $\varphi(q^{2k})=1+O(q^{2k})$ gives the stated
asymptotics and square-summability. (Numerical confirmation to $10^{-40}$: checks S1, S1$'$.)
\end{proof}

\begin{remark}[the $\rho$-shift]\label{rem:rhoshift}
At the discrete-series value $\omega=\mu(q^{2\ell-1})$ the minimal solution decays like
$q^{(2\ell-1)k}$ while the discrete bilocal (Convention~\ref{conv:chord}) weights chord states
by $q^{2\ell k}$; the mismatch $q^{-k}$, constant $=1$ numerically (check S4), is the usual
Weyl-vector shift between an operator weight and a wavefunction decay rate, recorded here so
that it is not mistaken for an inconsistency downstream. The same shift appears in the strange
family: bilocal weight $(-1)^{k}q^{2\ell k}$ versus minimal-solution decay
$(-1)^{k}q^{2ak}=(-1)^{k}q^{(2\ell-1)k}$ under $a=\ell-\tfrac12$.
\end{remark}

\subsection{Bilocal coefficients between principal states}

The equal-time overlap kernels of the strange bilocals are Lemma~\ref{lem:mehler}: for
$\ell\in\tfrac12\Z_{>0}$ the chord sum converges absolutely ($|t|=q^{2\ell}<1$) and
\begin{equation}
  \bra{\P^{\theta_1}}\mathcal{S}^{\ell}(0)\ket{\P^{\theta_2}}
  =\frac{(q^{4\ell};q^{2})_{\infty}}{(-q^{2\ell}e^{i(\pm\theta_1\pm\theta_2)};q^{2})_{\infty}},
\end{equation}
holomorphic in $q^{2\ell}$ on the unit disk; no analytic continuation from any compact-quotient
computation is used anywhere in this section (everything is defined directly on the non-compact
lattice), so the caveat of arXiv:2412.19681 about continuation-based derivations does not apply
here.

\subsection{Orthogonality and Plancherel: what is available and what is not}

\begin{proposition}[available]\label{prop:planchereldata}
\PROVED\ (in the sources.) (i) On the chord sector, the principal eigenfunctions are complete
and delta-orthonormal with the DSSYK Plancherel density
$\frac{d\theta}{2\pi}(q^{2},e^{\pm2i\theta};q^{2})_{\infty}$ [SI (4.11);
Proposition~\ref{prop:chordprincipal}]. (ii) At the unrescaled level, the GKK sector
decompositions (Convention~\ref{conv:plancherel}) provide a rigorous Plancherel theory in which
the strange series occurs with counting measure at $q^{2a}=pq^{1+2l}$.
\end{proposition}

\begin{remark}[not available: rescaled Plancherel for the strange family]\label{rem:noplancherel}
\GAP{14}\ No orthogonality or completeness statement for strange-series coefficients
\emph{on the DSSYK space} is available: the strange radial coefficients
\eqref{eq:minimalsolution} are solutions at spectral values off the spectrum of the self-adjoint
transfer matrix, and whether the GKK Plancherel decomposition survives SI's rescaling
$r\to\infty$ is not established --- by us or by \cite{Schouten:2025dssyk}. Every statement in
this project that would sum or integrate over strange labels on the DSSYK space is conditional
on issue \#14 and says so. The unconditional statements are: the GKK theory at finite scaling,
the chord-sector principal completeness, the kernel computations of Section~\ref{sec:intro},
and Theorem~\ref{thm:strangeradial}.
\end{remark}

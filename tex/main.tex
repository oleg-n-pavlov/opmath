\subsection{The claim, verbatim}
\label{sec:claimverbatim}

The object of this project is the following assertion of Schouten--Isachenkov
\cite{Schouten:2025dssyk} (arXiv:2512.10101v2; all quotations and equation numbers refer to v2;
the full quotation ledger with locations is \texttt{lit/notes/2512.10101.md}):

\begin{quote}
``Lastly, we make remarks on the correlation function related to the strange series
representation, which is argued to interpolate between the AdS and dS regions of our
$q$-homogeneous space.'' \hfill (abstract)
\end{quote}
\begin{quote}
``such that the strange series has a non-zero amplitude between the lower part and upper part of
the spectrum~. It was proposed in [arXiv:2411.16922] and [arXiv:2505.08116] that zooming into the
upper part of the spectrum, i.e.\ $\theta=|\ln(q^2)|k$, corresponds to de Sitter JT-gravity.
If this proposal were proven to be correct, the strange series would therefore have a non-zero
amplitude between the anti-de Sitter and de Sitter regions.'' \hfill (\S4.3, end)
\end{quote}
\begin{quote}
``Moreover, supposing that dS space corresponds to the upper part of the spectrum, it was shown
in Section~4.3 that the strange series gives non-zero amplitudes between AdS and dS regions, and
thus in some sense interpolates between them. It is currently unknown what the correct physics
interpretation of such an amplitude would be.'' \hfill (\S5, Discussion, item 1)
\end{quote}

The authors' own epistemic labelling is therefore: \emph{argued}, twice conditional --- on the
dS identification of the upper spectral edge, and on an interpretation of the amplitude that they
declare unknown. The phrase ``in some sense interpolates'' is theirs; making the sense precise is
the purpose of this section.

\subsection{Setting and the paper's computation}
\label{sec:setting}

Throughout, conventions are those of Section~\ref{sec:conventions}. The DSSYK spectral parameter
is $\theta\in[0,\pi]$, $E(\theta)=2\cos\theta/\sqrt{1-q^{2}}$ (Convention~\ref{conv:q}); the
principal-series eigenvectors $\ket{\P^{\theta}}$ and the chord states $\ket{k}$ are as in
Conventions \ref{conv:casimir} and \ref{conv:chord}.

\begin{definition}[edge regimes]\label{def:edges}
For $k\ge0$ fixed and $q\to1^{-}$, the \emph{lower-edge regime} is the family
$\theta^{L}_{q}(k)\coloneqq\pi-|\ln q^{2}|\,k$ and the \emph{upper-edge regime} is
$\theta^{U}_{q}(k)\coloneqq|\ln q^{2}|\,k$.
\end{definition}

\begin{definition}[AdS region]\label{def:ads}
The \emph{AdS region} is the lower-edge regime. Justification (not a convention): in this regime
the $q\to1$ limit of DSSYK is the Schwarzian theory on AdS$_2$ [SI \S4.2, eq.\ (4.17); standard
DSSYK literature \cite{Berkooz:2018largeN}], and every bulk-dual proposal on the market agrees on
this identification.
\end{definition}

\begin{definition}[dS region --- two inequivalent candidates]\label{def:ds}
\emph{(a) Edge-dS (dS-JT):} the upper-edge regime $\theta^{U}_{q}(k)$
[arXiv:2411.16922 \cite{Blommaert:2024entropic}; arXiv:2505.08116 \cite{Okuyama:2025dsjt}].
\emph{(b) Centre-dS:} the middle of the band, $\theta=\pi/2$, $E=0$
[arXiv:2310.16994 \cite{Narovlansky:2023lfz} and successors].
The claim of \cite{Schouten:2025dssyk} is formulated with (a); its footnote records (b) as a
rival. Any statement below that depends on the choice says so explicitly.
\end{definition}

The paper's strange-series computation concerns the bilocal operators of
Convention~\ref{conv:chord},
$\O^{\ell}$ (discrete series) and $\mathcal{S}^{\ell}$ (strange series), $\ell\in\tfrac12\Z_{>0}$,
through their equal-time overlap kernels
\begin{equation}\label{eq:kernels}
  \K^{\O}_{q}(\theta_{1},\theta_{2})
  \coloneqq\bra{\P^{\theta_1}}\O^{\ell}(0)\ket{\P^{\theta_2}}
  =\frac{(q^{4\ell};q^{2})_{\infty}}{(q^{2\ell}e^{i(\pm\theta_1\pm\theta_2)};q^{2})_{\infty}},
  \qquad
  \K^{\mathcal{S}}_{q}(\theta_{1},\theta_{2})
  \coloneqq\bra{\P^{\theta_1}}\mathcal{S}^{\ell}(0)\ket{\P^{\theta_2}},
\end{equation}
where the denominator is the product over the four sign choices [SI (4.15)--(4.16), (4.19)].

\begin{lemma}[$q$-Mehler form; chord-side characterisation]\label{lem:mehler}
\DERIVED\ For $t\in\C$, $|t|<1$,
\begin{equation}\label{eq:qmehler}
  \sum_{k\ge0}\frac{t^{k}}{(q^{2};q^{2})_{k}}\,H_{k}(\cos\theta_1;q^{2})H_{k}(\cos\theta_2;q^{2})
  =\frac{(t^{2};q^{2})_{\infty}}{(t\,e^{i(\pm\theta_1\pm\theta_2)};q^{2})_{\infty}},
\end{equation}
the $q$-Mehler formula for continuous $q$-Hermite polynomials
(Gasper--Rahman; \cite{Berkooz:2018largeN} eq.\ (2.28)). Consequently
$\K^{\O}_{q}$ is \eqref{eq:qmehler} at $t=q^{2\ell}$, and
$\K^{\mathcal{S}}_{q}$ is \eqref{eq:qmehler} at $t=-q^{2\ell}$:
the strange bilocal inserts matter with \emph{alternating chord weights}
$(-1)^{k}q^{2k\ell}$, versus $q^{2k\ell}$ for the discrete bilocal. In particular
$\K^{\mathcal{S}}_{q}(\theta_1,\theta_2)
=(q^{4\ell};q^{2})_{\infty}/(-q^{2\ell}e^{i(\pm\theta_1\pm\theta_2)};q^{2})_{\infty}$,
which is SI (4.19) before its limit.
\end{lemma}

\begin{proof}
$\hat\rho_{st}\ket{k}=\tfrac12 q^{-1}q^{-2k}\ket{k}$ [SI (4.13)] gives
$(2q\hat\rho_{st})^{-\ell-\frac{i\pi}{2\ln q}}\ket{k}
=q^{2k(\ell+\frac{i\pi}{2\ln q})}\ket{k}=(-1)^{k}q^{2k\ell}\ket{k}$, since
$q^{2k\cdot i\pi/(2\ln q)}=e^{i\pi k}$. Expanding both $\ket{\P^{\theta_i}}$ in chord states
(Convention~\ref{conv:casimir}) yields the left side of \eqref{eq:qmehler} with $t=\mp q^{2\ell}$;
\eqref{eq:qmehler} itself is classical.
\end{proof}

\begin{lemma}[exact reflection identity]\label{lem:reflection}
\DERIVED\ For all $q\in(0,1)$, $\ell\in\tfrac12\Z_{>0}$ and $\theta_{1},\theta_{2}$,
\begin{equation}\label{eq:reflection}
  \K^{\mathcal{S}}_{q}(\theta_{1},\theta_{2})
  =\K^{\O}_{q}(\theta_{1},\,\pi-\theta_{2})
  =\K^{\O}_{q}(\pi-\theta_{1},\,\theta_{2}).
\end{equation}
That is: at finite $q$, the strange-series equal-time kernel is the discrete-series kernel with
one spectral argument reflected through the middle of the band ($E\to-E$).
\end{lemma}

\begin{proof}
$e^{\pm i(\pi-\theta_2)}=-e^{\mp i\theta_2}$, so the four denominator arguments
$-q^{2\ell}e^{i(\pm\theta_1\pm\theta_2)}$ of $\K^{\mathcal{S}}_{q}(\theta_1,\theta_2)$
(Lemma~\ref{lem:mehler}) coincide, as a set, with the four arguments
$q^{2\ell}e^{i(\pm\theta_1\pm(\pi-\theta_2))}$ of
$\K^{\O}_{q}(\theta_1,\pi-\theta_2)$; the numerators are equal. The second equality holds because
the kernel depends on $(\theta_1,\theta_2)$ only through the four sums $\pm\theta_1\pm\theta_2$,
and reflecting both arguments returns the original kernel:
$\K^{\mathcal{S}}$ and $\K^{\O}$ are each invariant under simultaneous reflection.
\end{proof}

\begin{proposition}[the paper's limit statements, re-derived]\label{prop:limits}
\DERIVED\ Fix $\ell\in\tfrac12\Z_{>0}$ and $k_{1},k_{2}\ge0$. As $q\to1^{-}$:
\begin{enumerate}[label=(\roman*)]
\item \emph{Same edge, discrete (Schwarzian benchmark):}
$\K^{\O}_{q}(\theta^{L}_{q}(k_1),\theta^{L}_{q}(k_2))
\sim N(q)\,\dfrac{\Gamma(\ell\pm ik_1\pm ik_2)}{\Gamma(2\ell)}$,
where $\Gamma(\ell\pm ik_1\pm ik_2)$ denotes the product over the four sign choices and
$N(q)\coloneqq(1-q^{2})^{2\ell-3}(q^{2};q^{2})_{\infty}^{-3}$
[SI (4.17); the $q$-gamma limit $\Gamma_{q^2}\to\Gamma$].
\item \emph{Same edge, strange:} the normalisation-free ratio vanishes,
$\K^{\mathcal{S}}_{q}(\theta^{L}_{q}(k_1),\theta^{L}_{q}(k_2))\,/\,
 \K^{\O}_{q}(\theta^{L}_{q}(k_1),\theta^{L}_{q}(k_2))\longrightarrow 0$
[this is the content of SI (4.19)].
\item \emph{Opposite edges, strange:} exactly, at every $q$,
$\K^{\mathcal{S}}_{q}(\theta^{L}_{q}(k_1),\theta^{U}_{q}(k_2))
=\K^{\O}_{q}(\theta^{L}_{q}(k_1),\theta^{L}_{q}(k_2))$,
hence with the same normalisation $N(q)$ as in (i) the limit is the Schwarzian $3j$-symbol
$\Gamma(\ell\pm ik_1\pm ik_2)/\Gamma(2\ell)$ [this is SI (4.20), sharpened to an identity].
\item \emph{Opposite edges, discrete:} the ratio
$\K^{\O}_{q}(\theta^{L}_{q}(k_1),\theta^{U}_{q}(k_2))\,/\,
 \K^{\O}_{q}(\theta^{L}_{q}(k_1),\theta^{L}_{q}(k_2))\longrightarrow 0$:
ordinary (discrete-series) matter does \emph{not} connect the two edges.
\end{enumerate}
\end{proposition}

\begin{proof}
$\theta^{L}_{q}(k)$ gives $e^{i\theta^{L}_{q}(k)}=-q^{2ik}$ and $\theta^{U}_{q}(k)$ gives
$e^{i\theta^{U}_{q}(k)}=q^{-2ik}$. (iii) is then Lemma~\ref{lem:reflection} evaluated at
$\pi-\theta^{U}_{q}(k_2)=\theta^{L}_{q}(k_2)$ --- the reflection is exact, no limit needed ---
combined with (i). For (i): each denominator factor is
$(q^{2(\ell\pm ik_1\pm ik_2)};q^{2})_{\infty}$, and
$(q^{2x};q^{2})_{\infty}=(1-q^{2})^{1-x}(q^{2};q^{2})_{\infty}/\Gamma_{q^{2}}(x)$
[SI (A.3)] with $\Gamma_{q^{2}}(x)\to\Gamma(x)$ gives the stated asymptotics; the imaginary
parts cancel in the accumulated exponent of $(1-q^{2})$. For (ii) and (iv): by
Lemma~\ref{lem:reflection} the two ratios are equal, so it suffices to prove (iv). By (i)'s
factorisation, the ratio in (iv) is
$\prod_{\pm\pm}(q^{2(\ell\pm ik_1\pm ik_2)};q^{2})_{\infty}
/\prod_{\pm\pm}(-q^{2(\ell\pm ik_1\pm ik_2)};q^{2})_{\infty}$.
Fix one of the four sign choices and set $x\coloneqq\ell\pm ik_{1}\pm ik_{2}$, so that
\[
  q^{2x+2n}=\rho_{n}\,e^{i\varphi_{q}},\qquad
  \rho_{n}\coloneqq q^{2\ell+2n}\in(0,1),\qquad
  \varphi_{q}\coloneqq 2(\pm k_{1}\pm k_{2})\ln q ,
\]
the phase $\varphi_{q}$ being \emph{independent of $n$}. Both infinite products converge
absolutely, and $|1+q^{2x+2n}|\ge1-\rho_{n}\ge1-q^{2\ell}>0$ for every $n$, so each factor
$r_{x}(q)\coloneqq(q^{2x};q^{2})_{\infty}/(-q^{2x};q^{2})_{\infty}$ is a well-defined
nonzero number with
\[
  |r_{x}(q)|^{2}
  =\prod_{n\ge0}\frac{|1-\rho_{n}e^{i\varphi_{q}}|^{2}}{|1+\rho_{n}e^{i\varphi_{q}}|^{2}}
  =\prod_{n\ge0}\frac{1-2\rho_{n}\cos\varphi_{q}+\rho_{n}^{2}}
                     {1+2\rho_{n}\cos\varphi_{q}+\rho_{n}^{2}} .
\]
Assume $q$ close enough to $1$ that $2(k_{1}+k_{2})|\ln q|\le\pi/3$ (for $k_{1}=k_{2}=0$ no
condition is needed); then $|\varphi_{q}|\le\pi/3$, hence $\cos\varphi_{q}\ge\tfrac12$, for
all four sign choices. The per-factor ratio is decreasing in $\cos\varphi_{q}$ (its
derivative in $c=\cos\varphi_{q}$ is $-4\rho_{n}(1+\rho_{n}^{2})$ divided by the squared
denominator, hence negative), so for every $n$
\[
  \frac{1-2\rho_{n}\cos\varphi_{q}+\rho_{n}^{2}}{1+2\rho_{n}\cos\varphi_{q}+\rho_{n}^{2}}
  \;\le\;\frac{1-\rho_{n}+\rho_{n}^{2}}{1+\rho_{n}+\rho_{n}^{2}}\eqqcolon g(\rho_{n})\;<\;1 .
\]
The function $g$ is strictly decreasing on $(0,1)$: the numerator of $g'$ is
$(2\rho-1)(1+\rho+\rho^{2})-(1-\rho+\rho^{2})(2\rho+1)=2\rho^{2}-2<0$. Hence
$g(\rho_{n})\le g(\tfrac12)=\tfrac37$ whenever $\rho_{n}\ge\tfrac12$, which happens exactly
for $n\le N_{\ell}(q)\coloneqq\ln2/(2|\ln q|)-\ell$; the number of such lattice sites tends
to $\infty$ as $q\to1^{-}$. Every remaining factor is $\le1$. Therefore
\[
  |r_{x}(q)|^{2}\;\le\;\Bigl(\tfrac37\Bigr)^{\lfloor N_{\ell}(q)\rfloor+1}
  \xrightarrow[q\to1^{-}]{}0
  \qquad\text{(for $q$ with $N_{\ell}(q)\ge0$),}
\]
so each of the four factors tends to $0$, and with it the ratio in (iv); (ii) then follows
from the equality of the two ratios established above. The quantitative bound is verified
numerically in \texttt{src/check\_claim\_kernels.py}, check (K5) (output
\texttt{results/check\_claim\_kernels.out}): at $q=0.9,\,0.99,\,0.999$ (with $\ell=1$,
$k_{1}=0.8$, $k_{2}=1.3$) the smallness hypothesis holds ($2(k_1+k_2)|\ln q|\le0.443$),
every per-factor ratio with $\rho_{n}\ge\tfrac12$ is $\le\tfrac37$, and the worst measured
$|r_{x}(q)|^{2}$ over the four sign choices is $2.2\times10^{-7}$ against the bound
$7.9\times10^{-2}$ at $q=0.9$, $8.3\times10^{-103}$ against $3.1\times10^{-13}$ at
$q=0.99$, and $3.2\times10^{-1066}$ against $4.8\times10^{-128}$ at $q=0.999$ (the bound is
far from tight, as expected of a counting bound); the full ratio falls to
$10^{-21419}$ by $q=0.9999$, the renormalised mixed-edge strange kernel agrees
with $\Gamma(\ell\pm ik_1\pm ik_2)/\Gamma(2\ell)$ to $4.7\times10^{-4}$ relative at $q=0.9999$,
and the reflection identity holds to better than $9\times10^{-37}$ relative at all sampled $q$,
with further scans in issue~\#10. \emph{Note}: SI's printed (4.19) writes the strange shift as
$i\pi/\ln q$ (the full period of
$\Gamma_{q^{2}}$, under which the corresponding $q$-gamma ratio would \emph{not} vanish);
consistency with their own (4.18) and \S4.3
text requires $i\pi/(2\ln q)$; see \texttt{lit/notes/2512.10101.md} \S7. \emph{Tag
bookkeeping} (issue \#18): the vanishing in (ii)/(iv) is proved in full by the product
bound above; the proposition as a whole keeps the tag \DERIVED{} because (i) and (iii) rest
on the $q$-gamma limit $\Gamma_{q^{2}}\to\Gamma$ and on Lemma~\ref{lem:mehler}, which are
complete calculations but not independently re-proved here.
\end{proof}

\subsection{Formalisations of the claim}
\label{sec:formalisations}

The informal claim admits inequivalent formalisations. We state all that we will use, from
weakest to strongest, and identify which one this project attacks.

\begin{claim}[C-amp: non-vanishing AdS$\leftrightarrow$dS amplitude; SI's literal claim]
\label{claim:camp}
Assume Definition~\ref{def:ds}(a) (edge-dS). Then the strange-series bilocals
$\mathcal{S}^{\ell}$, $\ell\in\tfrac12\Z_{>0}$, have equal-time overlap kernels between the AdS
region and the dS region with finite non-zero $q\to1$ limit (after the normalisation $N(q)$ that
renders the discrete-series AdS kernel finite), given by the Schwarzian $3j$-symbol; their
AdS$\leftrightarrow$AdS kernels vanish in the same normalisation; and for the discrete-series
bilocals the situation is reversed. \emph{Status:} \DERIVED\ --- this is
Proposition~\ref{prop:limits}\,(ii)--(iv), conditional \emph{only} on the meaning of ``dS''.
\end{claim}

\begin{claim}[C-geo: geometric interpolation; the open claim this project attacks]
\label{claim:cgeo}
There exists a radial Casimir problem on the quantum homogeneous space carrying the DSSYK
dynamics (issues \#5--\#7) --- a second-order $q$-difference operator in an invariant coordinate,
with an effective potential depending on the representation label --- together with a criterion,
fixed in advance (issue \#8), for when such a radial problem is of AdS$_2$ type and when of
dS$_2$ type; and under this criterion the strange series realises the dS$_2$ type, or crosses
from one type to the other as its label varies, while the representations with classical
analogues realise the AdS$_2$ type. \emph{Status:} \CONJ\ --- this is the precise reading of
``genuinely interpolate''; nothing in \cite{Schouten:2025dssyk} proves or refutes it, and the
authors call the interpretation of the amplitude ``currently unknown''. \emph{Resolution
(issues \#7--\#8):} refuted --- see Theorem~\ref{thm:verdict}, whose classes (I)--(III)
exhaust the readings of this claim.
\end{claim}

\begin{remark}[what would settle the question]\label{rem:settle}
Claim~\ref{claim:camp} is settled (Proposition~\ref{prop:limits}), \emph{given} its
conditionality; what remains open in it is exactly its hypothesis, Definition~\ref{def:ds}(a).
Claim~\ref{claim:cgeo} is open and falsifiable: computing the radial Casimir operator on the
space fixed in issue \#5, restricted to the strange series (issues \#6--\#7), and evaluating the
issue-\#8 criterion on it, either exhibits the AdS$\to$dS crossing (proving
Claim~\ref{claim:cgeo}), or exhibits an obstruction (refuting it). A specialist can check either
outcome against Sections \ref{sec:space}--\ref{sec:criterion}.
\end{remark}

\begin{remark}[dS-identification dependence]\label{rem:dsdependence}
\ARGUED\ Claim~\ref{claim:camp} is \emph{not} robust against replacing
Definition~\ref{def:ds}(a) by (b). At $\theta_2=\pi/2$ the four denominator arguments of
$\K^{\mathcal{S}}_{q}(\theta^{L}_{q}(k_1),\pi/2)$ tend to $\pm i$, and the standard asymptotics
$\ln(u;q^{2})_{\infty}\sim-\mathrm{Li}_2(u)/|\ln q^{2}|$ gives
\[
\K^{\mathcal{S}}_{q}(\theta^{L}_{q}(k_1),\pi/2)\,/\,
\K^{\O}_{q}(\theta^{L}_{q}(k_1),\theta^{L}_{q}(k_2))
\;\sim\;\exp\!\Bigl(-\frac{3\pi^{2}}{4\,|\ln q^{2}|}\Bigr)\longrightarrow 0
\]
(using $\mathrm{Li}_2(i)+\mathrm{Li}_2(-i)=-\pi^{2}/24$, $\mathrm{Li}_2(1)=\pi^{2}/6$):
the strange amplitude between the AdS edge and the band centre is exponentially small in the
same normalisation in which the edge-to-edge amplitude is finite. Numerical check (issue \#10
scans pending): the ratio is $5.6\times10^{-11}$ at $q=0.9$, $1.8\times10^{-153}$ at $q=0.99$,
$1.3\times10^{-313}$ at $q=0.995$; the measured rate $|\ln q^{2}|\cdot\ln(\mathrm{ratio})$ is
$-4.97,\,-7.07,\,-7.22$ against the predicted $-3\pi^{2}/4=-7.40$ (subleading corrections are
$O(|\ln q^{2}|\ln|\ln q^{2}|)$ in the exponent). So if the centre-dS identification
(b) is the correct one, the strange series does \emph{not} connect AdS to dS in the amplitude
sense --- the claim's truth value depends on which dS proposal one adopts. This
criterion-sensitivity is exactly the situation flagged in issue \#8 (``if the criterion is
ambiguous, that is itself the finding''); the quantitative treatment belongs to \#8/\#11.
\end{remark}

\begin{remark}[discreteness of the strange family]\label{rem:discreteness}
``Interpolate'' suggests a continuous deformation. Three parameters could supply it:
(1) the abstract strange label $a>0$, continuous in the classification
(Convention~\ref{conv:series}) but \emph{discrete} ($a\in\tfrac12\Z_{>0}$) in the Plancherel
decomposition of the regular representation (Convention~\ref{conv:plancherel}) --- and the
bilocal construction of SI uses $\ell\in\tfrac12\Z_{>0}$ because integer powers of the spherical
element are what exists in the coordinate algebra (their footnote on the universal cover for
continuous weights applies here too);
(2) the spectral parameter $\theta$ (continuous; this is the parameter in
Claim~\ref{claim:camp});
(3) the radial coordinate of the homogeneous space (continuous $q$-lattice; this is the
parameter in Claim~\ref{claim:cgeo}).
Any proof or refutation of Claim~\ref{claim:cgeo} must say against which parameter the
``crossing'' happens; a formalisation in which no continuous parameter crosses anything is not
an interpolation statement but a two-point statement, and must be reported as such.
\end{remark}

\begin{remark}[degenerate readings, excluded]\label{rem:degenerate}
Two readings are excluded as contentless. (a) ``The strange series are the negative-$\tCas$
counterparts of the discrete series'' is true by construction (Convention~\ref{conv:series});
without an independent geometric criterion it says nothing about AdS or dS. (b) Given
Definition~\ref{def:ds}(a) and Proposition~\ref{prop:limits}, calling the non-zero mixed-edge
kernel ``interpolation'' is definitionally true; the content resides in the hypothesis (a) and in
the \emph{selectivity} of the property --- among the bilocal families computed so far (discrete,
strange) exactly the strange one connects the edges. Whether the selectivity survives the
inclusion of principal-series bilocals is, to our knowledge, not computed anywhere; we record it
as an open sub-question for issue \#6.
\end{remark}

\subsection{The normaliser origin of the reflection identity}
\label{sec:normaliserorigin}

% Issue #15. Connects Lemma \ref{lem:reflection} to the Z_2-grading of the dual von Neumann
% algebra of GKK; establishes the fusion of strange and discrete series inside single
% irreducible corepresentations of the normaliser quantum group; and derives the covariance
% obstruction that informs issue #8. Numerical verification:
% src/check_normaliser_grading.py (results/check_normaliser_grading.out, 72/72 checks pass;
% cited as (G1)--(G7) below).

Lemma~\ref{lem:reflection} and the normaliser structure of
Convention~\ref{conv:normaliser} are two occurrences of one mechanism. This subsection makes
the connection precise: the extra factor distinguishing the strange bilocal from the discrete
bilocal is a self-adjoint unitary $\Pi$ that anticommutes with the transfer matrix ---
the chord-sector shadow of the $\Z_2$-grading of the dual von Neumann algebra of
\cite{Groenevelt:2009gkk} --- and at the unrescaled level the strange and discrete series are
\emph{fused into single irreducible corepresentations} of $\SUq\rtimes\Z_2$. The structural
consequence (Theorem~\ref{thm:covobstruction}) is an obstruction relevant to issue \#8.

\begin{lemma}[the grading unitary on the chord sector]\label{lem:gradingchord}
\PROVED\ Let $q\in(0,1)$, let $H_{\mathrm{ch}}$ be the closure of the span of the chord states
$\{\ket{k}\}_{k\in\Z_{\ge0}}$ (an orthonormal basis, Convention~\ref{conv:chord}, boundary
conditions $\eta=\xi=1$), and let $2q\hat\rho_{st}$ be the positive self-adjoint diagonal
operator $2q\hat\rho_{st}\ket{k}=q^{-2k}\ket{k}$ [SI (4.13)]. Define, by the Borel functional
calculus with the real branch of $\ln$ on $(0,\infty)$,
\begin{equation}\label{eq:Pidef}
  \Pi \coloneqq \bigl(2q\hat\rho_{st}\bigr)^{-\frac{i\pi}{2\ln q}},
  \qquad z^{-\frac{i\pi}{2\ln q}}\coloneqq\exp\Bigl(-\frac{i\pi\ln z}{2\ln q}\Bigr).
\end{equation}
Then:
\begin{enumerate}[label=(\roman*)]
\item $\Pi\ket{k}=(-1)^{k}\ket{k}$; consequently $\Pi=\Pi^{*}=\Pi^{-1}$, $\Pi^{2}=1$:
$\Pi$ is the chord-parity involution, the nontrivial character of the spectral lattice
$q^{-2\Z_{\ge0}}$ of $2q\hat\rho_{st}$.
\item $\Pi\,\tCas\,\Pi=-\tCas$ on $H_{\mathrm{ch}}$, hence $\Pi T\Pi=-T$ and
$\Pi e^{\tau T}\Pi=e^{-\tau T}$ for every $\tau\in\C$
($T=2\tCas/\sqrt{1-q^{2}}$, Convention~\ref{conv:casimir}).
\item Coefficientwise on principal eigenvectors: with
$v_{k}(\theta)\coloneqq H_{k}(\cos\theta;q^{2})/\sqrt{(q^{2};q^{2})_{k}}$
(Convention~\ref{conv:casimir}), $(-1)^{k}v_{k}(\theta)=v_{k}(\pi-\theta)$ for all
$k\ge0$ and $\theta\in[0,\pi]$; that is, $\Pi\ket{\P^{\theta}}=\ket{\P^{\pi-\theta}}$.
\end{enumerate}
\end{lemma}

\begin{proof}
(i) On the eigenvalue $z=q^{-2k}$, $\ln z=-2k\ln q$, so
$z^{-i\pi/(2\ln q)}=\exp(i\pi k)=(-1)^{k}$. The function takes values in $\{\pm1\}\subset\R$,
so $\Pi$ is a self-adjoint unitary with $\Pi^{2}=1$.
(ii) $\tCas$ is bounded on $H_{\mathrm{ch}}$: writing $\tCas=A+A^{*}$ with
$A\ket{k}=\tfrac12\sqrt{1-q^{2k}}\ket{k-1}$, $\|A\|\le\tfrac12$ gives $\|\tCas\|\le1$. On
basis vectors,
$\Pi\tCas\Pi\ket{k}
=(-1)^{k}\bigl(\tfrac12\sqrt{1-q^{2k+2}}(-1)^{k+1}\ket{k+1}
+\tfrac12\sqrt{1-q^{2k}}(-1)^{k-1}\ket{k-1}\bigr)=-\tCas\ket{k}$,
and both sides are bounded, so the identity extends to $H_{\mathrm{ch}}$ by linearity and
continuity. The exponential statement follows term by term from the (norm-convergent, since
$T$ is bounded) power series of $e^{\tau T}$: $\Pi T^{n}\Pi=(-T)^{n}$.
(iii) $H_{k}(-x;q^{2})=(-1)^{k}H_{k}(x;q^{2})$ by induction on $k$ from the recurrence
$H_{k+1}(x)=2xH_{k}(x)-(1-q^{2k})H_{k-1}(x)$: it holds for $H_{0}=1$ and $H_{1}=2x$, and if it
holds at $k-1,k$ then
$H_{k+1}(-x)=-2xH_{k}(-x)-(1-q^{2k})H_{k-1}(-x)
=(-1)^{k+1}\bigl(2xH_{k}(x)-(1-q^{2k})H_{k-1}(x)\bigr)=(-1)^{k+1}H_{k+1}(x)$. Apply this at
$x=\cos\theta$, $-x=\cos(\pi-\theta)$. (Numerical confirmation to $10^{-59}$ or better:
checks (G1)--(G3).)
\end{proof}

\begin{proposition}[factorisation of the strange bilocal; the exact content of
Lemma~\ref{lem:reflection}]\label{prop:factorisation}
\PROVED\ Let $q\in(0,1)$ and $\ell\in\tfrac12\Z_{>0}$, with
$\O^{\ell},\mathcal{S}^{\ell}$ as in Convention~\ref{conv:chord}. Then:
\begin{enumerate}[label=(\alph*)]
\item \emph{(factorisation)} As bounded operators on $H_{\mathrm{ch}}$,
\begin{equation}\label{eq:factorisation}
  \mathcal{S}^{\ell}(0)=\O^{\ell}(0)\,\Pi=\Pi\,\O^{\ell}(0),
  \qquad
  \mathcal{S}^{\ell}(\tau)=\O^{\ell}(\tau)\;e^{\tau T}\Pi e^{-\tau T}
  =\O^{\ell}(\tau)\,\Pi\,e^{-2\tau T}\quad(\tau\in\C).
\end{equation}
\item \emph{(the conjugation reading is false)} $\Pi\,\O^{\ell}(0)\,\Pi^{-1}=\O^{\ell}(0)$:
conjugation by $\Pi$ fixes every function of $\hat\rho_{st}$, and
$\Pi\O^{\ell}(0)\Pi^{-1}\neq\mathcal{S}^{\ell}(0)$ (they differ by the factor $(-1)^{k}$ on
every chord state with $k$ odd). The strange bilocal is \emph{not} the discrete bilocal
conjugated by a normaliser element; it is the discrete bilocal \emph{multiplied} by the
grading unitary $\Pi$.
\item \emph{(kernel identity through $\Pi$)} For $\theta_{1},\theta_{2}\in[0,\pi]$ the chord
sums
\begin{equation}\label{eq:kernelsums}
  \sum_{k\ge0}v_{k}(\theta_{1})\,(-1)^{k}q^{2\ell k}\,v_{k}(\theta_{2})
  \;=\;\sum_{k\ge0}v_{k}(\theta_{1})\,q^{2\ell k}\,v_{k}(\pi-\theta_{2})
\end{equation}
converge absolutely and are equal; i.e.\
$\bra{\P^{\theta_1}}\mathcal{S}^{\ell}(0)\ket{\P^{\theta_2}}
=\bra{\P^{\theta_1}}\O^{\ell}(0)\ket{\P^{\pi-\theta_2}}$ termwise. Combined with
Lemma~\ref{lem:mehler} this reproduces Lemma~\ref{lem:reflection}; the identity
\eqref{eq:kernelsums} itself is proved here without the $q$-Mehler formula.
\end{enumerate}
\end{proposition}

\begin{proof}
(a) The Borel functional calculus of the positive self-adjoint operator $2q\hat\rho_{st}$ is
multiplicative, and
$z^{-\ell-\frac{i\pi}{2\ln q}}=z^{-\ell}\cdot z^{-\frac{i\pi}{2\ln q}}$ pointwise on
$(0,\infty)$; evaluating on the spectrum, $\mathcal{S}^{\ell}(0)\ket{k}=(-1)^{k}q^{2\ell k}
\ket{k}=\O^{\ell}(0)\Pi\ket{k}$. All three operators are diagonal in the chord basis, hence
commute pairwise. The $\tau$-dependent statement follows by conjugating with $e^{\tau T}$
(bounded for all $\tau\in\C$ since $T$ is bounded) and using
$\Pi e^{-\tau T}=e^{\tau T}\Pi$ (Lemma~\ref{lem:gradingchord}(ii)).
(b) Diagonal operators commute, so $\Pi\O^{\ell}(0)\Pi^{-1}=\O^{\ell}(0)$; on $\ket{k}$ with
$k$ odd, $\O^{\ell}(0)\ket{k}=q^{2\ell k}\ket{k}\neq-q^{2\ell k}\ket{k}
=\mathcal{S}^{\ell}(0)\ket{k}$.
(c) Absolute convergence: by the explicit expansion
$H_{k}(\cos\theta;q^{2})=\sum_{j=0}^{k}\genfrac[]{0pt}{1}{k}{j}_{q^{2}}e^{i(k-2j)\theta}$
[KLS \cite{KLS2010} \S14.26] and
$\genfrac[]{0pt}{1}{k}{j}_{q^{2}}\le(q^{2};q^{2})_{\infty}^{-2}$, one has
$|H_{k}(\cos\theta;q^{2})|\le(k+1)(q^{2};q^{2})_{\infty}^{-2}$, so both series in
\eqref{eq:kernelsums} are dominated by
$\sum_{k}(k+1)^{2}(q^{2};q^{2})_{\infty}^{-5}\,q^{2\ell k}<\infty$. Termwise equality is
Lemma~\ref{lem:gradingchord}(iii) applied to $v_{k}(\theta_{2})$. (Checks (G4)--(G5):
factorisation and conjugation triviality exact; kernel sums agree to relative
$3\times10^{-60}$ and match the closed form of
$\K^{\O}_{q}(\theta_{1},\pi-\theta_{2})$ to relative $4\times10^{-60}$.)
\end{proof}

\begin{proposition}[the $\Z_2$-grading of the dual algebra; GKK]\label{prop:gkkgrading}
\PROVED\ (in \cite{Groenevelt:2009gkk}; transcription, with the dictionary of
Convention~\ref{conv:translation}; numerically verified on truncations, checks (G6).)
Let $(M,\Delta)$ be the von Neumann algebraic quantum group $\SUq\rtimes\Z_2$ and $\hat M$ its
dual, acting on the GNS space $\mathcal{K}$ of the Haar weight. Then:
\begin{enumerate}[label=(\roman*)]
\item With $\mathcal{K}_{\pm}\coloneqq\overline{\mathrm{span}}\{f_{m,p,t}:\operatorname{sgn}(pt)
=\pm\}$ and $\hat M_{\pm}\coloneqq\{x\in\hat M: x\,\mathcal{K}_{\pm}\subseteq
\mathcal{K}_{\pm(\mp)}\}$ one has $\hat M=\hat M_{+}\oplus\hat M_{-}$, and the Casimir
satisfies the graded commutation relations $x\,\Omega_{\mathrm{GKK}}\subseteq
\Omega_{\mathrm{GKK}}\,x$ for $x\in\hat M_{+}$ and $y\,\Omega_{\mathrm{GKK}}\subseteq
-\Omega_{\mathrm{GKK}}\,y$ for $y\in\hat M_{-}$ [GKK Def.\ (defKpmandMpm), Prop.\
(decompMintoM+andM-andgradedcommutation)]. Since $\tCas=-\Omega_{\mathrm{GKK}}$
(Convention~\ref{conv:translation}), the same relations hold verbatim for $\tCas$.
\item $\hat M$ is generated by $K$, $E$, $U^{+-}_{0}$, $U^{-+}_{0}$, where the two extra
generators lie in the odd part $\hat M_{-}$ and are \emph{necessary}: $E$ and $K$ alone
do not suffice, precisely because they commute with the Casimir while $\hat M$ does not
[GKK Thm.\ (generatorsforhatM), Prop.\ (QppngeneratedualM), Lemma\
(polardecompositionQppn)].
\item On the eigenvectors $e^{\epsilon,\eta}_{m}(p,x)$ of $\Omega_{\mathrm{GKK}}$ for the
eigenvalue $\epsilon\eta x$ in the sector $(p,m,\epsilon,\eta)$, the generators act by
[GKK Lemma (actionsofgeneratorsoneigvetsOm), eq.\ (introprincipalseries)]
\begin{equation}\label{eq:gkkgradingaction}
\begin{split}
  K\,e^{\epsilon,\eta}_{m}(p,x)&=p^{\frac12}q^{m}\,e^{\epsilon,\eta}_{m}(p,x),\qquad
  (q^{-1}-q)\,E\,e^{\epsilon,\eta}_{m}(p,x)=q^{-m-\frac12}p^{-\frac12}
  \sqrt{1+2\epsilon\eta xq^{2m+1}p+q^{4m+2}p^{2}}\;e^{\epsilon,\eta}_{m+1}(p,x),\\
  U^{+-}_{0}\,e^{\epsilon,\eta}_{m}(p,x)&=\eta\,(-1)^{\upsilon(p)}\,
  e^{\epsilon,-\eta}_{m}(p,x),\qquad
  U^{-+}_{0}\,e^{\epsilon,\eta}_{m}(p,x)=\epsilon\,\eta^{\chi(p)}(-1)^{m}\,
  e^{-\epsilon,\eta}_{m}(p,x),
\end{split}
\end{equation}
where $\chi(p)=\log_{q}p$ and $\upsilon(p)=k$ for $p=q^{2k}$ or $p=q^{2k-1}$. In particular
both $U^{+-}_{0}$ and $U^{-+}_{0}$ flip the sign $\epsilon\eta$, hence map
$\Omega_{\mathrm{GKK}}$-eigenvectors for $\lambda$ to eigenvectors for $-\lambda$; they
commute with $K$; on the principal corepresentation space
$\mathcal{L}_{p,x}=\bigoplus_{\epsilon,\eta}\ell^{2}_{\epsilon,\eta}(p,x)$ they are unitary
and satisfy $(U^{+-}_{0})^{2}=(U^{-+}_{0})^{2}=-1$; and conjugation by them maps the
$E$-action to the $(\epsilon\eta\mapsto-\epsilon\eta)$-twisted action, with an extra factor
$-1$ for $U^{-+}_{0}$ (the $(-1)^{m}$ in \eqref{eq:gkkgradingaction} --- the same parity
twist as $\Pi$, in the $K$-weight index). (Checks (G6a)--(G6d): all verified to
$1.5\times10^{-59}$ or exactly, at $(q,\chi(p))=(0.6,0)$ and $(0.8,1)$.)
\end{enumerate}
\end{proposition}

\begin{theorem}[fusion of strange and discrete series in the normaliser quantum group]
\label{thm:fusion}
\PROVED\ (in \cite{Groenevelt:2009gkk}, Prop.\ (discretesubrepresinregrepr) and eq.\
(sudecompdiscretesubrepresinregrepr); label computation and transcription ours, numerically
verified, checks (G7).) Let $q\in(0,1)$, $p\in q^{\Z}$, and $x=\mu(\lambda)$ with
$\lambda\in-q^{2\Z+1}p\cup q^{2\Z+1}p$, $|\lambda|>1$; define $j,l\in\Z$ by
$|\lambda|=q^{1-2j}p^{-1}=q^{1+2l}p$ (equivalently $l=-j-\chi(p)$) and
\begin{equation}\label{eq:fusionlabel}
  \ell\coloneqq j+\tfrac12\chi(p)\in\tfrac12\Z,\qquad
  q^{2\ell-1}=|\lambda|^{-1},\qquad |\lambda|>1\iff\ell\ge1 .
\end{equation}
Then $(p,x)$ determines an irreducible discrete-series corepresentation $W_{p,x}$ of
$\SUq\rtimes\Z_2$ inside the left regular corepresentation, and as a $\Uq$-module
\begin{equation}\label{eq:fusion}
  W_{p,x}\big|_{\Uq}\;\cong\;\pi^{S}_{\ell-\frac12,\,\epsilon(p)}\;\oplus\;D^{-}_{\ell}
  \;\oplus\;D^{+}_{\ell},
  \qquad \epsilon(p)=\tfrac12\chi(p)\bmod 1,
\end{equation}
with $\tCas=-\mu(q^{2\ell-1})$ on the strange summand and $\tCas=+\mu(q^{2\ell-1})$ on both
discrete summands. Moreover:
\begin{enumerate}[label=(\alph*)]
\item \emph{(exhaustiveness)} $W\cong\bigoplus_{p\in q^{\Z}}\bigl(\int_{-1}^{1}W_{p,x}\,dx
\oplus\bigoplus_{x\in\sigma_{d}(\Omega_{p})}W_{p,x}\bigr)$ [GKK Thm.\
(decompositionofleftregularcorep)], and the corepresentations in the continuous part
restrict to direct sums of two principal series [GKK Rmk.\
(propintroprincipalseriescorepresentation)]. Hence \emph{every} occurrence of a strange
series in the Plancherel decomposition of $L^{2}_{q}(\mathrm{SU}(1,1)\rtimes\Z_2)$ lies
inside a single irreducible corepresentation \eqref{eq:fusion} whose other
$\Uq$-constituents are $D^{\pm}_{\ell}$ with $a=\ell-\tfrac12$ --- \emph{exactly} the label
substitution $\ell\to\ell+\tfrac{i\pi}{2\ln q}$, $a=\ell-\tfrac12$, by which
\cite{Schouten:2025dssyk} generate the strange series (Convention~\ref{conv:series}).
\item \emph{(the odd generators realise the exchange)} In the case $x>0$ the strange
summand is the block $(\epsilon,\eta)=(+,+)$ (all $m\in\Z$), and $U^{+-}_{0}$ maps the
$D^{+}_{\ell}$ summand (block $(+,-)$, $m\ge j$) isometrically into it, while $U^{-+}_{0}$
maps the $D^{-}_{\ell}$ summand (block $(-,+)$, $m\le l$) isometrically into it; restricted
to $W_{p,x}$ they are partial isometries. The cases $x<0$ are identical with the roles of
the blocks permuted [GKK Prop.\ (discretesubrepresinregrepr), cases (ii)--(iii)].
\end{enumerate}
\end{theorem}

\begin{proof}[Proof of the label computation and the transcription]
GKK state the decomposition (in their conventions, with their labels) as
$\pi^{S}_{\frac12(\chi(p)-1)+j,\,\epsilon(p)}\oplus D^{-}_{-\frac12\chi(p)-l}\oplus
D^{+}_{\frac12\chi(p)+j}$. From $l=-j-\chi(p)$ (which follows from
$q^{1-2j}p^{-1}=q^{1+2l}p$), the $D^{-}$ label is
$-\tfrac12\chi(p)-l=j+\tfrac12\chi(p)=\ell$, equal to the $D^{+}$ label, and the strange
label is $a=\tfrac12(\chi(p)-1)+j=\ell-\tfrac12$. The Casimir values: on the strange block
$\Omega_{\mathrm{GKK}}=+|x|=\mu(q^{2a})$ with $q^{2a}=q^{2j-1}p=|\lambda|^{-1}$, and on the
discrete blocks $\Omega_{\mathrm{GKK}}=-|x|=-\mu(q^{1-2\ell})$; translation by
$\tCas=-\Omega_{\mathrm{GKK}}$ gives the values stated. The identification of the blocks
with the printed matrix elements [GKK eqs.\ (strangeserrep), (posdiscrserrep),
(negdiscrserrep)] under the index maps $n+\epsilon(p)=m+\tfrac12\chi(p)$ (strange block),
$n=m-j$ ($D^{+}$ block) and $n=l-m$ ($D^{-}$ block) is a direct computation with the
factorisation $1+2\epsilon\eta xq^{2m+1}p+q^{4m+2}p^{2}
=(1+\epsilon\eta q^{2m+1}p\lambda)(1+\epsilon\eta q^{2m+1}p\lambda^{-1})$; it is carried
out numerically to $4\times10^{-61}$ relative for all three blocks at
$(q,\chi(p),j)=(0.6,0,2),(0.8,1,1),(0.7,2,3)$ (checks (G7), including the vanishing of the
lower-edge factor $1-q^{2m+2-2j}$ at $m=j-1$, which is what makes the one-sided blocks
invariant). Part (b) is \eqref{eq:gkkgradingaction} restricted to $W_{p,x}$, where
$e^{\epsilon,-\eta}_{m}(p,x)$, resp.\ $e^{-\epsilon,\eta}_{m}(p,x)$, is zero whenever the
corresponding eigenvalue is not in the discrete spectrum of the target sector [GKK,
definition of $e^{\epsilon,\eta}_{m}(p,x)$ in Sec.\ (ssec:discreteseries)].
\end{proof}

\begin{remark}[GKK's own summary]\label{rem:gkkquote}
\cite{Groenevelt:2009gkk} state the phenomenon verbatim in their introduction: ``However, in
the quantum group case the discrete series are no longer split up into a positive discrete
series and a negative discrete series.'' Theorem~\ref{thm:fusion} is the precise form: the
normaliser fuses $D^{+}_{\ell}$, $D^{-}_{\ell}$ \emph{and} $\pi^{S}_{\ell-1/2}$ into one
irreducible corepresentation. Classically the normaliser fuses only $D^{+}_{\ell}\oplus
D^{-}_{\ell}$ (an L-packet, as \cite{Schouten:2025dssyk} \S2.3 note); the adjunction of the
strange constituent is the quantum novelty.
\end{remark}

\begin{lemma}[essential uniqueness of $\Pi$]\label{lem:uniqueness}
\PROVED\ Let $V$ be a unitary operator on $H_{\mathrm{ch}}$ with $V\tCas=-\tCas V$. Then
$V=\Pi f(\tCas)$ for some Borel function $f$ with $|f|=1$ almost everywhere with respect to
the spectral measure of $\tCas$ at $\ket{0}$. In particular $\Pi$ is the unique
self-adjoint-unitary implementation of the grading on the chord sector up to a spectral
phase.
\end{lemma}

\begin{proof}
$\Pi V$ commutes with $\tCas$: $\Pi V\tCas=-\Pi\tCas V=\tCas\Pi V$ by
Lemma~\ref{lem:gradingchord}(ii). By Proposition~\ref{prop:chordprincipal}, $\tCas$ on
$H_{\mathrm{ch}}$ is unitarily equivalent to multiplication by $x$ on
$L^{2}([-1,1],\nu)$ with $\nu$ the (absolutely continuous) $q^{2}$-Hermite orthogonality
measure and cyclic vector $1$. A bounded operator $B$ on $L^{2}(\nu)$ commuting with
multiplication by $x$ commutes with multiplication by any polynomial, hence
$B(g)=B(1)\cdot g$ first for polynomials $g$ and then, by continuity of $B$ and density,
for all $g\in L^{2}(\nu)\cap L^{\infty}(\nu)$; $h\coloneqq B(1)$ satisfies
$\|hg\|_{2}\le\|B\|\,\|g\|_{2}$ for all such $g$, which forces $|h|\le\|B\|$ a.e.\ (apply it
to indicators of the sets $\{|h|>\|B\|+\epsilon\}$), so $B=M_{h}$ with
$h\in L^{\infty}(\nu)$. Thus $\Pi V=f(\tCas)$; unitarity of $\Pi V$ gives $|f|=1$
$\nu$-a.e., and $V=\Pi f(\tCas)$ since $\Pi^{2}=1$.
\end{proof}

\begin{remark}[what is \emph{not} established: $\Pi$ as a rescaled limit]\label{rem:pilimit}
\GAP{16}\ Lemma~\ref{lem:gradingchord} and Proposition~\ref{prop:gkkgrading} identify $\Pi$
with the normaliser grading \emph{structurally}: $\Pi$ anticommutes with $\tCas$ on the chord
sector exactly as the odd part $\hat M_{-}$ anticommutes with $\tCas$ on the GNS space, and
by Lemma~\ref{lem:uniqueness} it is the essentially unique such unitary. What is not
established is the \emph{limit} identification: whether $\Pi$ is the $r\to\infty$ rescaled
limit (SI (4.2)) of a concrete element of $\hat M_{-}$ (for instance of $U^{-+}_{0}$ in the
appropriate sector, whose $(-1)^{m}$ factor in \eqref{eq:gkkgradingaction} is the same
parity twist). SI's rescaling is performed on states and on the spherical element, not on
$\hat M_{-}$; issue \#16 records this gap. No statement below depends on it.
\end{remark}

\begin{definition}[normaliser-covariant assignment]\label{def:covariant}
Let $\mathcal{R}$ be the set of irreducible admissible unitary $*$-representations of $\Uq$
(Convention~\ref{conv:series}). An \emph{assignment} is a map $C$ from $\mathcal{R}$ to an
arbitrary set of values (e.g.\ $\{\mathrm{AdS},\mathrm{dS},\ldots\}$). $C$ is
\emph{normaliser-covariant} if $C(\pi)=C(\pi')$ whenever $\pi$ and $\pi'$ occur as
$\Uq$-constituents of one and the same irreducible unitary corepresentation of
$\SUq\rtimes\Z_2$ in the left regular corepresentation.
\end{definition}

\begin{theorem}[covariance obstruction]\label{thm:covobstruction}
\PROVED\ Let $C$ be any normaliser-covariant assignment. Then for every
$\ell\in\tfrac12\Z_{\ge1}$, with $\epsilon=0$ for integer $\ell$ and $\epsilon=\tfrac12$ for
half-odd $\ell$,
\begin{equation}\label{eq:covobstruction}
  C\bigl(\pi^{S}_{\ell-\frac12,\,\epsilon}\bigr)=C\bigl(D^{+}_{\ell}\bigr)
  =C\bigl(D^{-}_{\ell}\bigr).
\end{equation}
In particular no normaliser-covariant assignment separates the strange family at the
Plancherel labels $a=\ell-\tfrac12\in\tfrac12\Z_{>0}$ from the discrete family: a criterion
that classifies the discrete series as ``AdS type'' and the strange series as ``dS type''
cannot be normaliser-covariant.
\end{theorem}

\begin{proof}
Fix $\ell$. For integer $\ell$ take $p=1$ ($\chi(p)=0$), $j=\ell$; for half-odd $\ell$ take
$p=q$ ($\chi(p)=1$), $j=\ell-\tfrac12$. In both cases $\lambda=q^{1-2j}p^{-1}=q^{1-2\ell}>1$,
so Theorem~\ref{thm:fusion} applies with $x=\mu(\lambda)$ and produces an irreducible
corepresentation $W_{p,x}$ whose $\Uq$-constituents are precisely
$\pi^{S}_{\ell-1/2,\epsilon(p)}$, $D^{-}_{\ell}$ and $D^{+}_{\ell}$; covariance forces
\eqref{eq:covobstruction}. The parenthetical range statement: by
Theorem~\ref{thm:fusion}(a) the strange labels occurring in the Plancherel decomposition are
exactly $a=\ell-\tfrac12$, $\ell\in\tfrac12\Z_{\ge1}$, i.e.\ $a\in\tfrac12\Z_{>0}$
(Convention~\ref{conv:plancherel}).
\end{proof}

\begin{corollary}[radial-problem version]\label{cor:radialparity}
\PROVED\ Let $\omega\in\R$ and consider the chord-lattice radial equation
\eqref{eq:radialrecurrence} of Section~\ref{sec:matrixcoeffs}. The parity map
$(c_{k})_{k\ge0}\mapsto((-1)^{k}c_{k})_{k\ge0}$ is an involutive linear bijection from the
solution space at $\omega$ onto the solution space at $-\omega$ which preserves absolute
values pointwise --- hence preserves square-summability of the normalised sequences, decay
and growth rates, minimality of solutions, and maps the regular solution at $\omega$ (the one
with $c_{-1}=0$) to the regular solution at $-\omega$. Consequently any AdS/dS criterion
that is a function of parity-invariant data of the radial problem assigns the same type to
the spectral values $\omega=\mu(q^{2\ell-1})$ (discrete) and $\omega=-\mu(q^{2\ell-1})$
(strange).
\end{corollary}

\begin{proof}
If $2\omega c_{k}=c_{k+1}+(1-q^{2k})c_{k-1}$ for all $k\ge1$, then $c'_{k}=(-1)^{k}c_{k}$
satisfies $2(-\omega)c'_{k}=(-1)^{k+1}2\omega c_{k}
=(-1)^{k+1}c_{k+1}+(1-q^{2k})(-1)^{k-1}c_{k-1}=c'_{k+1}+(1-q^{2k})c'_{k-1}$. The map squares
to the identity and $|c'_{k}|=|c_{k}|$; $c_{-1}=0$ iff $c'_{-1}=0$. All the listed
properties depend on the sequences only through absolute values or through the vanishing of
$c_{-1}$.
\end{proof}

\begin{remark}[reading: a candidate obstruction for Claim~\ref{claim:cgeo}, and the

\end{remark}

\begin{convention}[Deformation parameter; DSSYK dictionary]\label{conv:q}
$q$ is real, $0<q<1$. The DSSYK chord parameter is $q_{\mathrm{SYK}}\coloneqq q^{2}$
[SI \S4.2: ``under the redefinition $q^2\to q$'']; $q=e^{-\lambda/2}$, $q_{\mathrm{SYK}}
=e^{-\lambda}$ with $\lambda=2p^{2}/N$ the DSSYK coupling [SI \S4.1]. The energy of the
principal-series eigenstate $\theta\in[0,\pi]$ is $E(\theta)=2\cos\theta/\sqrt{1-q^{2}}$
[SI \S4.2]. The lower spectral edge is $\theta=\pi$, the upper $\theta=0$; the Schwarzian/AdS$_2$
regime is $\theta=\pi-|{\ln q^{2}}|k$, $q\to1$ [SI (4.17)]; the de Sitter identifications
appearing in the claim under test are recorded in Section~\ref{sec:intro}.
\end{convention}

\begin{convention}[$\Uq$]\label{conv:Uq}
Generators $E,F,K^{\pm1}$ with [SI (3.3)--(3.4)]
\begin{equation}\label{eq:ourUq}
  KK^{-1}=K^{-1}K=1,\quad KE=qEK,\quad KF=q^{-1}FK,\quad
  EF-FE=\frac{K^{2}-K^{-2}}{q-q^{-1}},
\end{equation}
$*$-structure $K^{*}=K$, $E^{*}=-F$; Hopf structure $\Delta(K)=K\otimes K$,
$\Delta(E)=K\otimes E+E\otimes K^{-1}$, $\Delta(F)=K\otimes F+F\otimes K^{-1}$,
$S(K)=K^{-1}$, $S(E)=-q^{-1}E$, $S(F)=-qF$, $\epsilon(K)=1$, $\epsilon(E)=\epsilon(F)=0$.
Classically $K=q^{H/2}$.
\end{convention}

\begin{convention}[Casimir and transfer matrix]\label{conv:casimir}
\begin{equation}\label{eq:ourCasimir}
  \Cas \coloneqq \frac{q^{-1}K^{-2}+qK^{2}-2}{(q-q^{-1})^{2}}+FE,
  \qquad
  \tCas \coloneqq \tfrac12\big((q-q^{-1})^{2}\Cas+2\big)
  =\tfrac12\big(qK^{2}+q^{-1}K^{-2}\big)+\tfrac12 (q-q^{-1})^{2}FE
\end{equation}
[SI (3.5) and \S3.3, \S3.5]. The DSSYK transfer matrix is $T=2\tCas/\sqrt{1-q^{2}}$
[SI \S4.2], acting on chord states by
$\tCas\ket{k}=\tfrac12\sqrt{1-q^{2k+2}}\,\ket{k+1}+\tfrac12\sqrt{1-q^{2k}}\,\ket{k-1}$
[SI (4.9)], with eigenvectors
$\ket{\P^{\theta}}=\sum_{k\ge0}H_{k}(\cos\theta;q^{2})/\sqrt{(q^{2};q^{2})_{k}}\,\ket{k}$
of eigenvalue $\cos\theta$ [SI (4.10)]. \DERIVED{} (check C5). We write
$\mu(y)\coloneqq\tfrac12(y+y^{-1})$ and $[z]_{q}\coloneqq(q^{z}-q^{-z})/(q-q^{-1})$
[SI (A.1)].
\end{convention}

\begin{convention}[Coordinate algebra and the normaliser extension]\label{conv:normaliser}
$\A_q(\mathrm{SU}(1,1))$: generators $\alpha,\gamma$ with relations, coproduct, counit, antipode
as in SI (3.1)--(3.2). By Woronowicz's no-go theorem there is no locally compact quantum group
$\SUq$; the extension $\SUq\rtimes\Z_2$ adds a central group-like self-adjoint $e$, $e^{2}=1$,
with $\alpha\alpha^{*}-q^{2}\gamma^{*}\gamma=e=\alpha^{*}\alpha-\gamma^{*}\gamma$ and the
$e$-twisted coproduct [SI (3.14)--(3.15)]; its irreducible $*$-representations $\pi_{\theta}$ act
on $L^{2}_{q^{2}}(I_{q^{2}})$, $I_{q}=(-\infty,-1)_{q}\cup[0,\infty)_{q}$, with position operator
$e\gamma^{*}\gamma$ [SI (3.18)]. The $\Z_2$ is the deck transformation between the two components
$\mathrm{SU}(1,1)$ and $\mathrm{SU}(1,1)v$ of the classical normaliser of $\mathrm{SU}(1,1)$ in
$\mathrm{SL}(2,\C)$ [SI (2.25)]; the extension is needed because only for the normaliser do the
self-adjoint extensions of $\Delta(e\gamma^{*}\gamma)$ admit coproduct-stable domains
(``smoothness conditions'', SI (3.19)), making a von Neumann algebraic quantum group possible.
\end{convention}

\begin{convention}[Representation series; the strange series]\label{conv:series}
Admissible $*$-representations have $K$-eigenvalues in $q^{\frac12\Z}$ with finite
multiplicities [GKK \S4.4]. The irreducible admissible unitary $*$-representations, in our
conventions (dictionary of Convention~\ref{conv:translation}):

\subsection{Candidates}
\label{sec:spacecandidates}

The phrase ``quantum AdS$_2$'' is used for inequivalent quotients in the literature; the
computation of Sections \ref{sec:matrixcoeffs}--\ref{sec:criterion} depends on which one is
meant. The candidates appearing in the sources:

\begin{definition}[quantum disk]\label{def:qdisk}
The Cartan-type quotient $\SUq/\mathrm{U}(1)$ (invariant algebra generated by the spherical
element $e\gamma^{*}\gamma$ in the normaliser setting; classically the unit disk, i.e.\
Euclidean AdS$_2$). This is the quantum bounded symmetric domain of Vaksman et al. Its
transfer-matrix eigenfunctions are Al-Salam--Chihara--type (two levels up the $q$-Askey scheme
from DSSYK). \cite{Schouten:2025dssyk} \S5 states explicitly that it is ``another (distinct from
the DSSYK coset considered in this paper) quantum homogeneous space of the von Neumann algebraic
quantum group, corresponding to its quotient by a maximal compact subgroup.'' \emph{Not the DSSYK
space.}
\end{definition}

\begin{definition}[quantum hyperboloid]\label{def:qhyperboloid}
The split-torus quotient (classically the single-sheeted hyperboloid $=$ Lorentzian AdS$_2$).
It does not occur in \cite{Schouten:2025dssyk}'s reduction; we record it only to exclude it:
no chord structure arises along this quotient in any source we know of. \emph{Not the DSSYK
space.}
\end{definition}

\begin{definition}[twisted-primitive double quotient; the DSSYK space]\label{def:dssykspace}
For $s,t$ with $|s|,|t|>q$, the ``$(s,t)$-Gauss'' double quotient of
$\mathrm{SU}_q(1,1)\rtimes\Z_2$ by the one-parameter quantum subgroups generated by the twisted
primitive elements $Y_{s},Y_{t}$ [SI (3.43)], with invariant $*$-algebra
\begin{equation}\label{eq:invariantalgebra}
  \A_q\bigl((Y_s)\backslash \mathrm{SU}(1,1)\rtimes\Z_2/(Y_t)\bigr)=\langle\rho_{st},e\rangle,
  \qquad
  \rho_{st}\coloneqq e\gamma^{*}_{s,qt}\gamma_{s,qt}-s^{-2}-q^{-2}t^{-2}
\end{equation}
[SI (3.47)--(3.48)]. Functions on the total space diagonalising the symmetrised twisted
primitives $\tilde Y^{L}_{t},\tilde Y^{R}_{s}$ [SI (3.45)] with eigenvalues $\mu(\xi),\mu(\eta)$
(quantised momenta) and $\rho_{st}$-eigenvalue $\mu(y)$ are the mixed matrix elements
$\delta^{\epsilon,(st)}_{\eta,y,\xi}$ [SI \S3.4]; classically $\rho_{-1,1}=-2+4e^{2\varphi}$, so
the $\rho_{st}$-eigenvalue is the radial/length coordinate [SI below (3.49)].
\end{definition}

\subsection{The choice, justified from the sources}

\begin{proposition}[where DSSYK lives]\label{prop:where}
\DERIVED\ (from \cite{Schouten:2025dssyk} \S4, whose derivations we follow; the chord-sector
spectral statement is proved below.) The DSSYK dynamics lives on the quotient of
Definition~\ref{def:dssykspace}, in the rescaled regime SI (4.2), as follows.
\begin{enumerate}[label=(\alph*)]
\item \emph{Two-sided (chord) sector:} the span of the chord states
$\ket{k}_{\eta,\xi}$, $y=q\eta\xi q^{2k}$, $k\in\Z_{\ge0}$ [SI (4.1), (4.5)], which sit at
$\epsilon=-1$ --- the $\mathrm{SU}(1,1)v$ component of the normaliser. On it the symmetrised
Casimir acts as the DSSYK transfer matrix (Convention~\ref{conv:casimir}); DSSYK boundary
conditions are $\eta=\xi=1$.
\item \emph{One-sided space (``reduced quantum AdS$_{2,q}$''):} fixing only the right datum
$\xi\in q^{2\Z}$ gives the restricted $q$-lattice
\begin{equation}\label{eq:ourlattice}
  \R^{2}_{q^{2}}(\xi)=\{(-q^{2n},q^{2m+1})\}_{n,m\in\Z}\cup
  \{(q^{2n},q^{2m+1}):q^{2(n+m)}\le\xi\},
\end{equation}
with points $(\chi,y)$, $\chi=-\epsilon\eta^{-1}$ [SI (4.22)], inner product
\begin{equation}\label{eq:ourinnerproduct}
  \langle f,g\rangle_{\xi}
  =\int_{-\infty}^{\infty}\!\!d_{q^{2}}\chi\int_{0}^{\infty}\!\!d_{q^{2}}y\;
  (q\xi^{-1}\chi y;q^{2})_{\infty}\,\overline{f(\chi,y)}\,g(\chi,y)
\end{equation}
[SI (4.24)], and Casimir action
\begin{equation}\label{eq:ourradialoperator}
  \Cas=\frac{q^{-1}\mathcal{T}_{q}^{-2}+q\,\mathcal{T}_{q}^{2}-2}{(q-q^{-1})^{2}}
  -\mu_{\xi}\,y\,\frac{\chi}{1-q^{2}}\,\mathcal{T}_{q}^{-2},
  \qquad \mu_{\xi}\coloneqq\frac{\xi^{-1}}{1-q^{2}},
\end{equation}
where $(\mathcal{T}_{q}f)(\chi,y)=f(\chi,qy)$ [SI (4.25)]. Classical coordinates:
$p=\tfrac12\chi/(q^{-1}-q)$, $e^{\varphi}=\tfrac12 y^{-1/2}$ [SI (4.30)]; the weight in
\eqref{eq:ourinnerproduct} is the $q$-exponential
$E_{q^{2}}(-2py\xi^{-1}(1-q^{2}))\to e^{-2py\xi^{-1}}$ under asymptotic ($p_{R}\sim\lambda^{-1}$)
boundary conditions, and $\to1$ at fixed $p_R$, recovering classical AdS$_2$ data as $q\to1$
[SI \S4.5].
\end{enumerate}
The DSSYK length dictionary: chord number $k$ with $y=q\eta\xi q^{2k}$; renormalised length
$\varphi$ with $e^{\varphi}=\tfrac12y^{-1/2}$; bare geodesic length $\ell_{\mathrm{length}}
=\lambda k$ in the triple-scaling regime, $\lambda=-2\ln q$; length positivity $=$ the lattice
restriction $q^{2(n+m)}\le\xi$ [SI \S4.1].
\end{proposition}

\begin{remark}[position space is assumed, not constructed]\label{rem:positionspace}
\cite{Schouten:2025dssyk} \S4.4 constructs the space in \emph{momentum} coordinates $(\chi,y)$
and writes: ``we assume there exists an invertible $q$-Fourier transform'' and ``an appropriate
explicit expression for the inner product in the position basis, which is compatible with the
constraint and the inner product in the momentum space, is currently not available to us.''
All computations in this project stay on the momentum-space lattice \eqref{eq:ourlattice}, where
every object is defined; nothing below uses the assumed $q$-Fourier transform. If a later issue
needs the position picture, that requires a \texttt{gap} issue first.
\end{remark}

\subsection{Where the strange series lives}
\label{sec:wherestrange}

\begin{proposition}[chord-sector states are principal only]\label{prop:chordprincipal}
\PROVED\ The closure of the span of the chord states $\{\ket{k}\}_{k\ge0}$ decomposes under the
symmetrised Casimir $\tCas$ with purely absolutely continuous spectrum $[-1,1]$ and spectral
multiplicity one; i.e.\ only principal-series Casimir values occur, and no discrete, strange, or
complementary values.
\end{proposition}

\begin{proof}
$\tCas$ acts on $\ell^{2}(\Z_{\ge0})$ as the Jacobi matrix with off-diagonal
$a_{k}=\tfrac12\sqrt{1-q^{2k}}$ and zero diagonal (Convention~\ref{conv:casimir}). It is
bounded and self-adjoint (all $|a_k|\le\tfrac12$), so its spectral measure at the cyclic vector
$\ket{0}$ is the unique orthogonality measure of the corresponding orthonormal polynomials,
which are the normalised continuous $q^{2}$-Hermite polynomials; their orthogonality measure is
$\frac{(q^{2};q^{2})_{\infty}}{2\pi}\,|(e^{2i\theta};q^{2})_{\infty}|^{2}\,d\theta$ on
$\cos\theta\in[-1,1]$ --- absolutely continuous with no mass points (Askey--Wilson $a=b=c=d=0$;
\cite{KLS2010} \S14.26). Cyclicity of $\ket{0}$ (three-term recurrence with $a_{k}>0$ for
$k\ge1$) gives multiplicity one. By Convention~\ref{conv:series}, values in $[-1,1]$ are
principal-series values.
\end{proof}

\begin{proposition}[where the strange series does occur]\label{prop:strangewhere}
\PROVED\ (in \cite{Groenevelt:2009gkk}; not re-derived here.) In the decomposition of the GNS
space of the Haar weight of $L^{\infty}_{q}(\mathrm{SU}(1,1)\rtimes\Z_2)$ as a $\Uq$-module, the
strange series occurs as discrete direct summands in the sectors $(\varepsilon,\eta)\in
\{(+,-),(-,+),(+,+)\}$ with labels $q^{2a}=pq^{1+2l}$ (Convention~\ref{conv:plancherel}).
\end{proposition}

\begin{remark}[consequence for the programme; the precise sense of ``on the space'']
\label{rem:strangeonspace}
Propositions \ref{prop:chordprincipal} and \ref{prop:strangewhere} together fix what ``matrix
coefficients of the strange series on the DSSYK space'' can mean, and this constrains issues
\#6--\#8:
\begin{enumerate}[label=(\roman*)]
\item \emph{As states of the DSSYK sector:} the strange series does \textbf{not} occur. The
chord sector is principal only. Any interpolation mechanism through strange \emph{states} of the
transfer matrix is excluded.
\item \emph{As operators (bilocals):} the strange series acts on the chord sector through
$\mathcal{S}^{\ell}$ (Convention~\ref{conv:chord}), with the kernel properties of
Section~\ref{sec:intro}.
\item \emph{As functions on the group:} strange-series matrix coefficients are elements of
$L^{2}_{q}(\mathrm{SU}(1,1)\rtimes\Z_2)$ (Proposition~\ref{prop:strangewhere}); their
$(s,t)$-reduction obeys the radial Casimir equation \eqref{eq:ourradialoperator} at spectral
value $\tCas=-\mu(q^{2a})<-1$, i.e.\ \emph{below the principal band}. The radial problem is the
same $q$-difference operator for every series; series membership enters only through the
spectral value. This is the object issues \#6--\#7 compute.
\end{enumerate}
\end{remark}

\begin{proposition}[strange series in our conventions]\label{prop:strangeexplicit}

\subsection{Radial coefficients on the chord lattice}

By Remark~\ref{rem:strangeonspace}, the meaningful ``matrix coefficients on the DSSYK space''
are the solutions of the radial Casimir equation at the strange spectral value. On the chord
lattice the radial equation is the three-term relation of Convention~\ref{conv:casimir}; in
unnormalised form ($c_{k}=v_{k}\sqrt{(q^{2};q^{2})_{k}}$):
\begin{equation}\label{eq:radialrecurrence}
  2\omega\,c_{k}=c_{k+1}+(1-q^{2k})\,c_{k-1},\qquad k\ge1,
  \tag{R}
\end{equation}
with the \emph{regular} solution fixed by $c_{1}=2\omega c_{0}$ (equivalently $c_{-1}=0$), i.e.\
$c^{\mathrm{reg}}_{k}=H_{k}(\omega;q^{2})$.

\begin{theorem}[strange radial coefficients, closed forms]\label{thm:strangeradial}
Let $0<s<1$ and $\omega=-\mu(s)<-1$ (strange values: $s=q^{2a}$, $a>0$;
Convention~\ref{conv:series}).
\begin{enumerate}[label=(\alph*)]
\item \PROVED\ \emph{(minimal solution.)} The sequence
\begin{equation}\label{eq:minimalsolution}
  c^{S}_{k}(a)\coloneqq(-1)^{k}\,q^{2ak}\;
  {}_{2}\varphi_{1}\!\left(\begin{matrix}0,\,0\\ q^{4a+2}\end{matrix};\,q^{2},\,
  q^{2k+2}\right),
  \qquad
  {}_{2}\varphi_{1}\!\left(\begin{matrix}0,0\\ c\end{matrix};q,z\right)
  =\sum_{j\ge0}\frac{z^{j}}{(q;q)_{j}(c;q)_{j}},
\end{equation}
solves \eqref{eq:radialrecurrence} at $\omega=-\mu(q^{2a})$ for all $k\ge1$, and
$c^{S}_{k}(a)=(-1)^{k}q^{2ak}\,(1+O(q^{2k}))$; the normalised sequence
$v^{S}_{k}=c^{S}_{k}/\sqrt{(q^{2};q^{2})_{k}}$ is square-summable. (Check S1$'$;
proof below.)
\item \DERIVED\ \emph{(regular solution and growth.)} The regular solution is
$c^{\mathrm{reg}}_{k}=H_{k}(-\mu(q^{2a});q^{2})=(-1)^{k}H_{k}(\mu(q^{2a});q^{2})$ with
$H_{k}(\mu(s);q^{2})\,s^{k}\,(s^{2};q^{2})_{\infty}\to1$ as $k\to\infty$; it grows like
$q^{-2ak}$ and is not square-summable --- consistently with
Proposition~\ref{prop:chordprincipal} (no strange states). (Check S2.)
\item \DERIVED\ \emph{(Wronskian and connection formula.)} With
$W_{k}\coloneqq c^{\mathrm{reg}}_{k+1}c^{\mathrm{min}}_{k}-c^{\mathrm{min}}_{k+1}
c^{\mathrm{reg}}_{k}$ one has $W_{k}=(q^{2};q^{2})_{k}\,W_{0}$ (\PROVED, one-line induction from
\eqref{eq:radialrecurrence}), and
\begin{equation}\label{eq:jost}
  W_{0}=\frac{s^{-1}-s}{(s^{2};q^{2})_{\infty}(q^{2};q^{2})_{\infty}}\;\neq\;0
  \quad\text{for all }0<s<1,
\end{equation}
equivalently the connection formula (generic $s$, $s^{2}\notin q^{2\Z}$)
\begin{equation}\label{eq:connection}
  H_{k}(\mu(s);q^{2})
  =\frac{c^{\mathrm{min}}_{k}(s^{-1})}{(s^{2};q^{2})_{\infty}}
  +\frac{c^{\mathrm{min}}_{k}(s)}{(s^{-2};q^{2})_{\infty}},
  \qquad
  c^{\mathrm{min}}_{k}(s)\coloneqq s^{k}\,
  {}_{2}\varphi_{1}\!\left(\begin{matrix}0,0\\ s^{2}q^{2}\end{matrix};q^{2},q^{2k+2}\right),
\end{equation}
verified to relative $10^{-33}$ or better at multiple $(q,s)$ (checks S3, S3$'$, S5). At the
\emph{resonant} points $s^{2}\in q^{2\Z}$ --- which include every strange label occurring in the
Plancherel decomposition, $a\in\tfrac12\Z_{>0}$ --- the branch $c^{\mathrm{min}}(s^{-1})$
degenerates and \eqref{eq:connection} requires a confluent limit; the minimal solution
\eqref{eq:minimalsolution} itself remains well-defined there.
\end{enumerate}
\end{theorem}

\begin{proof}[Proof of (a)]
Substitute $c_{k}=s^{k}\varphi(q^{2k})$, $\varphi(z)=\sum_{j\ge0}d_{j}z^{j}$, into
\eqref{eq:radialrecurrence} at $\omega=\mu(s)$ (the value $-\mu(s)$ follows by the parity map
$c_{k}\to(-1)^{k}c_{k}$, which flips the sign of $\omega$ in \eqref{eq:radialrecurrence}).
Matching coefficients of $z^{j}$ in $(s^{2}+1)\varphi(z)=s^{2}\varphi(q^{2}z)
+(1-z)\varphi(q^{-2}z)$ gives
$d_{j}\bigl[(s^{2}+1)-s^{2}q^{2j}-q^{-2j}\bigr]=-d_{j-1}q^{2-2j}$, i.e.\
$d_{j}=d_{j-1}\,q^{2}/\bigl[(1-q^{2j})(1-s^{2}q^{2j})\bigr]$, so
$d_{j}=q^{2j}/\bigl[(q^{2};q^{2})_{j}(s^{2}q^{2};q^{2})_{j}\bigr]$ with $d_{0}=1$ ---
the series \eqref{eq:minimalsolution}. For $0<s<1$ the factors $(1-s^{2}q^{2i})$ are bounded
below by $1-s^{2}q^{2}>0$, so $|d_{j}|\le C q^{2j}$ and $\varphi$ is analytic in
$|z|<q^{-2}$, which contains every lattice point $z=q^{2k}$, $k\ge0$; the term-by-term
manipulation is therefore justified, and $\varphi(q^{2k})=1+O(q^{2k})$ gives the stated
asymptotics and square-summability. (Numerical confirmation to $10^{-40}$: checks S1, S1$'$.)
\end{proof}

\begin{remark}[the $\rho$-shift]\label{rem:rhoshift}
At the discrete-series value $\omega=\mu(q^{2\ell-1})$ the minimal solution decays like
$q^{(2\ell-1)k}$ while the discrete bilocal (Convention~\ref{conv:chord}) weights chord states
by $q^{2\ell k}$; the mismatch $q^{-k}$, constant $=1$ numerically (check S4), is the usual
Weyl-vector shift between an operator weight and a wavefunction decay rate, recorded here so
that it is not mistaken for an inconsistency downstream. The same shift appears in the strange
family: bilocal weight $(-1)^{k}q^{2\ell k}$ versus minimal-solution decay
$(-1)^{k}q^{2ak}=(-1)^{k}q^{(2\ell-1)k}$ under $a=\ell-\tfrac12$.
\end{remark}

\subsection{Slice reduction: the radial operator}

\begin{proposition}[slice reduction of the momentum-space Casimir]\label{prop:slicereduction}
\PROVED\ (computation; checks (R1).) Let $q\in(0,1)$, $\xi\in q^{2\Z}$, and let $\Cas$ act on
functions on the restricted lattice $\R^{2}_{q^{2}}(\xi)$ of \eqref{eq:ourlattice} by
\eqref{eq:ourradialoperator}, with the inner product \eqref{eq:ourinnerproduct}. Then:
\begin{enumerate}[label=(\alph*)]
\item \emph{(slices)} $\Cas$ commutes with multiplication by $\chi$, hence preserves each slice
$\{(\chi, q^{2m+1})\}_{m}$; with $f_{m}\coloneqq f(\chi,q^{2m+1})$ the symmetrised Casimir
$\tCas=\tfrac12\bigl((q-q^{-1})^{2}\Cas+2\bigr)$ acts on the slice by the three-term
$q$-difference operator
\begin{equation}\label{eq:sliceoperator}
  (\tCas f)_{m}=\tfrac{q}{2}\,f_{m+1}+\tfrac{1}{2q}\bigl(1-c\,q^{2m}\bigr)\,f_{m-1},
  \qquad c\coloneqq\chi/\xi\in\pm q^{2\Z},
\end{equation}
which depends on the slice only through $c$.
\item \emph{(symmetry)} \eqref{eq:sliceoperator} is symmetric with respect to the slice weight
induced by \eqref{eq:ourinnerproduct},
$w_{m}=q^{2m+1}(cq^{2m+2};q^{2})_{\infty}$: one has $w_{m}\cdot\tfrac{q}{2}
=w_{m+1}\cdot\tfrac{1}{2q}(1-cq^{2m+2})$ for all $m$. For $c=q^{2\nu}>0$ the weight vanishes
identically for $m<m_{0}\coloneqq-\nu$ (an exactly vanishing factor of the $q$-Pochhammer):
the length-positivity truncation of Proposition~\ref{prop:where} is carried by the measure.
\item \emph{(orthonormal form)} In the orthonormalised basis $g_{m}=\sqrt{w_{m}}f_{m}$, $\tCas$
acts as the Jacobi operator
\begin{equation}\label{eq:Jc}
  (J_{c}\,g)_{m}=a_{m}\,g_{m+1}+a_{m-1}\,g_{m-1},\qquad
  a_{m}=\tfrac12\sqrt{1-c\,q^{2m+2}},
\end{equation}
with zero diagonal, on $\ell^{2}(\Z)$ for $c<0$ and on $\ell^{2}(\{m\ge m_{0}\})$ for $c>0$.
\item \emph{(two radial operators)} The unit shift $m\mapsto m+1$ intertwines $J_{c}$ with
$J_{cq^{2}}$. Hence, up to unitary equivalence, exactly \emph{two} radial operators occur on
the whole space: for every slice with $\chi\xi>0$ (the $\epsilon=-1$, chord/normaliser wedge),
$J_{+}\coloneqq J_{c>0}$ \emph{is the DSSYK transfer matrix} $\tCas$ of
Convention~\ref{conv:casimir} under $k=m-m_{0}$; and for every slice with $\chi\xi<0$ (the
$\epsilon=+1$ wedge), $J_{-}\coloneqq J_{c=-1}$, with $a_{m}=\tfrac12\sqrt{1+q^{2m+2}}$,
$m\in\Z$.
\end{enumerate}
\end{proposition}

\begin{proof}
(a) \eqref{eq:ourradialoperator} contains only $y$-shifts $\mathcal{T}_{q}^{\pm2}$ and
multiplication operators; $\tfrac12(q-q^{-1})^{2}$ times the potential term gives, at
$y=q^{2m+1}$, the coefficient $\tfrac12(q-q^{-1})^{2}\,\xi^{-1}\chi y/(1-q^{2})^{2}
=\tfrac12 q^{-2}\chi\xi^{-1}q^{2m+1}=\tfrac{1}{2q}\,cq^{2m}$, since
$(q-q^{-1})^{2}/(1-q^{2})^{2}=q^{-2}$; the kinetic part contributes
$\tfrac{q}{2}f_{m+1}+\tfrac{1}{2q}f_{m-1}$ after the $+1$ from the symmetrisation cancels the
$-f_m$. (b) $w_{m}\tfrac q2=\tfrac12 q^{2m+2}(cq^{2m+2};q^{2})_{\infty}$ and
$w_{m+1}\tfrac{1}{2q}(1-cq^{2m+2})=\tfrac12 q^{2m+2}(1-cq^{2m+2})(cq^{2m+4};q^{2})_{\infty}$,
equal by $(u;q^{2})_{\infty}=(1-u)(uq^{2};q^{2})_{\infty}$; for $c=q^{2\nu}$ and $m\le-\nu-1$
the argument $cq^{2m+2}\in q^{-2\Z_{\ge0}}$ makes a factor of $(cq^{2m+2};q^{2})_{\infty}$
vanish exactly. (c) $a_{m}=\tfrac q2\sqrt{w_{m}/w_{m+1}}=\tfrac12\sqrt{1-cq^{2m+2}}$.
(d) Substituting $m=k+m_0$, $c=q^{2\nu}$, $m_0=-\nu$ into \eqref{eq:Jc} gives
$a=\tfrac12\sqrt{1-q^{2k+2}}$, the chord coefficients; all statements are verified
numerically to $10^{-60}$ or exactly in checks (R1).
\end{proof}

\begin{remark}[the effective potential; where the representation label sits]
\label{rem:effectivepotential}
The radial ``potential'' is the single $q$-exponential coefficient
\begin{equation}\label{eq:qpotential}
  1-c\,q^{2m}=1-4\,p\,p_{R}\,(q^{-1}-q)^{2}\,y/q,\qquad y=q^{2m+1},\quad
  e^{\varphi}=\tfrac12 y^{-1/2},
\end{equation}
using SI's dictionary $p=\tfrac12\chi/(q^{-1}-q)$, $p_{R}=\tfrac12\xi^{-1}/(q^{-1}-q)$
[SI (4.30), (4.4)]: a $q$-Liouville (one-exponential) potential, the complete
$a,b,c,d\to0$ degeneration of the Askey--Wilson ($q$-P\"oschl--Teller) operator
SI (3.85) whose parameters are $(qs\eta,qt^{-1}\xi,qt\xi,qs^{-1}\eta)$
(Convention~\ref{conv:translation}). Issue \#7's task ``write the effective potential as a
function of the strange-series label'' is therefore answered in the negative direction the
data force: \emph{the potential carries no representation label}. The label enters the radial
problem only (i) as the spectral value $\omega=\tCas$-eigenvalue ($-\mu(q^{2a})$ strange,
$+\mu(q^{2\ell-1})$ discrete, $\cos\theta$ principal) --- as already recorded in
Remark~\ref{rem:strangeonspace}(iii) --- and (ii) as the boundary decay exponent of the
radial solution, $g^{(\omega)}_{m}\sim q^{1/2}\sigma^{m}/(\sigma^{2}q^{2};q^{2})_{\infty}$
as $m\to+\infty$ with $\sigma=-q^{2a}$, resp.\ $q^{2\ell-1}$
(Lemma~\ref{lem:wedgesolutions}).
\end{remark}

\subsection{The classical limit}
\label{sec:radialclassical}

\begin{theorem}[$q\to1$ contraction of the radial operator]\label{thm:radialclassical}
\DERIVED\ (checks (R2).) Fix $p,p_{R}\in\R\setminus\{0\}$ and set
$\chi=2p(q^{-1}-q)$, $\xi^{-1}=2p_{R}(q^{-1}-q)$, $\lambda=-2\ln q$, and
$e^{\varphi}=\tfrac12y^{-1/2}$ [SI (4.4), (4.30)]. Then, pointwise on smooth functions of
$\varphi$ and with deviation $O(\lambda^{2})$,
\begin{equation}\label{eq:classcontraction}
  \frac{2(J_{c}-1)}{\lambda^{2}}
  \;\longrightarrow\;
  h_{\mathrm{cl}}\coloneqq\tfrac14\,\partial_{\varphi}^{2}-p\,p_{R}\,e^{-2\varphi},
\end{equation}
the Liouville radial operator: a wall for $p\,p_{R}>0$ (the chord wedge) and an inverted
(bottomless) exponential well for $p\,p_{R}<0$ (the $\epsilon=+1$ wedge). Equivalently, in
terms of SI's classical radial Casimir action (2.26) [label
\texttt{eq:classicalCasimirAction}],
$(-\Cas_{\mathrm{cl}}f)(\varphi)=(-\tfrac14\partial^{2}_{\varphi}-\tfrac12\partial_{\varphi}
+p_{L}p_{R}e^{-2\varphi})f$ on $L^{2}(e^{2\varphi}d\varphi)$: conjugation by $e^{\varphi}$
gives the unitary equivalence $-\Cas_{\mathrm{cl}}\cong\tfrac14-h_{\mathrm{cl}}$, so
\eqref{eq:classcontraction} reproduces SI's classical radial part with the band-edge constant
$\tfrac14$ accounted for (principal eigenvalues: $-\Cas_{\mathrm{cl}}=\tfrac14+s^{2}$
$\leftrightarrow$ $h_{\mathrm{cl}}$-value $-s^{2}$). Moreover:
\begin{enumerate}[label=(\alph*)]
\item \emph{(both edges)} The contraction \eqref{eq:classcontraction} captures the upper
spectral edge ($\tCas\approx+1$); the lower (Schwarzian/AdS) edge is the \emph{same} classical
operator reached through the grading unitary: exactly, at every $q$,
$-(\tCas+1)\circ\Pi=\Pi\circ(\tCas-1)$ on each slice
(Lemma~\ref{lem:gradingchord}; check (R2), deviation $0$), so
$-2(\tCas+1)/\lambda^{2}$ on the $(-1)^{m}$-twisted sector contracts to the same
$h_{\mathrm{cl}}$.
\item \emph{(eigenfunction check)} On the wall side, $h_{\mathrm{cl}}$ has the generalised
eigenfunctions $K_{2is}\bigl(2\sqrt{pp_{R}}\,e^{-\varphi}\bigr)$ with
$h_{\mathrm{cl}}$-eigenvalue $-s^{2}$ (the Liouville/Schwarzian wavefunctions of SI (2.27));
numerically, $2(J_{c}-1)/\lambda^{2}$ applied to the sampled Bessel function returns the
eigenvalue with deviation $3.9\times10^{-3}$, $8.0\times10^{-4}$, $1.8\times10^{-4}$ at
$\lambda=0.2,0.1,0.05$ (ratios $4.9$, $4.4$; predicted rate $4$), and the operator-level
deviations on test functions have measured ratios $4.8,4.3$ ($pp_{R}>0$) and $4.3,4.2$
($pp_{R}<0$) per halving of $\lambda$ (checks (R2)).
\item \emph{(gauge caveat, recorded)} In the unnormalised function picture
\eqref{eq:sliceoperator} the same contraction produces the similarity-equivalent form
$\tfrac14(1-\partial_{\varphi})^{2}-pp_{R}e^{-2\varphi}$ but with an $O(\lambda)$ midpoint
artifact in pointwise comparisons; the gauge-invariant $O(\lambda^{2})$ statement is
\eqref{eq:classcontraction} in the orthonormal gauge. (First run of check (R2) was written in
the function gauge and FAILED with rate ratios $\approx2$; recorded per
\texttt{CONTRIBUTING.md} \S5.)
\end{enumerate}
\end{theorem}

\begin{proof}[Derivation]
With $\ln y$-derivative $D=y\partial_{y}=-\tfrac12\partial_{\varphi}$ one has
$g(q^{\pm2}y)=e^{\mp\lambda D}g$, so the kinetic part of \eqref{eq:Jc} is
$\tfrac12(e^{\lambda D}+e^{-\lambda D})g-g=\tfrac{\lambda^{2}}{2}D^{2}g+O(\lambda^{4})$,
giving $2(\cdot)/\lambda^{2}\to D^{2}=\tfrac14\partial_{\varphi}^{2}$. The potential part:
$a_{m}-\tfrac12=-\tfrac14 cq^{2m+2}+O(c^{2}q^{4m})$ with
$cq^{2m+2}=4pp_{R}(q^{-1}-q)^{2}qy=4pp_{R}\lambda^{2}y(1+O(\lambda^{2}))$, and the two
neighbours contribute $qy$ and $q^{-1}y$ whose average is $y\cosh(\lambda/2)\cdot(1+O(\lambda^2))$, so
$2(a_{m}(g_{m+1})+a_{m-1}(g_{m-1})-\text{kinetic})/\lambda^{2}\to-4pp_{R}y=-pp_{R}e^{-2\varphi}$
with $O(\lambda^{2})$ error. The unitary equivalence with SI (2.26): for $u=e^{\varphi}f$,
$e^{\varphi}(-\tfrac14\partial^{2}-\tfrac12\partial+V)e^{-\varphi}u
=(-\tfrac14\partial^{2}+\tfrac14+V)u$. (a) is Lemma~\ref{lem:gradingchord}(ii) restricted to
a slice ($J_{c}$ has zero diagonal, so $(-1)^{m}$-conjugation flips its sign); (b) is the
standard Bessel computation, $\tfrac14\partial_{\varphi}^{2}K_{2is}(ue^{-\varphi}\!\cdot)
=(u^{2}e^{-2\varphi}/4-s^{2})K_{2is}$, verified numerically as stated.
\end{proof}

\begin{remark}[comparison with the Matsuki-decomposition machinery]\label{rem:matsuki}
\ARGUED\ Issue \#7 asks whether the classical Matsuki picture of
\cite{Schlosser:2024matsuki} has a $q$-analogue here. What the computation delivers is the
$q$-analogue \emph{of its output}, not of the decomposition theory itself: classically, the
double reduction by the two parabolic (twisted-primitive limit) subgroups splits the group
into the two open regions $p_{L}p_{R}\gtrless0$, on each of which the Casimir has the radial
part SI (2.26) --- the Liouville wall/well pair; on the quantum side,
Proposition~\ref{prop:slicereduction} splits the momentum lattice into the two slice families
$\chi\xi\gtrless0$ ($\epsilon=\mp1$), carrying $J_{+}$ (wall, truncating) and $J_{-}$ (well,
non-truncating), and Theorem~\ref{thm:radialclassical} shows each contracts onto its
classical counterpart. The classical radial coordinate ranges over a continuum touching the
orbit boundaries; the $q$-lattice $\pm q^{2\Z}\times q^{2\Z+1}$ never meets the walls
$\chi=0$, $y=0$. We do \emph{not} construct a $q$-Matsuki decomposition theory (no claim
about general double cosets is made or needed: SI's quantum Gauss decomposition already
supplies the reduction); this is the stated deviation-with-justification from
\texttt{PROBLEM.md} \S4's method sketch. SI themselves note the agreement of
\eqref{eq:ourradialoperator} with the hands-on lattice Casimir of BINN
\cite{Berkooz:2022qgroup} (4.10) [SI \S4.4].
\end{remark}

\subsection{The non-chord wedge: solutions, deficiency indices, extensions}
\label{sec:wedge}

Throughout this subsection $c=-\gamma<0$ with $\gamma\in q^{2\Z}$; by
Proposition~\ref{prop:slicereduction}(d) we may and do take $\gamma=1$, i.e.\
$J_{-}$ with $a_{m}=\tfrac12\sqrt{1+q^{2m+2}}$ on $\ell^{2}(\Z)$; general $\gamma$ statements
follow by the shift. The unnormalised picture is
\begin{equation}\label{eq:wedgerecurrence}
  2\omega\,c_{m}=c_{m+1}+(1+\gamma q^{2m})\,c_{m-1},\qquad m\in\Z,
  \tag{R$'$}
\end{equation}
the analytic continuation of the chord recurrence \eqref{eq:radialrecurrence} under
$q^{2k}\mapsto-\gamma q^{2m}$, i.e.\ under the \emph{same} half-period shift
$k\mapsto m+\tfrac{\ln\gamma}{2\ln q}+\tfrac{i\pi}{2\ln q}$ that generates the strange series
(Convention~\ref{conv:series}).

\begin{lemma}[closed-form solutions on the wedge]\label{lem:wedgesolutions}
\DERIVED\ (checks (R0), (R3).) For $\sigma\in\C^{\times}$, $|\sigma|<1$, and
$\omega=\mu(\sigma)$:
\begin{enumerate}[label=(\alph*)]
\item the sequence
$c_{m}(\sigma)\coloneqq\sigma^{m}\,{}_{2}\varphi_{1}\!\left(\begin{matrix}0,0\\
\sigma^{2}q^{2}\end{matrix};q^{2},-\gamma q^{2m+2}\right)$ solves \eqref{eq:wedgerecurrence}
wherever the series converges ($\gamma q^{2m+2}<1$); the derivation of
Theorem~\ref{thm:strangeradial}(a) carries over verbatim with $q^{2k}\to-\gamma q^{2m}$;
\item by the $a\to0$, $b\to0$ confluence of Heine's transformation
[\cite{GasperRahman} (1.4.1)],
\begin{equation}\label{eq:phi11identity}
  {}_{2}\varphi_{1}\!\left(\begin{matrix}0,0\\ C\end{matrix};q,Z\right)
  =\frac{{}_{1}\varphi_{1}\!\left(\begin{matrix}Z\\ 0\end{matrix};q,C\right)}
        {(C;q)_{\infty}\,(Z;q)_{\infty}},\qquad
  {}_{1}\varphi_{1}\!\left(\begin{matrix}Z\\ 0\end{matrix};q,C\right)
  =\sum_{n\ge0}\frac{(Z;q)_{n}}{(q;q)_{n}}(-1)^{n}q^{\binom n2}C^{n},
\end{equation}
whose right side is entire in $Z$; hence
\begin{equation}\label{eq:wedgeclosed}
  c_{m}(\sigma)=\frac{\sigma^{m}\;{}_{1}\varphi_{1}\!\left(\begin{matrix}-\gamma
  q^{2m+2}\\ 0\end{matrix};q^{2},\sigma^{2}q^{2}\right)}
  {(\sigma^{2}q^{2};q^{2})_{\infty}\,(-\gamma q^{2m+2};q^{2})_{\infty}}
\end{equation}
extends $c_{m}(\sigma)$ to \emph{every} $m\in\Z$ (the denominator never vanishes for
$\gamma>0$), and $c_{m}(\sigma)=\sigma^{m}(1+O(q^{2m}))$ as $m\to+\infty$;
\item the weighted solution is
$g^{(\omega)}_{m}=q^{1/2}\sqrt{(-\gamma q^{2m+2};q^{2})_{\infty}}\;c_{m}(\sigma)$.
\end{enumerate}
The identity \eqref{eq:phi11identity} and the solution property of \eqref{eq:wedgeclosed} at
$m\le0$ are verified to $10^{-57}$ and $10^{-158}$ relative (checks (R0), (R3)).
\end{lemma}

\begin{theorem}[deficiency indices of the wedge radial operator]\label{thm:wedgedeficiency}
\PROVED\ (in \cite{Koelink:2003oneparam}; parameter identification ours; numerically
verified, checks (R4), (R5).) The operator $J_{-}$, defined on finitely supported sequences
in $\ell^{2}(\Z)$, is unbounded, symmetric, and \emph{not} essentially self-adjoint: its
deficiency indices are $(1,1)$. Specifically, $2J_{-}$ is the operator $L$ of
\cite{Koelink:2003oneparam}, case (1) of his Lemma 3.1 with $\psi=0$: his
$a_{k}=|r|^{-1}q^{-k}\sqrt{1-2rq^{k}\sin\psi+r^{2}q^{2k}}$ satisfies
$a_{k}^{2}=1+r^{-2}q^{-2k}$ at $\psi=0$, which equals our $(2a_{m})^{2}=1+\gamma q^{2m+2}$
under $e_{m}\mapsto e_{-m}$ and $r=\gamma^{-1/2}$ (the bond $(m,m+1)$ maps to his bond
$(-m-1,-m)$). The structure of the two ends:
\begin{enumerate}[label=(\alph*)]
\item \emph{(horizon end, limit circle)} As $m\to-\infty$ (i.e.\ $y\to\infty$,
$\varphi\to-\infty$) \emph{every} solution of \eqref{eq:wedgerecurrence} is
square-summable in the weighted norm, with $|g_{m}|\sim\mathrm{const}\cdot q^{|m|/2}$;
measured slopes of $\ln_{q}|g_{m}|$ over $m\in[-32,-20]$: $0.500008$ and $0.500008$ for the
two basis solutions at a strange-type $\omega$, $0.499989$, $0.499989$ at $\omega=i$
(predicted $\tfrac12$, tolerance $1.8\times10^{-3}$ from the $O(q^{|m|})$ correction scale;
checks (R4)).
\item \emph{(boundary end, limit point)} As $m\to+\infty$ the coefficients are bounded
($a_{m}\to\tfrac12$), so the half-line restriction is bounded and self-adjoint there; for
$\omega\notin[-1,1]$ exactly one solution ($c_{m}(\sigma)\sim\sigma^{m}$, $|\sigma|<1$)
decays, the other grows like $\sigma^{-m}$.
\end{enumerate}
Consequently the eigenvalue equation at $\omega=i$ has exactly one $\ell^{2}(\Z)$ solution:
deficiency indices $(1,1)$, and the self-adjoint extensions $J_{-}^{(\theta)}$ form a
$\mathrm{U}(1)$ family.
\end{theorem}

\begin{remark}[first analysis, and why it was wrong]\label{rem:deficiencyfirsttry}
Our first asymptotic analysis of the horizon end concluded limit point (a Gaussian/
anti-Gaussian solution pair), through a mis-estimated reduction-of-order sum; the correct
statement is Koelink's: in the unnormalised picture all solutions decay at the same
\emph{Gaussian} leading order $q^{m^{2}+O(m)}$ (the recurrence degenerates to a one-term
recursion downward), the weight grows like $q^{-(m+O(1))^{2}}$, and the products sit at the
common \emph{exponential} order $q^{|m|/2}$ --- both square-summable. Recorded per
\texttt{CONTRIBUTING.md} \S5; the numerical slopes above are the check that settled it.
The extension dependence is also directly visible numerically: out-of-band eigenvalues of
Dirichlet vs.\ Neumann truncations at $m=-M$ remain separated by gaps in $[0.11,0.84]$ as
$M=8,12,16$ grows (check (R5)) --- for an essentially self-adjoint operator they would
converge to each other.
\end{remark}

\begin{theorem}[spectra of the self-adjoint extensions]\label{thm:wedgespectra}
Let $J^{(\theta)}_{-}$, $\theta\in\mathrm{U}(1)$, be the self-adjoint extensions of $J_{-}$.
\begin{enumerate}[label=(\alph*)]
\item \PROVED\ (in \cite{Koelink:2003oneparam}, Thm.\ 5.8 and Rmk.\ 5.11
(arXiv labels: \texttt{thm:case1spectraldecomp}, \texttt{rmk:discspectralE}), transcribed
through the dictionary of Theorem~\ref{thm:wedgedeficiency}.) Every $J^{(\theta)}_{-}$ has
absolutely continuous spectrum $[-1,1]$ of multiplicity one, plus simple point spectrum
\begin{equation}\label{eq:twogrids}
  \sigma_{\mathrm{pp}}\bigl(J^{(\theta)}_{-}\bigr)
  =\mu\Bigl(\bigl(y_{1}(\theta)\,q^{2\Z}\cup y_{2}(\theta)\,q^{2\Z}\bigr)
  \cap\{|y|>1\}\Bigr),
\end{equation}
\emph{two bilateral $q^{2}$-grids}, unbounded in both directions, determined by the two
roots $y_{1},y_{2}$ (per fundamental cell) of the equation $g(y)=\mathrm{const}(\theta)$
for an explicit elliptic function $g$ of order two (a quotient of two theta functions;
proof of (b) below); the eigenfunction at
$\omega=\mu(y_{i}q^{2j})$ is the explicit $g^{(\omega)}$ of
Lemma~\ref{lem:wedgesolutions}(c).
\item \DERIVED\ \emph{(root product rule.)} The two roots obey
\begin{equation}\label{eq:productrule}
  y_{1}\,y_{2}\in-\gamma\,q^{2\Z} .
\end{equation}
Consequently $y_{1}$ and $y_{2}$ are real of \emph{opposite signs}: every self-adjoint
extension carries one infinite \emph{discrete-type} tower ($\tCas=\mu(y)>1$) and one
infinite \emph{strange-type} tower ($\tCas=\mu(y)<-1$), log-periodically locked by
\eqref{eq:productrule}. In particular, on the $\gamma=1$ slice the extension whose spectrum
contains the strange value $-\mu(q^{2a_{0}})$ ($a_{0}>0$) has
\begin{equation}\label{eq:extensionspectrum}
  \sigma_{\mathrm{pp}}=\bigl\{-\mu(q^{2a_{0}+2j})\bigr\}_{j\in\Z,\;q^{2a_{0}+2j}<1}
  \;\cup\;\bigl\{+\mu(q^{2j-2a_{0}})\bigr\}_{j\in\Z,\;q^{2j-2a_{0}}<1}
\end{equation}
(the two parametrisations $\mu(y)=\mu(1/y)$ are used with arguments inside $(0,1)$, so the
$|y|>1$ window of \eqref{eq:twogrids} is automatic).
\item \DERIVED\ \emph{(numerical verification.)} With the bilinear eigenvector test
$h(\omega)=\sum_{m}\hat g^{(\omega_{0})}_{m}\hat g^{(\omega)}_{m}$ (two real normalisable
solutions belong to one extension iff $h=0$; Green's identity), at $q=0.6$, $\gamma=1$:
for $a_{0}=\tfrac12$, $|h|$ at the four predicted grid points
$-\mu(q^{3}),-\mu(q^{5}),+\mu(q),+\mu(q^{3})$ is
$7.8\times10^{-42}$, $1.3\times10^{-41}$, $1.4\times10^{-41}$, $7.8\times10^{-42}$
(the numerical floor of the $m$-cutoffs), against off-grid controls $0.209$ and $0.097$;
for generic $a_{0}=0.37$, the strange grid point and the two product-rule partners give
$|h|\le1.3\times10^{-41}$ while the \emph{naive mirror points} $+\mu(q^{2a_{0}})$,
$+\mu(q^{2a_{0}+2})$ give $0.043$, $0.063$: the partner grid is
\eqref{eq:extensionspectrum}, not the mirror. (Checks (R6); grid persistence at $q=0.9$,
check (R7).)
\end{enumerate}
\end{theorem}

\begin{proof}[Proof of (b)]
In Koelink's variables (case (1), $\psi=0$, $a=q^{1/2}$, $t=ir$, $r=\gamma^{-1/2}$, base
$q$ with our $q^{2}$-grids arising as his ``$q^{2}$-quadratic grids'') the point-spectrum
condition is $A\,\theta(yq^{1/2}ir)+\bar A\,\theta(-yq^{1/2}ir)=0$ with
$\theta(x)=(x,q/x;q)_{\infty}$ and $A=A(\theta)$ [\cite{Koelink:2003oneparam}, Rmk.\ 5.11,
display \texttt{eq:weightE1}]. Writing $y=e^{2\pi iw}$, $q=e^{\pi i\tau}$,
the function $g(w,\tau)=\theta(e^{2\pi iw}q^{1/2}ir)/\theta(-e^{2\pi iw}q^{1/2}ir)$ is
elliptic of order $2$ with periods $1,\tau$ (i.e.\ $y\mapsto yq^{2}$), and the condition is
the level set $g(w,\tau)=-\bar A/A$. By Abel's theorem the two roots $w_{1},w_{2}$ of a
level set of an elliptic function satisfy $w_{1}+w_{2}\equiv$ (sum of the poles) mod the
period lattice; the poles of $g$ are at $-yq^{1/2}ir\in q^{\Z}$, two per cell, with product
$y_{p1}y_{p2}=(-q^{-1/2}/ir)(-q^{1/2}/ir)=-1/r^{2}=-\gamma$; hence
$y_{1}y_{2}\equiv-\gamma\bmod q^{2\Z}$. A complex-conjugate root pair would have
$y_{1}y_{2}=|y_{1}|^{2}>0$, contradicting \eqref{eq:productrule}; so both roots are real, of
opposite signs. \eqref{eq:extensionspectrum} is \eqref{eq:productrule} solved for $y_{2}$
at $y_{1}=-q^{2a_{0}}$.
\end{proof}

\subsection{The strange series in the radial problem, and the $q\to1$ limit}

\begin{theorem}[where the strange values sit radially]\label{thm:radialstrange}
\DERIVED\ (from Theorems \ref{thm:wedgedeficiency}--\ref{thm:wedgespectra}; checks (R6).)
On the DSSYK space, the strange Casimir values $\omega=-\mu(q^{2a})<-1$ occur in the radial
problem exactly as follows.
\begin{enumerate}[label=(\roman*)]
\item On the chord wedge ($\chi\xi>0$): never (Proposition~\ref{prop:chordwedge}).
\item On the $\epsilon=+1$ wedge ($\chi\xi<0$): for \emph{every} $a>0$ the explicit
radial solution $g^{(-\mu(q^{2a}))}$ of Lemma~\ref{lem:wedgesolutions} is normalisable
(geometric decay $|\sigma|^{m}=q^{2am}$ at the boundary end, $q^{|m|/2}$ at the horizon

$J^{(\theta(a))}_{-}$; within that extension it comes locked to the discrete-type tower
\eqref{eq:extensionspectrum}. The discrete values $+\mu(q^{2\ell-1})$ enter in the
mirror-symmetric way. The strange and discrete families are therefore \emph{exactly on a
par} in the radial problem; no radial datum distinguishes them except the sign of
$\tCas$ --- the concrete realisation of Corollary~\ref{cor:radialparity} and
Theorem~\ref{thm:covobstruction} announced there.
\end{enumerate}
\end{theorem}

\begin{proposition}[$q\to1$: how the strange towers disappear]\label{prop:radialq1}
\DERIVED\ (checks (R7).) Fix $a>0$ and let $q\to1^{-}$.
\begin{enumerate}[label=(\roman*)]
\item The strange eigenvalue $-\mu(q^{2a})\to-1^{-}$: its distance to the continuum
$[-1,1]$ is $\mu(q^{2a})-1=2\sinh^{2}(a\lambda/2)=\tfrac{a^{2}\lambda^{2}}{2}+O(\lambda^{4})$
--- the strange towers merge into the band edge and no isolated spectrum survives at $q=1$
at fixed label. The strange \emph{representation} disappears (its Casimir parameter leaves
the strange range $(-\infty,-1)$), as required by \texttt{PROBLEM.md}.
\item In lower-edge (Schwarzian) units, $E^{\mathrm{low}}\coloneqq2(1+\tCas)/\lambda^{2}$
on the $\Pi$-twisted sector (Theorem~\ref{thm:radialclassical}(a)): the continuum becomes
$[0,\infty)$ ($E^{\mathrm{low}}\to k^{2}$) and the strange tower survives at
\begin{equation}\label{eq:strangetower}
  E^{\mathrm{low}}\bigl(-\mu(q^{2a_{0}+2j})\bigr)
  =\frac{2\bigl(1-\mu(q^{2a_{0}+2j})\bigr)}{\lambda^{2}}
  \longrightarrow -(a_{0}+j)^{2}:
\end{equation}
the bound-state tower of the classical inverted-Liouville well. Its $-\Cas_{\mathrm{cl}}$
values are $\tfrac14-(a_{0}+j)^{2}=\ell_j(1-\ell_j)$ with $\ell_j=a_{0}+j+\tfrac12$: the
\emph{classical discrete-series} Casimir values, under the fusion label map
$a=\ell-\tfrac12$ once more. Measured deviations from $-(a_{0}+j)^{2}$ at
$a_{0}=\tfrac12$: $(-2.3\times10^{-4},-1.9\times10^{-2},-1.5\times10^{-1})$ for $j=0,1,2$
at $q=0.9$ and $(-2.1\times10^{-6},-1.7\times10^{-4},-1.3\times10^{-3})$ at $q=0.99$;
deviation ratio $109.9$ against the predicted $(\ln0.9/\ln0.99)^{2}=109.9$.
\item At the same (lower) edge the \emph{discrete} tower is expelled:
$E^{\mathrm{low}}(+\mu(q^{2\ell-1}))=2(1+\mu(q^{2\ell-1}))/\lambda^{2}
=4/\lambda^{2}+(\ell-\tfrac12)^{2}+O(\lambda^{2})\to+\infty$ (measured: $90.33$ vs
$4/\lambda^{2}=90.08$ at $q=0.9$; $9900.33$ vs $9900.08$ at $q=0.99$). At the upper edge
the roles are exactly exchanged ($\Pi$-conjugation). So each classical edge theory keeps
one tower --- which becomes its ordinary discrete series --- and expels the mirror tower to
infinite energy; the coexistence of both towers in one self-adjoint radial operator is a
finite-$q$ ($\lambda>0$) phenomenon with no classical counterpart.
\item \ARGUED\ \emph{(a quantum-only window.)} The classical well
$\tfrac14\partial^{2}_{\varphi}-pp_{R}e^{-2\varphi}$, $pp_{R}<0$, on
$L^{2}(e^{2\varphi}d\varphi)$ admits bound states only for $\kappa>\tfrac12$
(normalisability of $e^{(2-4\kappa)\varphi}$ at $\varphi\to+\infty$, $E=-\kappa^{2}$),
matching the classical unitarity bound $\ell=\kappa+\tfrac12>1$; the $q$-side towers
\eqref{eq:strangetower} include $\kappa=a_{0}+j\in(0,\tfrac12)$ members whenever
$a_{0}<\tfrac12$, whose limits fall below that bound: eigenvalues with no classical
counterpart even edge-locally. Not needed downstream; recorded for issue \#8.
\end{enumerate}
\end{proposition}

\begin{remark}[summary for the criterion (issue \#8)]\label{rem:radialsummary}
The radial part of the Casimir on the DSSYK space is now explicit: the chord wedge carries
the transfer matrix (principal continuum only); the $\epsilon=+1$ wedge carries the
$q$-Liouville well operator $J_{-}$, whose self-adjoint realisations each hold one strange
and one discrete tower, locked by the product rule \eqref{eq:productrule}, collapsing at
$q\to1$ onto the classical discrete series of the respective edge with the mirror tower
expelled. Everything is exactly $\Pi$-covariant; the only non-covariant datum remains the
orientation of the spectral axis (which edge is ``low''), and the only unresolved freedom
is the extension choice (issue \#26), whose conjectured resolution
\eqref{eq:fusionspectrum} is itself $\Pi$-invariant. Any AdS/dS criterion built from the
radial problem therefore faces the obstruction of Theorem~\ref{thm:covobstruction} in the
sharpened form: it must break a symmetry that the radial operator, its extension family,
its explicit eigenfunctions, and its classical limits all respect.
\end{remark}

\label{sec:criterion}

\subsection{Criterion classes and the candidate list}

\begin{definition}[criterion classes]\label{def:criterionclasses}
An AdS/dS criterion is an assignment $C$ from the irreducible admissible unitary
$*$-representations of $\Uq$ (equivalently, from their occurrences in the radial problem of
Section~\ref{sec:radial}) to values in $\{\mathrm{AdS},\mathrm{dS},\ldots\}$, built from:
\begin{enumerate}[label=(\Roman*)]
\item \emph{intrinsic data}: the quantum group, the DSSYK space, the radial operators, their
extension theory, eigenfunctions and classical limits --- i.e.\ normaliser-covariant data in
the sense of Definition~\ref{def:covariant};
\item intrinsic data \emph{plus the orientation}: the identification of the lower spectral
edge $\theta=\pi$ as the ground-state (Schwarzian/AdS) side --- intrinsic to DSSYK as a
thermal theory ($e^{-\beta T}$ weights low energy; Definition~\ref{def:ads}), and the one
datum in the model that is odd under the grading $\Pi$;
\item class (II) data \emph{plus an external bulk proposal} --- the edge-dS or centre-dS
identifications of Definition~\ref{def:ds}.
\end{enumerate}
The candidates fixed in issue \#8: (a) the sign of the effective curvature extracted from the
radial operator; (b) the sign of the exponential term in the ($q$-)P\"oschl--Teller
potential; (c) the character of the spectrum (continuous vs.\ discrete, normalisable vs.\
delta-normalisable); (d) reality/unitarity conditions on the representation label; (e) the
signature of the invariant metric on the quantum homogeneous space.
\end{definition}

\subsection{Assessment of the candidates}

\begin{proposition}[candidates (a)--(d) cannot validate Claim~\ref{claim:cgeo}]
\label{prop:criterioncases}
\PROVED\ (from Section~\ref{sec:radial} and Theorem~\ref{thm:covobstruction}.) Each of the
criteria (a)--(d), evaluated on the DSSYK radial problem, is a normaliser-covariant
assignment; by Theorem~\ref{thm:covobstruction} it gives
$C(\pi^{S}_{\ell-1/2})=C(D^{\pm}_{\ell})$ and therefore cannot classify the strange family as
dS against an AdS discrete family. Case by case:
\begin{enumerate}[label=(\alph*)]
\item \emph{Effective curvature.} The radial potential \eqref{eq:qpotential} carries no
representation label (Remark~\ref{rem:effectivepotential}); its classical limits on the two
wedges are the AdS$_2$ Liouville wall and well of SI (2.26)
(Theorem~\ref{thm:radialclassical}) --- both wedges are regions of the \emph{same} classical
AdS$_{2}$ reduction. No curvature-like coefficient anywhere in the radial family depends on
which representation is inserted: criterion (a) is constant on representations.
\item \emph{Potential sign.} The sign of the exponential term is
$\operatorname{sgn}(\chi\xi)$ --- a wedge datum (which slice), not a representation datum.
Constant on representations.
\item \emph{Spectral character.} Strange and discrete values enter identically: simple
isolated eigenvalues of the same wedge extensions, in towers exchanged by $\Pi$ with all
normalisability data preserved (Theorem~\ref{thm:wedgespectra},
Corollary~\ref{cor:radialparity}); principal values are the common continuum. The one
apparent intrinsic asymmetry --- strange radial solutions alternate at lattice scale,
$c_{m}\sim(-1)^{m}q^{2am}$, so they have no continuum limit as functions --- is precisely
the parity twist, i.e.\ it is \emph{equivalent to} $\operatorname{sgn}\tCas$, whose reading
is the orientation question of class (II), not an intrinsic discriminator.
\item \emph{Label reality/unitarity.} Both families are unitary at every label
(Convention~\ref{conv:series}; the strange series for \emph{all} $a>0$), and the fusion
theorem pairs them label-by-label inside single irreducible corepresentations
(Theorem~\ref{thm:fusion}). Covariant, hence obstructed.
\end{enumerate}
\end{proposition}

\begin{remark}[candidate (e): not constructible in the present framework]
\label{rem:metriccriterion}
A metric-signature criterion cannot be evaluated: the DSSYK space is constructed in momentum
coordinates, the position picture rests on an assumed $q$-Fourier transform
(Remark~\ref{rem:positionspace}), and a metric requires the spectral-triple layer that
\cite{Schouten:2025dssyk} explicitly defer: ``the natural metric structure provided by a
spectral triple should shed light on the intriguing overlap of correlation functions between

\texttt{gap} in our argument); but it is the one listed candidate that could in principle
carry grading-breaking structure, and Conjecture~\ref{conj:residue} records what it would
have to do.
\end{remark}

\subsection{The oriented data}

\begin{proposition}[what the orientation separates]\label{prop:orientation}
\DERIVED\ (arithmetic from Section~\ref{sec:radial}; checks (R7).) Relative to the DSSYK
orientation (class (II)): on every wedge extension $J^{(\theta)}_{-}$, the strange tower
lies \emph{below the lower band edge} --- at energies $E=2\tCas/\sqrt{1-q^{2}}$ a distance
\begin{equation}\label{eq:bindingenergy}
  \Delta E_{a}=\frac{2\bigl(\mu(q^{2a})-1\bigr)}{\sqrt{1-q^{2}}}
  =\frac{4\sinh^{2}(a\lambda/2)}{\sqrt{1-e^{-\lambda}}}
\end{equation}
below the chord-sector ground-state energy $-E_{\max}$ --- and the discrete tower
symmetrically above $+E_{\max}$. As $q\to1$ the strange tower survives at the AdS
(ground-state) edge as the classical discrete-series bound states of the inverted-Liouville
well, and is expelled from the opposite edge; the discrete tower has the mirror fate
(Proposition~\ref{prop:radialq1}). The orientation therefore \emph{does} break the grading
and \emph{does} separate the two families --- but it attaches the \emph{strange} family to
the AdS side and the \emph{discrete} family to the opposite edge. (These towers live on the
$\epsilon=+1$ wedge of the reduced space, not in the chord sector, whose spectrum remains
exactly the band $[-E_{\max},E_{\max}]$: Proposition~\ref{prop:chordprincipal}; there is no
contradiction with DSSYK spectral data.)
\end{proposition}

\subsection{Verdict}

\begin{theorem}[verdict on Claim~\ref{claim:cgeo}]\label{thm:verdict}
Under the criterion classes of Definition~\ref{def:criterionclasses}:
\begin{enumerate}[label=(\roman*)]
\item \PROVED\ \emph{(class (I): refuted.)} No intrinsic criterion validates
Claim~\ref{claim:cgeo}: the listed candidates (a)--(d) fail by
Proposition~\ref{prop:criterioncases}, and \emph{any} intrinsic criterion is
normaliser-covariant, hence obstructed by Theorem~\ref{thm:covobstruction}; candidate (e)
is not constructible (Remark~\ref{rem:metriccriterion}).
\item \DERIVED\ \emph{(class (II): refuted with an inversion.)} The orientation separates
the families, but assigns the strange family to the \emph{AdS} side
(Proposition~\ref{prop:orientation}): under the edge-dS reading of the opposite edge, the
dS-edge-local family is the \emph{discrete} series. An oriented criterion therefore not
only fails to validate ``the strange series realises the dS$_{2}$ type'' --- it supports
the opposite assignment.
\item \DERIVED\ \emph{(class (III): reduces to Claim~\ref{claim:camp}.)} Under edge-dS,
what is true is exactly Proposition~\ref{prop:limits}: the strange bilocals are the
edge-crossing family (finite AdS$\leftrightarrow$dS amplitudes, vanishing
AdS$\leftrightarrow$AdS ones, the reverse for discrete bilocals) --- with the selectivity
refined on the complex-weight family by the transition of Theorem~\ref{thm:transition}
(Section~\ref{sec:transition}), and still open only for the non-spherical bilocals
(issue \#13, Remark~\ref{rem:degenerate}).
Under centre-dS, the strange AdS$\leftrightarrow$dS amplitude vanishes
(Remark~\ref{rem:dsdependence}) and no spectral statement singles out the band centre: the
claim fails in every class.
\item \DERIVED\ \emph{(No crossing.)} In no class does any quantity \emph{cross} as the strange label
$a$ varies: the eigenvalue moves monotonically along $(-\infty,-1)$, the bilocal amplitudes
vary smoothly, and the only distinguished labels are the resonances $2a\in\Z$ (Plancherel
labels, $\Pi$-invariant extensions, connection-formula degeneration
(Theorem~\ref{thm:strangeradial}(c))) --- structure conditions, not sign changes. The
``interpolation'' of Claim~\ref{claim:cgeo}, read as a crossing against any continuous
parameter (Remark~\ref{rem:discreteness}), does not occur.
\end{enumerate}
\emph{Conclusion for the epic.} Claim~\ref{claim:cgeo} is \textbf{refuted} as an
intrinsic-geometry statement, by the explicit obstruction now realised in the radial
problem; the defensible content of ``the strange series interpolates between AdS and dS''
is Claim~\ref{claim:camp} (conditional on edge-dS), \emph{together with} the structural
statement that the strange and discrete families form a single $\Pi$-orbit --- one fused
irreducible corepresentation of the normaliser (Theorem~\ref{thm:fusion}), one locked pair
of spectral towers per radial extension (Theorem~\ref{thm:wedgespectra}), the two towers
edge-local at the two ends of the spectrum with the strange component on the AdS side and
the cross-edge amplitudes carried by the strange bilocals. The AdS$\leftrightarrow$dS
bridge, to the extent the framework supports one, is the \emph{grading} $\Pi$; the strange
series is one half of its orbit, not a family that migrates from one geometry to the other.
\end{theorem}

\begin{claim}[the falsifiable residue]\label{conj:residue}
\CONJ\ If the spectral-triple metric structure announced in \cite{Schouten:2025dssyk} \S5

\subsection{The complex-weight bilocal family: an exact edge-connectivity transition}
\label{sec:transition}

% Issue #27 (sub-issue of #13). Numerical verification:
% src/check_principal_bilocals.py (results/check_principal_bilocals.out, all checks pass;
% cited as (P1)-(P5)).

The selectivity load on Theorem~\ref{thm:verdict}(iii) can be discharged for the largest
family the framework makes computable: the \emph{complex-weight} spherical-power bilocals.

\begin{lemma}[complex-weight bilocals; exact reflection]\label{lem:complexweight}
\PROVED\ (checks (P1).) For $w\in\C$ with $\operatorname{Re}w>0$, the operator
$(2q\hat\rho_{st})^{-w}$ is a bounded diagonal bilocal with chord weights $t^{k}$,
$t\coloneqq q^{2w}$, $0<|t|<1$, and its equal-time kernel between principal states is the
$q$-Mehler kernel \eqref{eq:qmehler} at complex $t$ (absolutely convergent; both sides of
\eqref{eq:qmehler} are analytic in $t$ on the unit disk). The reflection identity holds for
\emph{all} such $t$: $\K_{q}(t;\theta_{1},\theta_{2})=\K_{q}(-t;\theta_{1},\pi-\theta_{2})$,
by the proof of Lemma~\ref{lem:reflection} verbatim; hence the normalisation-free
edge ratio
\begin{equation}\label{eq:edgeratio}
  \mathrm{Rat}(t)\coloneqq
  \frac{\K_{q}(t;\theta^{L}_{q}(k_1),\theta^{U}_{q}(k_2))}
       {\K_{q}(t;\theta^{L}_{q}(k_1),\theta^{L}_{q}(k_2))}
  =\prod_{j=1}^{4}
  \frac{(t e^{i\varphi_{j}};q^{2})_{\infty}}{(-t e^{i\varphi_{j}};q^{2})_{\infty}},
  \qquad \varphi_{j}\in\{\pm\lambda_2(k_1{+}k_2),\pm\lambda_2(k_1{-}k_2)\},
\end{equation}
$\lambda_2=|\ln q^{2}|$, satisfies $\mathrm{Rat}(t)\,\mathrm{Rat}(-t)=1$ identically. On
the principal-weight circle $t=q^{1+2ib}$ ($\Delta=\tfrac12+ib$;
$b\in[0,b_{\max}]$, $b_{\max}=-\pi/(2\ln q)$, the parameter window of the principal
series), the reflection $b\mapsto b_{\max}-b$ sends $t\mapsto-\bar t$, so
$|\mathrm{Rat}|(b_{\max}-b)=1/|\mathrm{Rat}|(b)$: the family is exactly antisymmetric
about its midpoint.
\end{lemma}

\begin{theorem}[the edge-connectivity transition]\label{thm:transition}
Fix $k_{1},k_{2}\ge0$ and $\delta\in(0,\tfrac\pi2)$; let $t=\rho_{0}e^{i\alpha}$ with
$\rho_{0}\in[\tfrac12+\delta,\,1)$, and let $q\to1^{-}$.
\begin{enumerate}[label=(\roman*)]
\item \PROVED\ \emph{(suppressed half-disk.)} If $|\alpha|+\lambda_2(k_{1}+k_{2})\le
\tfrac\pi2-\delta$, then $|\mathrm{Rat}(t)|\to0$: each factor of \eqref{eq:edgeratio} obeys
the counting bound of Proposition~\ref{prop:limits}(iv) with
$c_{0}=\cos(|\alpha|+\lambda_2(k_1{+}k_2))\ge\sin\delta$ in place of $\tfrac12$ (the
per-factor ratio $g_{c}(\rho)=(1-2\rho c+\rho^{2})/(1+2\rho c+\rho^{2})$ is decreasing in
$\rho$ for $c>0$, derivative numerator $2c(\rho^{2}-1)<0$, so $g_{c_0}(\rho_{n})\le
g_{c_0}(\tfrac12)<1$ on the $\to\infty$ many sites with $\rho_{n}=\rho_{0}q^{2n}\ge\tfrac12$).
Quantitatively, $\ln|\mathrm{Rat}|=\sum_{j}\mathrm{Re}\bigl[\mathrm{Li}_2(-te^{i\varphi_j})
-\mathrm{Li}_2(te^{i\varphi_j})\bigr]/\lambda_2+\tfrac12\sum_{j}
\ln\bigl|\tfrac{1-te^{i\varphi_j}}{1+te^{i\varphi_j}}\bigr|+O(\lambda_2^{2})$; at
$|t|\to1$ the leading exponent per factor is $-(\tfrac{\pi^{2}}{4}-\tfrac{\pi|\alpha|}{2})
/\lambda_2$. Measured at $\alpha=\pi/4$, $\rho_0=q$: $\ln|\mathrm{Rat}|=-23.4186$, $-81.0067$,
$-245.504$ against predictions $-23.3979$, $-81.0058$, $-245.504$ at $q=0.9,0.97,0.99$;
residual scaling exponent measured $\approx2$ in $\lambda_2$ (checks (P2)).
\item \PROVED\ \emph{(enhanced half-disk.)} If $|\alpha|-\lambda_2(k_1{+}k_2)\ge\tfrac\pi2
+\delta$, then $|\mathrm{Rat}(t)|\to\infty$, reciprocally to the mirror point:
$\mathrm{Rat}(t)=1/\mathrm{Rat}(-t)$ exactly. (Measured: $|\mathrm{Rat}|=1.5\times10^{10}$
at $q=0.9$, $4.2\times10^{106}$ at $q=0.99$ for $\alpha=3\pi/4$; checks (P3).) The strange
ray $\alpha=\pi$ ($t=-q^{2\ell}$) is the boundary ray of maximal enhancement at given
$|t|$.
\item \PROVED\ \emph{(the marginal line is exact.)} For $t=i\rho$, $\rho\in(0,1)$,
\emph{at every} $q$:
$|\mathrm{Rat}(i\rho)|=1$ --- the cross-edge and same-edge kernels have exactly equal
modulus. Proof: the phase set $\{\varphi_{j}\}$ is symmetric, and for each pair
$(\varphi,-\varphi)$ and each $n\ge0$,
$|1\mp\rho_{n}e^{i(\pi/2+\varphi)}|^{2}\,|1\mp\rho_{n}e^{i(\pi/2-\varphi)}|^{2}
=(1\pm2\rho_{n}\sin\varphi+\rho_{n}^{2})(1\mp2\rho_{n}\sin\varphi+\rho_{n}^{2})$ is the
same for both signs, so numerator and denominator products in \eqref{eq:edgeratio} have
equal modulus factor by factor. (Verified to $5\times10^{-41}$; checks (P4).)
\item \PROVED\ \emph{(dictionary: the crossing sits exactly at the band centre.)} On the
principal-weight circle $t=q^{1+2ib}$ the point $b=b_{\max}/2$ has $t=-iq$, on the marginal
line; hence $|\mathrm{Rat}|=1$ there \emph{exactly}, with
$\ln|\mathrm{Rat}|<0$ below and $>0$ above (antisymmetric by
Lemma~\ref{lem:complexweight}; measured $\mp9.367$, $\mp32.4$, $\mp98.2$ at
$0.4\,b_{\max}$/$0.6\,b_{\max}$ for $q=0.9,0.97,0.99$; checks (P5)). The $\theta$-image of
the crossing weight ($\theta=\pi+2b\ln q$ at $b=b_{\max}/2$) is
$\theta=\pi/2$ --- \emph{the band centre, at every $q$}. (A first version of check (P5)
searched for an asymptotic drift of the crossing towards the midpoint and found none:
the measured drift equalled bisection precision at every $q$; the crossing is exact,
not asymptotic. Recorded per \texttt{CONTRIBUTING.md} \S5.)
\end{enumerate}
\end{theorem}

\begin{remark}[reading of the transition]\label{rem:transitionreading}
\ARGUED\ Three consequences.
(1) \emph{Binary selectivity fails for the extended family}: the edge-connecting set of
the weight disk is the open half $|\arg t|>\pi/2$, with the strange ray as its extreme;
Claim~\ref{claim:camp}'s selectivity statement is exact for DSSYK's standard matter
content (real positive weights and their strange partners) and becomes the
\emph{transition} of Theorem~\ref{thm:transition} on the full family. Theorem
\ref{thm:verdict}(iii) is refined accordingly.
(2) \emph{The transition locus is the $\Pi$-fixed locus, and its $\theta$-image is the
band centre} --- the locus of the centre-dS identification
(Definition~\ref{def:ds}(b)). Under edge-dS, edge-connectivity switches on exactly at the
weight whose spectral image is the rival proposal's ``dS point''. We record this
structural coincidence linking the two dS proposals without interpreting it.
(3) \emph{What remains outside this family}: bilocals transforming in $\pi_{b,\epsilon}$
as genuine non-spherical corepresentation intertwiners --- the operators SI describe
below their (4.14) as spanned by $\Lambda^{-1}(\delta^{\epsilon,(st)})$-elements, whose
action they found ``impractical'' to compute --- are not powers of the spherical element
and are not covered; issue \#13 remains open for them.
\end{remark}

\begin{remark}[what remains open after this section]\label{rem:criterionopen}
The verdict is unconditional on the radial side but leaves three loads on other issues:
the selectivity of C-amp beyond the spherical-power family
(issue \#13; partially resolved by Theorem~\ref{thm:transition}, issue \#27); the rescaled
Plancherel theory (issue \#14) for any statement summing over strange labels; and the
extension selection (issue \#26), which affects only \emph{which} locked tower pair is
realised, not the verdict (every extension is obstructed the same way, and the conjectured
canonical extension is itself $\Pi$-invariant). The centre-dS quantitative comparison
deferred from Remark~\ref{rem:dsdependence} is subsumed by
Theorem~\ref{thm:verdict}(iii) and Theorem~\ref{thm:transition}.
\end{remark}

\begin{proposition}[density of states]\label{prop:dos}
\DERIVED\ (checks (L1).) The chord-sector Plancherel density [SI (4.11);
Proposition~\ref{prop:planchereldata}] factorises \emph{exactly} at every $q$ through the
$q$-Gamma function: at $\theta=\pi-\lambda_2\tilde k$,
\begin{equation}\label{eq:dosfactorised}
  \rho_{q}(\theta)\coloneqq\frac{(q^{2};q^{2})_{\infty}}{2\pi}
  \bigl|(e^{2i\theta};q^{2})_{\infty}\bigr|^{2}
  =\frac{(q^{2};q^{2})_{\infty}^{3}(1-q^{2})^{2}}{2\pi}\,
  \frac{1}{|\Gamma_{q^{2}}(2i\tilde k)|^{2}}
  \;\xrightarrow[q\to1]{}\;
  \frac{(\cdots)}{2\pi}\cdot\frac{2\tilde k\sinh(2\pi\tilde k)}{\pi},
\end{equation}
the Schwarzian density of states, by $\Gamma_{q^{2}}\to\Gamma$ and
$|\Gamma(2i\tilde k)|^{2}=\pi/(2\tilde k\sinh2\pi\tilde k)$. The factorisation is verified
to $10^{-37}$ relative; the limit with relative deviations
$0.054/0.0056/0.00056$ at $q=0.9/0.99/0.999$ ($\tilde k=0.6$) and $0.47/0.057/0.0058$
($\tilde k=1.4$), measured rate ratios $0.78$--$0.99$ against the predicted $O(\lambda_2)$
scaling.
\end{proposition}

\begin{proposition}[wavefunctions: Bessel $K$ from the connection formula]\label{prop:besselK}
\DERIVED\ (checks (L2); derivation from this project's own closed forms.) Let
$\tilde s>0$, $s=q^{2i\tilde s}$ (principal value $\mu(s)=\cos(\tilde s\lambda_2)$), and
$u=2q^{k+1}/\lambda_2$ (the renormalised-length variable, $u=e^{-\tilde\ell/2}$ with
$\tilde\ell=\lambda_2 k-2\ln(\lambda_2/2)+O(\lambda_2)$, cf.\ the length dictionary of
Proposition~\ref{prop:where}). Then, as $q\to1$ at fixed $(\tilde s,u)$:
\begin{enumerate}[label=(\alph*)]
\item the minimal-solution kernel of Theorem~\ref{thm:strangeradial} confluences to the
modified Bessel function,
\begin{equation}\label{eq:Iconfluence}
  {}_{2}\varphi_{1}\!\left(\begin{matrix}0,0\\ s^{2}q^{2}\end{matrix};q^{2},
  \tfrac{\lambda_2^{2}u^{2}}{4}\right)\longrightarrow
  \Gamma(1+2i\tilde s)\,(u/2)^{-2i\tilde s}\,I_{2i\tilde s}(u),
\end{equation}
(measured relative deviations $6.6\times10^{-3}/5.8\times10^{-4}/5.7\times10^{-5}$ at
$q=0.9/0.99/0.999$, $u=0.4$; rates $1.10$, $1.01$ vs.\ predicted $O(\lambda_2)$);
\item combining the two branches through the connection formula \eqref{eq:connection} and
$\Gamma(1+\nu)\Gamma(-\nu)=-\pi/\sin\pi\nu$,
$K_{\nu}=\tfrac{\pi}{2\sin\pi\nu}(I_{-\nu}-I_{\nu})$:
\begin{equation}\label{eq:HtoK}
  H_{k}\bigl(\mu(q^{2i\tilde s});q^{2}\bigr)\;\longrightarrow\;
  \frac{2\,K_{2i\tilde s}(u)}{\lambda_2\,(q^{2};q^{2})_{\infty}},
\end{equation}
i.e.\ the transfer-matrix eigenvectors reduce to the Liouville/Schwarzian wavefunctions
$K_{2is}(e^{-\tilde\ell/2})$ of SI (2.27) at the upper edge, and times $(-1)^{k}$
(Lemma~\ref{lem:gradingchord}) at the lower/AdS edge. Measured relative deviations
$0.16/0.014/0.0014$ ($u_{0}=0.4$, oscillatory region, $k$ up to $7819$) and
$0.089/0.0081/0.0008$ ($u_{0}=1.5$), rates $1.01$--$1.06$ vs.\ predicted $O(\lambda_2)$.
\end{enumerate}
\end{proposition}
